4//! Inteligencia Artificial heurística basada en reglas para UNO.
5//!
6//! Prioridad de jugada (`Medium` / `Hard` / `Expert`):
7//! 1. `Wild Draw Four` (`CardValue::WildDrawFour`) si tiene ≤ 2 cartas en la mano y es legal.
8//! 2. Carta que cambia el color al color propio más frecuente en la mano (`most_frequent_color`).
9//! 3. `+2` (`CardValue::DrawTwo`) cuando algún oponente va ganando (tiene pocas cartas).
10//! 4. Carta que coincide con el color activo o el número/valor de la cima del descarte.
11//!
12//! En `AiDifficulty::Easy`: juega la primera carta válida de su mano sin considerar estrategia.
13

22/// Trait de acceso para permitir llamar `best_move(&uno_state, ...)` o `best_move(&session, ...)`.
23pub trait UnoStateAccess {
24    fn as_uno_state(&self) -> Option<&UnoState>;
25}
26
27impl UnoStateAccess for UnoState {
28    fn as_uno_state(&self) -> Option<&UnoState> {
29        Some(self)
30    }
31}
32
33impl UnoStateAccess for Session {
34    fn as_uno_state(&self) -> Option<&UnoState> {
35        self.world().get_resource::<UnoState>()
36    }
37}

39/// Comando devuelto por la IA de UNO.
40#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
41pub enum UnoCommand {
42    PlayCard(PlayCardCommand),
43    DrawCard(DrawCardCommand),
44    Intercept(InterceptCommand),
45    CallUno(CallUnoCommand),
46}
47
48impl UnoCommand {
49    /// Convierte el comando de IA en un `Box<dyn Command>` ejecutable por `Session`.
50    pub fn into_command(self) -> Box<dyn Command> {
51        match self {
52            Self::PlayCard(cmd) => Box::new(cmd),
53            Self::DrawCard(cmd) => Box::new(cmd),
54            Self::Intercept(cmd) => Box::new(cmd),
55            Self::CallUno(cmd) => Box::new(cmd),
56        }
57    }
58
59    pub fn card_index(&self) -> Option<usize> {
60        match self {
61            Self::PlayCard(cmd) => Some(cmd.card_index),
62            Self::Intercept(cmd) => Some(cmd.card_index),
63            Self::DrawCard(_) | Self::CallUno(_) => None,
64        }
65    }
66
67    pub fn chosen_color(&self) -> Option<CardColor> {
68        match self {
69            Self::PlayCard(cmd) => cmd.chosen_color,
70            _ => None,
71        }
72    }
73}

75/// Calcula el color estándar (`Red`, `Yellow`, `Green`, `Blue`) más frecuente en la mano.
76pub fn most_frequent_color(hand: &[Card]) -> CardColor {
77    let candidates = [
78        CardColor::Red,
79        CardColor::Yellow,
80        CardColor::Green,
81        CardColor::Blue,
82    ];
83    let mut best_color = CardColor::Red;
84    let mut best_count = 0usize;
85
86    for &color in &candidates {
87        let count = hand.iter().filter(|c| c.color == color).count();
88        if count > best_count {
89            best_count = count;
90            best_color = color;
91        }
92    }
93    best_color
94}

96/// Comprueba si `card` es legalmente jugable por `ai` considerando la restricción de `WildDrawFour`.
97fn is_legally_playable(state: &UnoState, hand: &[Card], card: &Card) -> bool {
98    if !state.is_playable(card) {
99        return false;
100    }
101    if card.value == CardValue::WildDrawFour {
102        let has_active_color = hand.iter().any(|c| c.color == state.active_color);
103        if has_active_color {
104            return false;
105        }
106    }
107    true
108}

110fn make_play_command(
111    ai: PlayerId,
112    card_index: usize,
113    card: &Card,
114    preferred_color: CardColor,
115) -> UnoCommand {
116    let chosen_color = if card.color == CardColor::WildColor {
117        Some(preferred_color)
118    } else {
119        None
120    };
121    UnoCommand::PlayCard(PlayCardCommand {
122        player: ai,
123        card_index,
124        chosen_color,
125    })
126}

128/// Selecciona la mejor jugada para `ai` en UNO según `difficulty`:
129/// - `Easy`: primera carta válida en la mano sin estrategia.
130/// - `Medium` / `Hard` / `Expert`:
131///   1) `Wild Draw Four` si ≤ 2 cartas en mano,
132///   2) Carta que cambia el color al color propio más frecuente,
133///   3) `+2` al jugador que va ganando,
134///   4) Carta que coincide con color/número.
135pub fn best_move<S: UnoStateAccess>(
136    state: &S,
137    ai: PlayerId,
138    difficulty: AiDifficulty,
139) -> Option<UnoCommand> {
140    best_move_with_randomness(state, ai, difficulty, 0, 0)
141}

143/// Selecciona la mejor jugada para `ai` en UNO con factor de aleatoriedad (`randomness_pct`, `0..=100`).
144pub fn best_move_with_randomness<S: UnoStateAccess>(
145    state: &S,
146    ai: PlayerId,
147    difficulty: AiDifficulty,
148    randomness_pct: u8,
149    seed: u64,
150) -> Option<UnoCommand> {
151    let uno = state.as_uno_state()?;
152    let hand = uno.hands.get(&ai)?;
153
154    if uno.draw_penalty > 0 {
155        return Some(UnoCommand::DrawCard(DrawCardCommand { player: ai }));
156    }
157
158    if hand.len() == 1 && !uno.uno_called.contains(&ai) {
159        let forgets = if difficulty == AiDifficulty::Easy {
160            (uno.discard_pile.len() * 37 + ai.0 as usize * 17) % 10 < 3
161        } else {
162            false
163        };
164        if !forgets {
165            return Some(UnoCommand::CallUno(CallUnoCommand { player: ai }));
166        }
167    }
168
169    let playable: Vec<(usize, Card)> = hand
170        .iter()
171        .copied()
172        .enumerate()
173        .filter(|(_, card)| is_legally_playable(uno, hand, card))
174        .collect();
175
176    if playable.is_empty() {
177        return Some(UnoCommand::DrawCard(DrawCardCommand { player: ai }));
178    }
179
180    let dominant_color = most_frequent_color(hand);
181
182    if difficulty == AiDifficulty::Easy && randomness_pct == 0 {
183        let (idx, card) = playable[0];
184        let fallback_color = if uno.active_color != CardColor::WildColor {
185            uno.active_color
186        } else {
187            CardColor::Red
188        };
189        return Some(make_play_command(ai, idx, &card, fallback_color));
190    }
191
192    let rival_leading = uno
193        .hands
194        .iter()
195        .any(|(&p, h)| p != ai && (h.len() <= hand.len() || h.len() <= 3));
196
197    if randomness_pct == 0 {
198        if hand.len() <= 2 {
199            if let Some(&(idx, card)) = playable
200                .iter()
201                .find(|(_, c)| c.value == CardValue::WildDrawFour)
202            {
203                return Some(make_play_command(ai, idx, &card, dominant_color));
204            }
205        }
206
207        if dominant_color != uno.active_color {
208            if let Some(&(idx, card)) = playable.iter().find(|(_, c)| c.color == dominant_color) {
209                return Some(make_play_command(ai, idx, &card, dominant_color));
210            }
211            if let Some(&(idx, card)) = playable.iter().find(|(_, c)| c.value == CardValue::Wild) {
212                return Some(make_play_command(ai, idx, &card, dominant_color));
213            }
214        }
215
216        if rival_leading {
217            if let Some(&(idx, card)) = playable.iter().find(|(_, c)| c.value == CardValue::DrawTwo)
218            {
219                return Some(make_play_command(ai, idx, &card, dominant_color));
220            }
221        }
222
223        if let Some(&(idx, card)) = playable
224            .iter()
225            .find(|(_, c)| c.color != CardColor::WildColor)
226        {
227            return Some(make_play_command(ai, idx, &card, dominant_color));
228        }
229
230        let (idx, card) = playable[0];
231        return Some(make_play_command(ai, idx, &card, dominant_color));
232    }
233
234    let scores: Vec<f32> = playable
235        .iter()
236        .map(|(_, c)| {
237            if c.value == CardValue::WildDrawFour && hand.len() <= 2 {
238                100.0
239            } else if dominant_color != uno.active_color && c.color == dominant_color {
240                85.0
241            } else if dominant_color != uno.active_color && c.value == CardValue::Wild {
242                80.0
243            } else if rival_leading && c.value == CardValue::DrawTwo {
244                75.0
245            } else if c.color != CardColor::WildColor {
246                50.0
247            } else {
248                25.0
249            }
250        })
251        .collect();
252
253    let safety_band = match difficulty {
254        AiDifficulty::Easy => 60.0,
255        AiDifficulty::Medium => 30.0,
256        AiDifficulty::Hard => 15.0,
257        AiDifficulty::Expert => 8.0,
258    };
259    let pick = AiDifficulty::select_top_k_with_noise(
260        &scores,
261        randomness_pct,
262        safety_band,
263        seed ^ (uno.discard_pile.len() as u64),
264    )
265    .unwrap_or(0);
266    let (idx, card) = playable[pick];
267    Some(make_play_command(ai, idx, &card, dominant_color))
268}

275    const P1: PlayerId = PlayerId(1);
276    const P2: PlayerId = PlayerId(2);

278    #[test]
279    fn uno_ai_changes_color_strategically() {
280        let mut session = new_uno_session(SessionId(1), 42, vec![P1, P2]);
281
282        {
283            let mut state = session.world_mut().resource_mut::<UnoState>();
284            // Color activo actual: Rojo (carta superior: 5 Rojo)
285            state.active_color = CardColor::Red;
286            state.discard_pile = vec![Card::new(CardColor::Red, CardValue::Number(5))];
287            state.draw_penalty = 0;
288
289            // Mano de P1:
290            // - Índice 0: 2 Rojo (mantiene Rojo)
291            // - Índice 1: 5 Azul (coincide por número 5 y cambia el color a Azul, que es el color más frecuente en la mano)
292            // - Índice 2: 7 Azul
293            // - Índice 3: 9 Azul
294            state.hands.insert(
295                P1,
296                vec![
297                    Card::new(CardColor::Red, CardValue::Number(2)),
298                    Card::new(CardColor::Blue, CardValue::Number(5)),
299                    Card::new(CardColor::Blue, CardValue::Number(7)),
300                    Card::new(CardColor::Blue, CardValue::Number(9)),
301                ],
302            );
303        }
304
305        // En Hard/Medium, la IA detecta que Azul es su color más frecuente (3 cartas vs 1 Roja)
306        // y juega la carta en índice 1 (5 Azul) para cambiar estratégicamente el color activo a Azul.
307        let hard_cmd = best_move(&session, P1, AiDifficulty::Hard).unwrap();
308        assert_eq!(
309            hard_cmd,
310            UnoCommand::PlayCard(PlayCardCommand {
311                player: P1,
312                card_index: 1,
313                chosen_color: None,
314            })
315        );
316
317        // En Easy, juega la primera carta válida sin considerar estrategia (índice 0: 2 Rojo)
318        let easy_cmd = best_move(&session, P1, AiDifficulty::Easy).unwrap();
319        assert_eq!(
320            easy_cmd,
321            UnoCommand::PlayCard(PlayCardCommand {
322                player: P1,
323                card_index: 0,
324                chosen_color: None,
325            })
326        );
327    }

329    #[test]
330    fn uno_ai_calls_uno_when_left_with_one_card() {
331        let mut session = new_uno_session(SessionId(1), 42, vec![P1, P2]);
332        {
333            let mut state = session.world_mut().resource_mut::<UnoState>();
334            state.active_color = CardColor::Red;
335            state.discard_pile = vec![Card::new(CardColor::Red, CardValue::Number(5))];
336            // P1 tiene exactamente 1 carta y no ha cantado UNO
337            state
338                .hands
339                .insert(P1, vec![Card::new(CardColor::Red, CardValue::Number(2))]);
340            state.uno_called.clear();
341        }
342
343        // En Hard, debe cantar UNO inmediatamente
344        let cmd = best_move(&session, P1, AiDifficulty::Hard).unwrap();
345        assert_eq!(cmd, UnoCommand::CallUno(CallUnoCommand { player: P1 }));
346
347        // Tras cantar UNO, vuelve a emitir la jugada de la carta
348        {
349            let mut state = session.world_mut().resource_mut::<UnoState>();
350            state.uno_called.insert(P1);
351        }
352        let play_cmd = best_move(&session, P1, AiDifficulty::Hard).unwrap();
353        assert_eq!(
354            play_cmd,
355            UnoCommand::PlayCard(PlayCardCommand {
356                player: P1,
357                card_index: 0,
358                chosen_color: None,
359            })
360        );
361    }
