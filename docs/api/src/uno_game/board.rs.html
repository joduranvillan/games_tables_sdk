4use open_table_engine::prelude::*;
5use open_table_macros::GameResource;
6use serde::{Deserialize, Serialize};
7use std::collections::{HashMap, HashSet};
8
9#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
10pub enum CardColor {
11    Red,
12    Yellow,
13    Green,
14    Blue,
15    WildColor,
16}
17
18#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
19pub enum CardValue {
20    Number(u8),
21    Skip,
22    Reverse,
23    DrawTwo,
24    Wild,
25    WildDrawFour,
26}
27
28#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
29pub struct Card {
30    pub color: CardColor,
31    pub value: CardValue,
32}
33
34impl Card {
35    pub fn new(color: CardColor, value: CardValue) -> Self {
36        Self { color, value }
37    }
38}
39
40#[derive(GameResource)]
41pub struct UnoState {
42    pub deck: Vec<Card>,
43    pub discard_pile: Vec<Card>,
44    pub hands: HashMap<PlayerId, Vec<Card>>,
45    pub active_color: CardColor,
46    pub draw_penalty: usize,
47    pub wild_draw_four_illegal: bool,
48    pub uno_called: HashSet<PlayerId>,
49    pub last_wild_draw_four_player: Option<PlayerId>,
50    pub allow_bluff_wild_draw_four: bool,
51}
52
53impl UnoState {
54    pub fn new(players: &[PlayerId], rng: &mut open_table_core::random::RandomService) -> Self {
55        let mut deck = Vec::new();
56        let colors = [
57            CardColor::Red,
58            CardColor::Yellow,
59            CardColor::Green,
60            CardColor::Blue,
61        ];
62
63        for &c in &colors {
64            deck.push(Card::new(c, CardValue::Number(0)));
65            for v in 1..=9 {
66                deck.push(Card::new(c, CardValue::Number(v)));
67                deck.push(Card::new(c, CardValue::Number(v)));
68            }
69            for _ in 0..2 {
70                deck.push(Card::new(c, CardValue::Skip));
71                deck.push(Card::new(c, CardValue::Reverse));
72                deck.push(Card::new(c, CardValue::DrawTwo));
73            }
74        }
75
76        for _ in 0..4 {
77            deck.push(Card::new(CardColor::WildColor, CardValue::Wild));
78            deck.push(Card::new(CardColor::WildColor, CardValue::WildDrawFour));
79        }
80
81        rng.shuffle(&mut deck);
82
83        let mut hands = HashMap::new();
84        for &p in players {
85            let mut hand = Vec::new();
86            for _ in 0..7 {
87                hand.push(deck.pop().unwrap());
88            }
89            hands.insert(p, hand);
90        }
91
92        let mut discard_pile = Vec::new();
93        // Make sure the first card is not a wild or action card for simplicity
94        let mut first_card_idx = 0;
95        for (i, card) in deck.iter().enumerate() {
96            if let CardValue::Number(_) = card.value {
97                first_card_idx = i;
98                break;
99            }
100        }
101        let first_card = deck.remove(first_card_idx);
102        let active_color = first_card.color;
103        discard_pile.push(first_card);
104
105        Self {
106            deck,
107            discard_pile,
108            hands,
109            active_color,
110            draw_penalty: 0,
111            wild_draw_four_illegal: false,
112            uno_called: HashSet::new(),
113            last_wild_draw_four_player: None,
114            allow_bluff_wild_draw_four: false,
115        }
116    }
117
118    pub fn draw_cards(
119        &mut self,
120        player: PlayerId,
121        count: usize,
122        rng: &mut open_table_core::random::RandomService,
123    ) -> Vec<Card> {
124        let mut drawn = Vec::new();
125        for _ in 0..count {
126            if self.deck.is_empty() {
127                if self.discard_pile.len() > 1 {
128                    let top = self.discard_pile.pop().unwrap();
129                    self.deck.append(&mut self.discard_pile);
130                    rng.shuffle(&mut self.deck);
131                    self.discard_pile.push(top);
132                } else {
133                    break; // No cards left anywhere
134                }
135            }
136            if let Some(c) = self.deck.pop() {
137                drawn.push(c);
138            }
139        }
140
141        if let Some(hand) = self.hands.get_mut(&player) {
142            hand.extend(drawn.clone());
143        }
144
145        drawn
146    }
147
148    pub fn is_playable(&self, card: &Card) -> bool {
149        if self.draw_penalty > 0 {
150            if let Some(top) = self.discard_pile.last() {
151                if top.value == CardValue::DrawTwo {
152                    return card.value == CardValue::DrawTwo
153                        || card.value == CardValue::WildDrawFour;
154                } else if top.value == CardValue::WildDrawFour {
155                    return card.value == CardValue::WildDrawFour;
156                }
157            }
158            return false;
159        }
160
161        if card.color == CardColor::WildColor {
162            return true;
163        }
164        if card.color == self.active_color {
165            return true;
166        }
167        if let Some(top) = self.discard_pile.last() {
168            if card.value == top.value {
169                return true;
170            }
171        }
172        false
173    }
174}
