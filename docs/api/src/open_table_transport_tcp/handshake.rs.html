1//! Protocolo de handshake, intercambiado ANTES de que una conexión empiece
2//! a mandar `CommandEnvelope`s normales.
3//!
4//! El problema que resuelve: cuando un cliente se cae y vuelve a
5//! conectarse, ¿cómo sabe el host que es el mismo jugador de antes y no
6//! alguien pretendiendo serlo? Un jugador que simplemente "dice" su
7//! `PlayerId` al reconectar tiene el mismo problema que ya resolvimos
8//! para los comandos (`issuer` vs `sender`): es trivial de falsificar.
9//!
10//! La solución acá es la misma idea aplicada a la conexión en vez de al
11//! comando: el host entrega un `reconnect_token` al aceptar un `Join`, y
12//! exige ese token de vuelta en un `Reconnect`. El cliente lo guarda
13//! localmente (fuera del alcance de este crate — en Godot sería
14//! `user://`, por ejemplo).
15
16use serde::{Deserialize, Serialize};
17
18use open_table_core::ids::PlayerId;
19
20pub const PROTOCOL_VERSION: &str = "1.0";
21
22fn default_protocol_version() -> String {
23    PROTOCOL_VERSION.to_string()
24}
25
26#[derive(Debug, Clone, Serialize, Deserialize)]
27pub enum HandshakeRequest {
28    /// Un jugador nuevo entrando a la sala por primera vez.
29    Join {
30        player_name: String,
31        #[serde(default = "default_protocol_version")]
32        protocol_version: String,
33        #[serde(default)]
34        route: Option<String>,
35    },
36    /// Un jugador que ya estaba, probando que es quien dice ser.
37    Reconnect {
38        player_id: PlayerId,
39        reconnect_token: String,
40        #[serde(default = "default_protocol_version")]
41        protocol_version: String,
42        #[serde(default)]
43        route: Option<String>,
44    },
45}
46
47impl HandshakeRequest {
48    pub fn protocol_version(&self) -> &str {
49        match self {
50            HandshakeRequest::Join {
51                protocol_version, ..
52            } => protocol_version,
53            HandshakeRequest::Reconnect {
54                protocol_version, ..
55            } => protocol_version,
56        }
57    }
58
59    pub fn route(&self) -> Option<&str> {
60        match self {
61            HandshakeRequest::Join { route, .. } => route.as_deref(),
62            HandshakeRequest::Reconnect { route, .. } => route.as_deref(),
63        }
64    }
65}
66
67#[derive(Debug, Clone, Serialize, Deserialize)]
68pub enum HandshakeResponse {
69    Welcome {
70        player_id: PlayerId,
71        reconnect_token: String,
72        #[serde(default = "default_protocol_version")]
73        protocol_version: String,
74    },
75    Rejected {
76        reason: String,
77    },
78    UnsupportedProtocolVersion {
79        expected: String,
80        got: String,
81    },
82}
