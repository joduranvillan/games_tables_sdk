18pub fn update_elo(winner: EloRating, loser: EloRating, k: f64) -> (EloRating, EloRating) {
19    let e_winner = 1.0 / (1.0 + 10.0_f64.powf((loser.0 - winner.0) / 400.0));
20    let e_loser = 1.0 / (1.0 + 10.0_f64.powf((winner.0 - loser.0) / 400.0));
21
22    let new_winner = winner.0 + k * (1.0 - e_winner);
23    let new_loser = loser.0 + k * (0.0 - e_loser);
24
25    (EloRating(new_winner), EloRating(new_loser))
26}

28pub fn update_elo_draw(a: EloRating, b: EloRating, k: f64) -> (EloRating, EloRating) {
29    let e_a = 1.0 / (1.0 + 10.0_f64.powf((b.0 - a.0) / 400.0));
30    let e_b = 1.0 / (1.0 + 10.0_f64.powf((a.0 - b.0) / 400.0));
31
32    let new_a = a.0 + k * (0.5 - e_a);
33    let new_b = b.0 + k * (0.5 - e_b);
34
35    (EloRating(new_a), EloRating(new_b))
36}

42    #[test]
43    fn test_victoria_sube_elo_del_ganador() {
44        let winner = EloRating(1200.0);
45        let loser = EloRating(1200.0);
46        let (new_w, new_l) = update_elo(winner, loser, 32.0);
47        assert!(new_w.0 > winner.0, "El ELO del ganador debe subir");
48        assert!(new_l.0 < loser.0, "El ELO del perdedor debe bajar");
49        assert_eq!(new_w.0, 1216.0);
50        assert_eq!(new_l.0, 1184.0);
