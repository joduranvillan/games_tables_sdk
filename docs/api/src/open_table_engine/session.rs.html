4//! `Session`: el contenedor de una partida en curso.
5//!
6//! Tres invariantes de seguridad/consistencia que este archivo hace
7//! cumplir:
8//! 1. `process_command` es el único método que muta el `World` (I3 del
9//!    RFC): no hay forma de tomar una referencia mutable al `World` desde
10//!    afuera.
11//! 2. El `sender` de un comando SIEMPRE viene de la capa de red (qué
12//!    conexión trajo estos bytes), nunca de un campo dentro del propio
13//!    comando. Si el comando declara un `issuer()` distinto del `sender`
14//!    real, se rechaza. Esto es lo que impide que un cliente mande
15//!    comandos "de parte de" otro jugador.
16//! 3. Cada comando aceptado se guarda en un log ordenado por tick. Ese
17//!    log alcanza para dos cosas sin necesitar serializar el `World`
18//!    entero: reproducir la partida completa (replay) y, filtrando los
19//!    eventos resultantes por `Visibility`, reconstruir lo que un cliente
20//!    que se reconectó se perdió (resync).
21

154struct TypedResourceWatcher<R: Resource + Clone + 'static, F: Fn(&R) + Send + Sync + 'static> {
155    handler: F,
156    _phantom: std::marker::PhantomData<R>,
157}
158
159impl<R: Resource + Clone + 'static, F: Fn(&R) + Send + Sync + 'static> ResourceWatcher
160    for TypedResourceWatcher<R, F>
161{
162    fn check_and_notify(&mut self, world: &BevyWorld) {
163        if world.is_resource_changed::<R>() {
164            if let Some(res) = world.get_resource::<R>() {
165                (self.handler)(res);
166            }
167        }
168    }
169}

215    ///
216    /// # Examples
217    /// ```
218    /// use open_table_engine::prelude::*;
219    /// let (delta_w, delta_l) = Session::compute_elo_delta(PlayerId(1), PlayerId(2), 32.0);
220    /// assert!(delta_w > 0.0 && delta_l < 0.0);
221    /// ```
222    pub fn compute_elo_delta(_winner: PlayerId, _loser: PlayerId, k: f64) -> (f64, f64) {
223        let (new_w, new_l) = crate::elo::update_elo(
224            crate::elo::EloRating(1200.0),
225            crate::elo::EloRating(1200.0),
226            k,
227        );
228        (new_w.0 - 1200.0, new_l.0 - 1200.0)
229    }

339    pub fn export_chess_pgn(&self) -> Option<String> {
340        let mut pgn = String::from("[Event \"Casual Game\"]\n[Site \"OpenTable\"]\n[Date \"????.??.??\"]\n[Round \"1\"]\n[White \"White\"]\n[Black \"Black\"]\n[Result \"*\"]\n\n");
341        let mut move_strs = Vec::new();
342        let mut move_num = 1;
343
344        let mut piece_positions = std::collections::HashMap::new();
345        for r in 0..8u8 {
346            for c in 0..8u8 {
347                let p = match (r, c) {
348                    (0, 0) | (0, 7) => Some(('R', true)),
349                    (0, 1) | (0, 6) => Some(('N', true)),
350                    (0, 2) | (0, 5) => Some(('B', true)),
351                    (0, 3) => Some(('Q', true)),
352                    (0, 4) => Some(('K', true)),
353                    (1, _) => Some(('P', true)),
354                    (7, 0) | (7, 7) => Some(('R', false)),
355                    (7, 1) | (7, 6) => Some(('N', false)),
356                    (7, 2) | (7, 5) => Some(('B', false)),
357                    (7, 3) => Some(('Q', false)),
358                    (7, 4) => Some(('K', false)),
359                    (6, _) => Some(('P', false)),
360                    _ => None,
361                };
362                if let Some((kind, is_white)) = p {
363                    piece_positions.insert((r, c), (kind, is_white));
364                }
365            }
366        }
367
368        let mut is_white_turn = true;
369
370        for (_tick, _elapsed, cmd) in &self.command_log {
371            if let Ok(val) = serde_json::to_value(cmd.as_ref()) {
372                let ct = val
373                    .get("command_type")
374                    .and_then(|v| v.as_str())
375                    .unwrap_or("");
376                if ct.contains("MovePieceCommand") {
377                    let from_r = val
378                        .get("from")
379                        .and_then(|f| f.get("row"))
380                        .and_then(|r| r.as_u64())
381                        .unwrap_or(0) as u8;
382                    let from_c = val
383                        .get("from")
384                        .and_then(|f| f.get("col"))
385                        .and_then(|c| c.as_u64())
386                        .unwrap_or(0) as u8;
387                    let to_r = val
388                        .get("to")
389                        .and_then(|t| t.get("row"))
390                        .and_then(|r| r.as_u64())
391                        .unwrap_or(0) as u8;
392                    let to_c = val
393                        .get("to")
394                        .and_then(|t| t.get("col"))
395                        .and_then(|c| c.as_u64())
396                        .unwrap_or(0) as u8;
397
398                    let piece_info = piece_positions.remove(&(from_r, from_c));
399                    let (kind, _is_white) = piece_info.unwrap_or(('P', is_white_turn));
400                    let captured = piece_positions.remove(&(to_r, to_c));
401
402                    let from_c_char = (b'a' + from_c) as char;
403                    let to_c_char = (b'a' + to_c) as char;
404                    let to_r_char = (b'1' + to_r) as char;
405
406                    let san_move = if kind == 'K' && from_c == 4 && to_c == 6 {
407                        "O-O".to_string()
408                    } else if kind == 'K' && from_c == 4 && to_c == 2 {
409                        "O-O-O".to_string()
410                    } else if kind == 'P' {
411                        if captured.is_some() || from_c != to_c {
412                            format!("{from_c_char}x{to_c_char}{to_r_char}")
413                        } else {
414                            format!("{to_c_char}{to_r_char}")
415                        }
416                    } else {
417                        if captured.is_some() {
418                            format!("{kind}x{to_c_char}{to_r_char}")
419                        } else {
420                            format!("{kind}{to_c_char}{to_r_char}")
421                        }
422                    };
423
424                    let promote_char = match val.get("promote_to").and_then(|p| p.as_str()) {
425                        Some("Queen") => "=Q",
426                        Some("Rook") => "=R",
427                        Some("Bishop") => "=B",
428                        Some("Knight") => "=N",
429                        _ => "",
430                    };
431                    let full_move = format!("{san_move}{promote_char}");
432
433                    let final_kind = match val.get("promote_to").and_then(|p| p.as_str()) {
434                        Some("Queen") => 'Q',
435                        Some("Rook") => 'R',
436                        Some("Bishop") => 'B',
437                        Some("Knight") => 'N',
438                        _ => kind,
439                    };
440                    piece_positions.insert((to_r, to_c), (final_kind, is_white_turn));
441
442                    if is_white_turn {
443                        move_strs.push(format!("{move_num}. {full_move}"));
444                    } else {
445                        if let Some(last) = move_strs.last_mut() {
446                            last.push_str(&format!(" {full_move}"));
447                        }
448                        move_num += 1;
449                    }
450                    is_white_turn = !is_white_turn;
451                }
452            }
453        }
454
455        pgn.push_str(&move_strs.join(" "));
456        Some(pgn)
457    }

507            self.world.insert_resource(EventHandlersResource::default());
508        }
509        let type_id = std::any::TypeId::of::<E>();
510        let wrapper = std::sync::Arc::new(move |event_dyn: &dyn Event| {
511            if let Some(concrete) = event_dyn.as_any().downcast_ref::<E>() {
512                handler(concrete);
513            }
514        });
515        let mut handlers = self.world.resource_mut::<EventHandlersResource>();
516        handlers.handlers.push((type_id, wrapper));
517    }

579    /// Único punto de entrada para mutar el estado de la partida.
580    ///
581    /// `sender` tiene que venir de `Transport::poll_received()` (la
582    /// identidad ligada a la conexión de red real), nunca de un campo
583    /// dentro de `command`. Si `command.issuer()` es `Some` y no coincide
584    /// con `sender`, se rechaza sin ejecutar nada — esto es lo que impide
585    /// que un cliente mande comandos en nombre de otro jugador.
586    ///
587    /// # Examples
588    /// ```
589    /// use open_table_engine::prelude::*;
590    /// let session = Session::new(SessionId(1), 42, vec![PlayerId(1), PlayerId(2)], RuleEngine::new());
591    /// assert_eq!(session.current_tick(), Tick(0));
592    /// ```
593    pub fn process_command(
594        &mut self,
595        sender: PlayerId,
596        command: Box<dyn Command>,
597    ) -> EngineResult<Vec<Box<dyn Event>>> {
598        if self.paused {
599            return Err(EngineError::SessionPaused);
600        }
601
602        if self
603            .world
604            .get_resource::<GameOverResource>()
605            .is_some_and(|r| r.0)
606            && !command.typetag_name().ends_with("RematchCommand")
607        {
608            return Err(EngineError::GameAlreadyOver);
609        }
610
611        if let Some(claimed_issuer) = command.issuer() {
612            if claimed_issuer != sender {
613                return Err(EngineError::Unauthorized(sender));
614            }
615        }
616
617        if let Some(mut rate_limit) = self.world.get_resource_mut::<RateLimitResource>() {
618            let max_cps = rate_limit.max_commands_per_second;
619            let now = Instant::now();
620            let history = rate_limit.history.entry(sender).or_default();
621            history.retain(|ts| now.duration_since(*ts) < Duration::from_secs(1));
622            if history.len() as u32 >= max_cps {
623                return Err(EngineError::RateLimitExceeded(sender));
624            }
625            history.push(now);
626        }
627
628        let envelope = CommandEnvelope {
629            tick: self.current_tick(),
630            payload: command.clone(),
631        };
632
633        for mw in &self.middlewares {
634            mw(&envelope, &mut self.world)?;
635        }
636
637        self.world.clear_trackers();
638        let events = self.rules.dispatch(command.as_ref(), &mut self.world)?;
639        let tick = self.clock.advance();
640        let elapsed = self.clock.elapsed_millis();
641        self.command_log.push((tick, elapsed, command));
642
643        for watcher in &mut self.watchers {
644            watcher.check_and_notify(&self.world);
645        }
646
647        if let Some(mut timer) = self.world.get_resource_mut::<TimerResource>() {
648            timer.reset();
649        }
650
651        if let Some(handlers_res) = self.world.get_resource::<EventHandlersResource>() {
652            for event in &events {
653                let event_type_id = event.as_any().type_id();
654                for (registered_type_id, handler) in &handlers_res.handlers {
655                    if *registered_type_id == event_type_id {
656                        handler(event.as_ref());
657                    }
658                }
659            }
660        }
661
662        Ok(events)
663    }

665    /// Comprueba si el temporizador de turno o el timeout de reconexión ha expirado.
666    pub fn check_timeout(&mut self) -> Option<PlayerId> {
667        if let Some(mut rec_res) = self.world.get_resource_mut::<ReconnectionTimeoutResource>() {
668            let now = Instant::now();
669            let timed_out_player =
670                rec_res
671                    .timeouts
672                    .iter()
673                    .find_map(|(&player, &(start, duration))| {
674                        if now.duration_since(start) >= duration {
675                            Some(player)
676                        } else {
677                            None
678                        }
679                    });
680
681            if let Some(player) = timed_out_player {
682                rec_res.timeouts.remove(&player);
683
684                let forfeit_event = ForfeitEvent { player };
685                if let Some(handlers_res) = self.world.get_resource::<EventHandlersResource>() {
686                    let event_type_id = std::any::TypeId::of::<ForfeitEvent>();
687                    for (registered_type_id, handler) in &handlers_res.handlers {
688                        if *registered_type_id == event_type_id {
689                            handler(&forfeit_event);
690                        }
691                    }
692                }
693
694                if let Some(mut turn) = self.world.get_resource_mut::<TurnResource>() {
695                    turn.0.advance();
696                }
697
698                return Some(player);
699            }
700        }
701
702        if let Some(timer) = self.world.get_resource::<TimerResource>() {
703            if timer.is_expired() {
704                return Some(self.current_player());
705            }
706        }
707        None
708    }

710    /// Revierte el último comando del log reconstruyendo el estado determinísticamente
711    /// desde un clon de `initial_state` + `history[..len-1]`.
712    /// Disponible solo si `SessionMode::Local`.
713    pub fn undo_last_command(&mut self) -> Result<(), EngineError> {
714        if self.mode != SessionMode::Local {
715            return Err(EngineError::InvalidCommand {
716                reason: "Undo solo disponible en modo Local".into(),
717            });
718        }
719
720        if self.command_log.is_empty() {
721            return Ok(());
722        }
723
724        let target_len = self.command_log.len() - 1;
725        let remaining_commands: Vec<_> = self
726            .command_log
727            .drain(..)
728            .take(target_len)
729            .map(|(_, _, cmd)| cmd)
730            .collect();
731
732        let init = self.initial_state.clone();
733        let legal_gen = self
734            .world
735            .get_resource::<LegalMoveGeneratorResource>()
736            .cloned();
737
738        let mut new_world = BevyWorld::new();
739        new_world.insert_resource(RandomResource(RandomService::from_seed(init.seed)));
740        new_world.insert_resource(TurnResource(TurnOrder::new(init.players)));
741        new_world.insert_resource(GameOverResource(false));
742        new_world.insert_resource(WinnerResource(None));
743        new_world.insert_resource(GameVariantResource(init.variant));
744        new_world.insert_resource(SessionMetadataResource(init.metadata));
745        if let Some(g) = legal_gen {
746            new_world.insert_resource(g);
747        }
748
749        if let Some(ref setup) = self.initial_setup {
750            setup(&mut new_world);
751        }
752
753        self.world = new_world;
754        self.clock = GameClock::new();
755
756        for command in remaining_commands {
757            let sender = command.issuer().unwrap_or_else(|| self.current_player());
758            self.process_command(sender, command)?;
759        }
760
761        Ok(())
762    }

890    /// Reproduce los comandos de `log_bytes` sobre una `Session` ya inicializada.
891    pub fn replay_on_session(
892        mut session: Session,
893        log_bytes: &[u8],
894    ) -> EngineResult<(Session, Vec<Box<dyn Event>>)> {
895        let log: Vec<LoggedCommand> = serde_json::from_slice(log_bytes)
896            .map_err(|e| EngineError::Replay(format!("no se pudo leer el log: {e}")))?;
897
898        let mut all_events = Vec::new();
899
900        for (_tick, _elapsed, command) in log {
901            let sender = command.issuer().unwrap_or_else(|| session.current_player());
902            let mut events = session.process_command(sender, command)?;
903            all_events.append(&mut events);
904        }
905
906        Ok((session, all_events))
907    }

917    /// Reproduce solo los comandos con tick mayor a `start_tick` desde `log_bytes`,

970    /// Restaura una `Session` directamente desde un snapshot guardado, sin reproducir los comandos.
971    pub fn from_snapshot(snapshot_bytes: &[u8], rules: RuleEngine) -> EngineResult<Session> {
972        let snapshot: SessionSnapshot = serde_json::from_slice(snapshot_bytes)
973            .map_err(|e| EngineError::Replay(format!("no se pudo leer el snapshot: {e}")))?;
974
975        let mut world = BevyWorld::new();
976        if let Some(tr) = snapshot.turn_resource {
977            world.insert_resource(tr);
978        }
979        if let Some(rr) = snapshot.random_resource {
980            world.insert_resource(rr);
981        }
982        if let Some(go) = snapshot.game_over_resource {
983            world.insert_resource(go);
984        }
985        if let Some(gv) = snapshot.game_variant_resource {
986            world.insert_resource(gv);
987        }
988        if let Some(sm) = snapshot.metadata_resource {
989            world.insert_resource(sm);
990        }
991
992        let seed = world
993            .get_resource::<RandomResource>()
994            .map(|r| r.0.seed())
995            .unwrap_or(0);
996        let players = world
997            .get_resource::<TurnResource>()
998            .map(|t| t.0.players().to_vec())
999            .unwrap_or_default();
1000        let variant = world
1001            .get_resource::<GameVariantResource>()
1002            .map(|v| v.0.clone())
1003            .unwrap_or(GameVariant::Standard);
1004        let metadata = world
1005            .get_resource::<SessionMetadataResource>()
1006            .map(|m| m.0.clone())
1007            .unwrap_or(SessionMetadata {
1008                game_name: "Unknown".into(),
1009                started_at: 0,
1010                max_players: players.len(),
1011            });
1012
1013        Ok(Session {
1014            id: snapshot.id,
1015            mode: SessionMode::Local,
1016            world,
1017            rules,
1018            clock: snapshot.clock,
1019            command_log: snapshot.command_log,
1020            initial_state: InitialSessionState {
1021                seed,
1022                players,
1023                variant,
1024                metadata,
1025            },
1026            initial_setup: None,
1027            paused: false,
1028            watchers: Vec::new(),
1029            middlewares: Vec::new(),
1030        })
1031    }

1081    /// Exporta el log de comandos serializado en JSON y firmado con HMAC-SHA256 usando `secret`.
1082    /// Añade los 32 bytes de la firma HMAC al final de los bytes del JSON.
1083    pub fn export_log_signed(&self, secret: &[u8]) -> EngineResult<Vec<u8>> {
1084        let json_bytes = self.export_log()?;
1085        let mut mac = HmacSha256::new_from_slice(secret)
1086            .map_err(|e| EngineError::Internal(format!("error al inicializar HMAC: {e}")))?;
1087        mac.update(&json_bytes);
1088        let signature = mac.finalize().into_bytes();
1089
1090        let mut signed_payload = Vec::with_capacity(json_bytes.len() + signature.len());
1091        signed_payload.extend_from_slice(&json_bytes);
1092        signed_payload.extend_from_slice(&signature);
1093        Ok(signed_payload)
1094    }

1096    /// Verifica la integridad del log firmado con HMAC-SHA256 antes de reproducirlo.
1097    pub fn verify_log(bytes: &[u8], secret: &[u8]) -> EngineResult<()> {
1098        if bytes.len() < 32 {
1099            return Err(EngineError::Replay(
1100                "log firmado inválido: tamaño insuficiente para incluir la firma HMAC".into(),
1101            ));
1102        }
1103
1104        let (json_bytes, signature_tag) = bytes.split_at(bytes.len() - 32);
1105
1106        let mut mac = HmacSha256::new_from_slice(secret)
1107            .map_err(|e| EngineError::Internal(format!("error al inicializar HMAC: {e}")))?;
1108        mac.update(json_bytes);
1109
1110        mac.verify_slice(signature_tag).map_err(|_| {
1111            EngineError::Replay(
1112                "firma HMAC inválida: el log ha sido alterado o la clave es incorrecta".into(),
1113            )
1114        })
1115    }
