11fn clone_checkers_session(session: &Session) -> Session {
12    let sides = session
13        .world()
14        .get_resource::<Sides>()
15        .copied()
16        .unwrap_or(Sides {
17            increasing: PlayerId(1),
18            decreasing: PlayerId(2),
19        });
20
21    let mut new_session =
22        crate::new_checkers_session(session.id, 0, sides.increasing, sides.decreasing);
23
24    if let Some(board) = session.world().get_resource::<BoardResource>() {
25        new_session.world_mut().insert_resource(board.clone());
26    }
27    if let Some(mc) = session.world().get_resource::<MultiCaptureResource>() {
28        new_session.world_mut().insert_resource(mc.clone());
29    }
30    if let Some(turn) = session.world().get_resource::<TurnResource>() {
31        new_session.world_mut().insert_resource(turn.clone());
32    }
33    if let Some(game_over) = session.world().get_resource::<GameOverResource>() {
34        new_session.world_mut().insert_resource(*game_over);
35    }
36    if let Some(winner) = session.world().get_resource::<WinnerResource>() {
37        new_session.world_mut().insert_resource(*winner);
38    }
39    new_session
40        .world_mut()
41        .insert_resource(LegalMoveGeneratorResource(std::sync::Arc::new(
42            CheckersLegalMoveGenerator,
43        )));
44
45    new_session
46}

48/// Selecciona una jugada legal para el jugador especificado en la sesión dada.
49///
50/// Si existen capturas legales, selecciona una al azar entre ellas.
51/// Si no hay capturas, selecciona al azar entre los movimientos simples legales.
52/// Devuelve `None` si la partida ha terminado, si no es el turno del jugador o
53/// si no hay movimientos legales disponibles.
54pub fn checkers_ai_pick_move(session: &Session, player: PlayerId) -> Option<MovePieceCommand> {
55    if session
56        .world()
57        .get_resource::<GameOverResource>()
58        .is_some_and(|r| r.0)
59    {
60        return None;
61    }
62
63    if session.current_player() != player {
64        return None;
65    }
66
67    let board = session.world().get_resource::<BoardResource>()?;
68
69    let mut captures = Vec::new();
70    let mut simple_moves = Vec::new();
71
72    for r in 0..8u8 {
73        for c in 0..8u8 {
74            let from = Position::new(r, c);
75            if let Some(piece) = board.get(from) {
76                if piece.owner != player {
77                    continue;
78                }
79
80                let offsets: &[(i8, i8)] = &[
81                    (-1, -1),
82                    (-1, 1),
83                    (1, -1),
84                    (1, 1),
85                    (-2, -2),
86                    (-2, 2),
87                    (2, -2),
88                    (2, 2),
89                ];
90
91                for &(dr, dc) in offsets {
92                    let nr = r as i8 + dr;
93                    let nc = c as i8 + dc;
94                    if (0..8).contains(&nr) && (0..8).contains(&nc) {
95                        let to = Position::new(nr as u8, nc as u8);
96                        if !to.is_dark_square() {
97                            continue;
98                        }
99
100                        let candidate_cmd = MovePieceCommand { player, from, to };
101                        let mut test_session = clone_checkers_session(session);
102
103                        if test_session
104                            .process_command(player, Box::new(candidate_cmd.clone()))
105                            .is_ok()
106                        {
107                            if dr.abs() == 2 {
108                                captures.push(candidate_cmd);
109                            } else {
110                                simple_moves.push(candidate_cmd);
111                            }
112                        }
113                    }
114                }
115            }
116        }
117    }
118
119    let candidate_pool = if !captures.is_empty() {
120        &captures
121    } else if !simple_moves.is_empty() {
122        &simple_moves
123    } else {
124        return None;
125    };
126
127    let idx = if let Some(rand_res) = session.world().get_resource::<RandomResource>() {
128        let mut rng = rand_res.0.clone();
129        (rng.next_u32() as usize) % candidate_pool.len()
130    } else {
131        (session.current_tick().0 as usize + candidate_pool.len()) % candidate_pool.len()
132    };
133
134    Some(candidate_pool[idx].clone())
135}

137/// Evalúa heurísticamente el tablero de Damas para `player` (material, damas coronadas, avance y centro).
138pub fn evaluate_checkers_board(session: &Session, player: PlayerId) -> f32 {
139    let Some(board) = session.world().get_resource::<BoardResource>() else {
140        return 0.0;
141    };
142    let sides = session
143        .world()
144        .get_resource::<Sides>()
145        .copied()
146        .unwrap_or(Sides {
147            increasing: PlayerId(1),
148            decreasing: PlayerId(2),
149        });
150    let mut score = 0.0f32;
151    for r in 0..8u8 {
152        for c in 0..8u8 {
153            if let Some(piece) = board.get(Position::new(r, c)) {
154                let base = if piece.is_king { 2.6 } else { 1.0 };
155                let adv = if piece.owner == sides.increasing {
156                    f32::from(r) * 0.05
157                } else {
158                    f32::from(7u8.saturating_sub(r)) * 0.05
159                };
160                let center = (3.5 - (f32::from(c) - 3.5).abs()).max(0.0) * 0.02;
161                let edge_safe = if c == 0 || c == 7 { 0.04 } else { 0.0 };
162                let total = base + adv + center + edge_safe;
163                if piece.owner == player {
164                    score += total;
165                } else {
166                    score -= total;
167                }
168            }
169        }
170    }
171    score
172}

174fn generate_checkers_moves(world: &BevyWorld, player: PlayerId) -> Vec<MovePieceCommand> {
175    let Some(board) = world.get_resource::<BoardResource>() else {
176        return vec![];
177    };
178    let Some(sides) = world.get_resource::<Sides>() else {
179        return vec![];
180    };
181
182    let mut candidates: Vec<MovePieceCommand> = Vec::new();
183    let captures = board.get_all_available_captures(sides, player);
184    let active_jumper = world
185        .get_resource::<MultiCaptureResource>()
186        .and_then(|m| m.active_piece);
187
188    if !captures.is_empty() {
189        for (from, to) in captures {
190            if let Some(req) = active_jumper {
191                if from != req {
192                    continue;
193                }
194            }
195            candidates.push(MovePieceCommand { player, from, to });
196        }
197    } else if active_jumper.is_none() {
198        let forward = sides.forward_direction(player);
199        for r in 0..8u8 {
200            for c in 0..8u8 {
201                let from = Position::new(r, c);
202                if let Some(piece) = board.get(from) {
203                    if piece.owner != player {
204                        continue;
205                    }
206                    let offsets: &[(i8, i8)] = if piece.is_king {
207                        &[(-1, -1), (-1, 1), (1, -1), (1, 1)]
208                    } else if forward == 1 {
209                        &[(1, -1), (1, 1)]
210                    } else {
211                        &[(-1, -1), (-1, 1)]
212                    };
213                    for &(dr, dc) in offsets {
214                        let nr = r as i8 + dr;
215                        let nc = c as i8 + dc;
216                        if (0..8).contains(&nr) && (0..8).contains(&nc) {
217                            let to = Position::new(nr as u8, nc as u8);
218                            if to.is_dark_square() && board.get(to).is_none() {
219                                candidates.push(MovePieceCommand { player, from, to });
220                            }
221                        }
222                    }
223                }
224            }
225        }
226    }
227
228    candidates
229}

231/// Selecciona la mejor jugada legal en Damas ajustando la evaluación según `AiDifficulty`.
232pub fn best_move(
233    session: &Session,
234    player: PlayerId,
235    difficulty: AiDifficulty,
236) -> Option<MovePieceCommand> {
237    best_move_with_randomness(session, player, difficulty, 0, 0)
238}

240/// Selecciona la mejor jugada legal en Damas con factor de aleatoriedad (`randomness_pct`, `0..=100`).
241pub fn best_move_with_randomness(
242    session: &Session,
243    player: PlayerId,
244    difficulty: AiDifficulty,
245    randomness_pct: u8,
246    seed: u64,
247) -> Option<MovePieceCommand> {
248    if difficulty == AiDifficulty::Easy && randomness_pct == 0 {
249        return checkers_ai_pick_move(session, player);
250    }
251    let legal = generate_checkers_moves(session.world(), player);
252    if legal.is_empty() {
253        return checkers_ai_pick_move(session, player);
254    }
255    let mut scored_cmds = Vec::with_capacity(legal.len());
256    let mut scores = Vec::with_capacity(legal.len());
257
258    for cmd in legal {
259        let mut next_session = clone_checkers_session(session);
260        if next_session
261            .process_command(player, Box::new(cmd.clone()))
262            .is_ok()
263        {
264            let mut score = evaluate_checkers_board(&next_session, player);
265            if matches!(difficulty, AiDifficulty::Hard | AiDifficulty::Expert) {
266                let opp = next_session.current_player();
267                if opp != player {
268                    let opp_moves = generate_checkers_moves(next_session.world(), opp);
269                    let mut worst_reply = score;
270                    for opp_cmd in opp_moves.into_iter().take(8) {
271                        let mut reply_sess = clone_checkers_session(&next_session);
272                        if reply_sess.process_command(opp, Box::new(opp_cmd)).is_ok() {
273                            let reply_eval = evaluate_checkers_board(&reply_sess, player);
274                            if reply_eval < worst_reply {
275                                worst_reply = reply_eval;
276                            }
277                        }
278                    }
279                    score = worst_reply;
280                }
281            }
282            scored_cmds.push(cmd);
283            scores.push(score);
284        }
285    }
286
287    let safety_band = match difficulty {
288        AiDifficulty::Easy => 1.5,
289        AiDifficulty::Medium => 0.65,
290        AiDifficulty::Hard => 0.30,
291        AiDifficulty::Expert => 0.15,
292    };
293    if let Some(idx) = AiDifficulty::select_top_k_with_noise(
294        &scores,
295        randomness_pct,
296        safety_band,
297        seed ^ session.current_tick().0 ^ session.id.0,
298    ) {
299        return scored_cmds.into_iter().nth(idx);
300    }
301    checkers_ai_pick_move(session, player)
302}

304#[cfg(test)]
305mod tests {
306    use super::*;
307    use crate::new_checkers_session;

309    const P1: PlayerId = PlayerId(1);
310    const P2: PlayerId = PlayerId(2);
311
312    #[test]
313    fn test_ai_always_returns_a_move_when_board_not_over() {
314        let mut session = new_checkers_session(SessionId(1), 42, P1, P2);
315
316        let is_over = session
317            .world()
318            .get_resource::<GameOverResource>()
319            .map_or(false, |r| r.0);
320        assert!(!is_over, "El tablero no debe estar terminado");
321
322        let current_player = session.current_player();
323        let mv = checkers_ai_pick_move(&session, current_player);
324
325        assert!(
326            mv.is_some(),
327            "La IA debe retornar un movimiento cuando la partida no ha terminado"
328        );
329
330        let chosen_move = mv.unwrap();
331        assert_eq!(chosen_move.player, current_player);
332
333        let res = session.process_command(current_player, Box::new(chosen_move));
334        assert!(
335            res.is_ok(),
336            "El movimiento retornado por la IA debe ser válido"
337        );
338    }
339
340    #[test]
341    fn test_ai_returns_none_when_game_over() {
342        let mut session = new_checkers_session(SessionId(1), 42, P1, P2);
343        session.world_mut().insert_resource(GameOverResource(true));
344
345        let mv = checkers_ai_pick_move(&session, P1);
346        assert!(
347            mv.is_none(),
348            "La IA debe retornar None cuando la partida terminó"
349        );
350    }
351
352    #[test]
353    fn test_ai_returns_none_when_not_player_turn() {
354        let session = new_checkers_session(SessionId(1), 42, P1, P2);
355        assert_eq!(session.current_player(), P1);
356
357        let mv = checkers_ai_pick_move(&session, P2);
358        assert!(
359            mv.is_none(),
360            "La IA debe retornar None si no es el turno del jugador"
361        );
362    }
