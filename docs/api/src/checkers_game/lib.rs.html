4//! Juego de referencia del OpenTable SDK: damas.
5//!
6//! Ejercita el SDK completo: `Command`/`Event` via los derives de
7//! `open_table_macros`, dos `Resource` (`BoardResource`, `Sides`),
8//! turnos (`TurnResource` del engine), y una condición de victoria.
9//! Sin información oculta (a propósito: damas es el caso más simple
10//! posible antes de meterse con un juego de cartas).
11//!
12//! Para jugarlo de punta a punta sin esperar a `open_table_godot`, ver
13//! `examples/local_match.rs` (dos jugadores por consola, mismo proceso).

35/// Arma una `Session` de damas lista para jugar: tablero inicial,
36/// jugadores, reglas registradas. Es el punto de entrada que tanto los
37/// tests como `examples/local_match.rs` usan para no repetir el setup.
38pub fn new_checkers_session(
39    session_id: SessionId,
40    seed: u64,
41    player_top: PlayerId,
42    player_bottom: PlayerId,
43) -> Session {
44    let mut session = Session::new_with_metadata(
45        session_id,
46        "Checkers",
47        seed,
48        vec![player_top, player_bottom],
49        build_checkers_rules(),
50    );
51    let setup = move |world: &mut BevyWorld| {
52        let (board, sides) = initial_board(player_top, player_bottom);
53        world.insert_resource(board);
54        world.insert_resource(sides);
55        world.insert_resource(MultiCaptureResource::default());
56        world.insert_resource(DrawClockResource::default());
57        world.insert_resource(LegalMoveGeneratorResource(std::sync::Arc::new(
58            CheckersLegalMoveGenerator,
59        )));
60    };
61    setup(session.world_mut());
62    session.set_initial_setup(setup);
63    session
64}
