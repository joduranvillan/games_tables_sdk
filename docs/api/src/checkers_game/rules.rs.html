6use crate::board::{BoardResource, DrawClockResource, MultiCaptureResource, Sides};
7use crate::commands::{MovePieceCommand, ResignCommand};
8use crate::events::{GameOverEvent, PieceMovedEvent};
9use crate::legal_moves::CheckersLegalMoveGenerator;
10use crate::position::Position;
11
12fn invalid(reason: impl Into<String>) -> EngineError {
13    EngineError::InvalidCommand {
14        reason: reason.into(),
15    }
16}
17
18fn has_any_capture(world: &BevyWorld, player: PlayerId) -> bool {
19    let board = world.resource::<BoardResource>();
20    let sides = world.resource::<Sides>();
21    !board.get_all_available_captures(sides, player).is_empty()
22}
23
24pub fn build_checkers_rules() -> RuleEngine {
25    let mut rules = RuleEngine::new();
26    rules.set_legal_move_generator(Box::new(CheckersLegalMoveGenerator));
27
28    rules.register::<MovePieceCommand, _>(|cmd, world| {
29        if !world.contains_resource::<BoardResource>() || !world.contains_resource::<Sides>() {
30            let players = world.resource::<TurnResource>().0.players();
31            let p1 = players[0];
32            let p2 = players.get(1).copied().unwrap_or(p1);
33            let (board, sides) = crate::board::initial_board(p1, p2);
34            world.insert_resource(board);
35            world.insert_resource(sides);
36        }
37        if !world.contains_resource::<MultiCaptureResource>() {
38            world.insert_resource(MultiCaptureResource::default());
39        }
40        if !world.contains_resource::<DrawClockResource>() {
41            world.insert_resource(DrawClockResource::default());
42        }
43        if !world.contains_resource::<LegalMoveGeneratorResource>() {
44            world.insert_resource(LegalMoveGeneratorResource(std::sync::Arc::new(
45                CheckersLegalMoveGenerator,
46            )));
47        }
48
49        // 1. Le tiene que tocar a este jugador.
50        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
51            return Err(invalid("no es tu turno"));
52        }
53
54        if !cmd.from.is_on_board() || !cmd.to.is_on_board() {
55            return Err(invalid("la casilla está fuera del tablero"));
56        }
57        if !cmd.from.is_dark_square() || !cmd.to.is_dark_square() {
58            return Err(invalid("damas solo se juega en casillas oscuras"));
59        }
60
61        // Si hay una captura múltiple activa, debe moverse con la misma ficha.
62        let active_jumper = world
63            .get_resource::<MultiCaptureResource>()
64            .and_then(|m| m.active_piece);
65        if let Some(required_from) = active_jumper {
66            if cmd.from != required_from {
67                return Err(invalid(
68                    "debés continuar la captura múltiple con la misma ficha",
69                ));
70            }
71        }
72
73        let opponent = world.resource::<Sides>().opponent_of(cmd.player);
74        let forward = world.resource::<Sides>().forward_direction(cmd.player);
75        let promotion_row = world.resource::<Sides>().promotion_row(cmd.player);
76        let sides_snapshot = *world.resource::<Sides>();
77        let captures_available = has_any_capture(world, cmd.player);
78
79        let mut board = world.resource_mut::<BoardResource>();
80
81        let piece = board
82            .get(cmd.from)
83            .ok_or_else(|| invalid("no hay ninguna ficha en la casilla de origen"))?;
84        if piece.owner != cmd.player {
85            return Err(invalid("esa ficha no es tuya"));
86        }
87        if board.get(cmd.to).is_some() {
88            return Err(invalid("la casilla de destino ya está ocupada"));
89        }
90
91        let (row_diff, col_diff) = cmd.from.diff(cmd.to);
92        if col_diff.abs() != row_diff.abs() {
93            return Err(invalid("el movimiento tiene que ser diagonal"));
94        }
95
96        if captures_available && row_diff.abs() != 2 {
97            return Err(invalid(
98                "hay capturas disponibles: la captura es obligatoria",
99            ));
100        }
101
102        let mut captured_pos: Option<Position> = None;
103
104        match row_diff.abs() {
105            1 => {
106                if !piece.is_king && row_diff.signum() != forward {
107                    return Err(invalid(
108                        "esa ficha todavía no es dama: solo puede avanzar hacia adelante",
109                    ));
110                }
111            }
112            2 => {
113                if !piece.is_king && row_diff.signum() != forward {
114                    return Err(invalid(
115                        "esa ficha todavía no es dama: solo puede capturar hacia adelante",
116                    ));
117                }
118                let mid = cmd.from.midpoint(cmd.to);
119                let mid_piece = board
120                    .get(mid)
121                    .ok_or_else(|| invalid("no hay ninguna ficha para saltar en el medio"))?;
122                if mid_piece.owner == cmd.player {
123                    return Err(invalid("no podés saltar tu propia ficha"));
124                }
125                captured_pos = Some(mid);
126            }
127            _ => return Err(invalid("las damas se mueven 1 casilla, o 2 al capturar")),
128        }
129
130        // --- Aplicar el movimiento ---
131        board.set(cmd.from, None);
132        if let Some(captured) = captured_pos {
133            board.set(captured, None);
134        }
135        let promoted = !piece.is_king && cmd.to.row == promotion_row;
136        board.set(
137            cmd.to,
138            Some(crate::piece::Piece {
139                owner: piece.owner,
140                is_king: piece.is_king || promoted,
141            }),
142        );
143
144        // Escanear si desde la casilla de llegada (cmd.to) la misma ficha tiene capturas adicionales disponibles.
145        // Si es así, se mantiene el turno del jugador y se registra la posición requerida (must_use_piece).
146        let has_further_capture = captured_pos.is_some()
147            && !promoted
148            && !board
149                .get_available_captures_for_piece(&sides_snapshot, cmd.player, cmd.to)
150                .is_empty();
151
152        let counts = board.count_pieces();
153        let opponent_blocked = if !has_further_capture {
154            board
155                .get_all_legal_moves(&sides_snapshot, opponent)
156                .is_empty()
157        } else {
158            false
159        };
160        let _ = board; // soltar el borrow antes de tocar otros resources
161
162        if let Some(mut multi) = world.get_resource_mut::<MultiCaptureResource>() {
163            if has_further_capture {
164                multi.active_piece = Some(cmd.to);
165            } else {
166                multi.active_piece = None;
167            }
168        } else if has_further_capture {
169            world.insert_resource(MultiCaptureResource {
170                active_piece: Some(cmd.to),
171            });
172        }
173
174        let half_moves = {
175            let mut clock = world.resource_mut::<DrawClockResource>();
176            if captured_pos.is_some() || promoted {
177                clock.half_moves_without_capture_or_promotion = 0;
178            } else {
179                clock.half_moves_without_capture_or_promotion += 1;
180            }
181            clock.half_moves_without_capture_or_promotion
182        };
183
184        let mut events: Vec<Box<dyn Event>> = vec![Box::new(PieceMovedEvent {
185            player: cmd.player,
186            from: cmd.from,
187            to: cmd.to,
188            captured: captured_pos,
189            promoted,
190        })];
191
192        // Condición de victoria: si el rival se quedó sin fichas o bloqueado sin movimientos legales.
193        let opponent_pieces_left = counts
194            .iter()
195            .find(|(id, _)| *id == opponent)
196            .map(|(_, count)| *count)
197            .unwrap_or(0);
198        if opponent_pieces_left == 0 || (!has_further_capture && opponent_blocked) {
199            world.insert_resource(GameOverResource(true));
200            world.insert_resource(WinnerResource(Some(cmd.player)));
201            events.push(Box::new(GameOverEvent {
202                winner: cmd.player,
203                is_draw: false,
204            }));
205        } else if !has_further_capture && half_moves >= 80 {
206            world.insert_resource(GameOverResource(true));
207            world.insert_resource(WinnerResource(None));
208            events.push(Box::new(GameOverEvent {
209                winner: PlayerId(0),
210                is_draw: true,
211            }));
212        }
213
214        if !has_further_capture {
215            world.resource_mut::<TurnResource>().0.advance();
216        }
217
218        Ok(events)
219    });
220
221    rules.register::<ResignCommand, _>(|cmd, world| {
222        let players = world.resource::<TurnResource>().0.players().to_vec();
223        if !players.contains(&cmd.player) {
224            return Err(invalid("no eres un jugador de esta partida"));
225        }
226        let opponent = players
227            .into_iter()
228            .find(|&p| p != cmd.player)
229            .unwrap_or(cmd.player);
230        world.insert_resource(GameOverResource(true));
231        world.insert_resource(WinnerResource(Some(opponent)));
232        Ok(vec![Box::new(GameOverEvent {
233            winner: opponent,
234            is_draw: false,
235        })])
236    });
237
238    rules
239}
