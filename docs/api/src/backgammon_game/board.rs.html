6pub const POINTS: usize = 24;
7pub const CHECKERS_PER_PLAYER: u32 = 15;

9/// El tablero: 24 puntos. Signo indica dueño (positivo = jugador
10/// `increasing`, negativo = jugador `decreasing`, ver `Sides`),
11/// magnitud indica cuántas fichas hay ahí — el truco clásico para
12/// representar un tablero de backgammon sin necesitar una struct
13/// `Option<(PlayerId, u32)>` por punto.
14#[derive(GameResource, Clone, Debug)]
15pub struct BoardResource {
16    pub points: [i32; POINTS],
17    pub bar: HashMap<PlayerId, u32>,
18    pub borne_off: HashMap<PlayerId, u32>,
19}
20
21impl BoardResource {
22    /// Configuración inicial estándar, adaptada a un único índice 0-23
23    /// compartido (en vez de la numeración clásica que cuenta distinto
24    /// para cada jugador) — simétrica: la posición del jugador
25    /// `decreasing` es el espejo de la del `increasing`.
26    pub fn standard_setup(increasing: PlayerId, decreasing: PlayerId) -> Self {
27        let mut points = [0i32; POINTS];
28        points[0] = 2;
29        points[11] = 5;
30        points[16] = 3;
31        points[18] = 5;
32        points[23] = -2;
33        points[12] = -5;
34        points[7] = -3;
35        points[5] = -5;
36
37        let mut bar = HashMap::new();
38        let mut borne_off = HashMap::new();
39        bar.insert(increasing, 0);
40        bar.insert(decreasing, 0);
41        borne_off.insert(increasing, 0);
42        borne_off.insert(decreasing, 0);
43
44        Self {
45            points,
46            bar,
47            borne_off,
48        }
49    }
50}

52#[derive(GameResource, Clone, Copy)]
53pub struct Sides {
54    pub increasing: PlayerId,
55    pub decreasing: PlayerId,
56}
57
58impl Sides {
59    pub fn owner_sign(&self, player: PlayerId) -> i32 {
60        if player == self.increasing {
61            1
62        } else {
63            -1
64        }
65    }
66
67    pub fn forward_direction(&self, player: PlayerId) -> i32 {
68        self.owner_sign(player)
69    }
70
71    pub fn opponent_of(&self, player: PlayerId) -> PlayerId {
72        if player == self.increasing {
73            self.decreasing
74        } else {
75            self.increasing
76        }
77    }
78
79    /// Punto de entrada al reingresar desde la barra, según el valor del
80    /// dado usado.
81    pub fn entry_point(&self, player: PlayerId, die: u32) -> usize {
82        if player == self.increasing {
83            (die - 1) as usize
84        } else {
85            (24 - die) as usize
86        }
87    }
88
89    /// Si el punto `idx` está dentro del cuadrante final de `player`
90    /// (condición para poder empezar a sacar fichas).
91    pub fn is_home_point(&self, player: PlayerId, idx: usize) -> bool {
92        if player == self.increasing {
93            idx >= 18
94        } else {
95            idx <= 5
96        }
97    }
98
99    pub fn all_checkers_home(&self, board: &BoardResource, player: PlayerId) -> bool {
100        if *board.bar.get(&player).unwrap_or(&0) > 0 {
101            return false;
102        }
103        let sign = self.owner_sign(player);
104        for (idx, &count) in board.points.iter().enumerate() {
105            if count * sign > 0 && !self.is_home_point(player, idx) {
106                return false;
107            }
108        }
109        true
110    }
111
112    /// Verifica si hay fichas del jugador en posiciones más lejanas de la meta dentro de su cuadrante final.
113    pub fn has_checkers_further_back(
114        &self,
115        board: &BoardResource,
116        player: PlayerId,
117        from: usize,
118    ) -> bool {
119        let sign = self.owner_sign(player);
120        if player == self.increasing {
121            (0..from).any(|idx| board.points[idx] * sign > 0)
122        } else {
123            ((from + 1)..POINTS).any(|idx| board.points[idx] * sign > 0)
124        }
125    }
126
127    /// Verifica si hay fichas más atrás del punto o del punto de salida correspondientes al dado.
128    pub fn has_checkers_behind_die_exit(
129        &self,
130        board: &BoardResource,
131        player: PlayerId,
132        from: usize,
133        die: u32,
134    ) -> bool {
135        let sign = self.owner_sign(player);
136        if player == self.increasing {
137            let exit_point = (24 - die as i32).max(0) as usize;
138            (0..exit_point).any(|idx| board.points[idx] * sign > 0)
139                || (0..from).any(|idx| board.points[idx] * sign > 0)
140        } else {
141            let exit_point = (die as usize).saturating_sub(1);
142            ((exit_point + 1)..POINTS).any(|idx| board.points[idx] * sign > 0)
143                || ((from + 1)..POINTS).any(|idx| board.points[idx] * sign > 0)
144        }
145    }
146
147    /// Comprueba si el jugador tiene al menos un movimiento válido con los dados disponibles.
148    pub fn has_any_valid_move(
149        &self,
150        board: &BoardResource,
151        player: PlayerId,
152        dice: &[u32],
153    ) -> bool {
154        let owner_sign = self.owner_sign(player);
155        let opponent_sign = -owner_sign;
156        let on_bar = *board.bar.get(&player).unwrap_or(&0);
157
158        if on_bar > 0 {
159            for &d in dice {
160                let entry = self.entry_point(player, d);
161                if board.points[entry] * opponent_sign <= 1 {
162                    return true;
163                }
164            }
165            return false;
166        }
167
168        let all_home = self.all_checkers_home(board, player);
169
170        for from in 0..POINTS {
171            if board.points[from] * owner_sign > 0 {
172                for &d in dice {
173                    let raw = from as i32 + self.forward_direction(player) * d as i32;
174                    if raw >= 0 && raw < POINTS as i32 {
175                        let dest = raw as usize;
176                        if board.points[dest] * opponent_sign <= 1 {
177                            return true;
178                        }
179                    } else if all_home {
180                        let distance = if player == self.increasing {
181                            24 - from as i32
182                        } else {
183                            from as i32 + 1
184                        };
185                        if d as i32 == distance
186                            || (d as i32 > distance
187                                && !self.has_checkers_behind_die_exit(board, player, from, d))
188                        {
189                            return true;
190                        }
191                    }
192                }
193            }
194        }
195
196        false
197    }
198}
