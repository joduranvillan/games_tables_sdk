28fn advance_turn_skip_bankrupt(world: &mut BevyWorld) {
29    for _ in 0..10 {
30        let next = world.resource_mut::<TurnResource>().0.advance();
31        let is_bankrupt = world.resource::<PlayersResource>().bankrupt.contains(&next);
32        if !is_bankrupt {
33            break;
34        }
35    }
36}

38fn handle_bankruptcy(
39    player: PlayerId,
40    creditor: Option<PlayerId>,
41    world: &mut BevyWorld,
42    events: &mut Vec<Box<dyn Event>>,
43) {
44    events.push(Box::new(BankruptcyEvent { player }));
45
46    let mut players = world.resource_mut::<PlayersResource>();
47    players.bankrupt.insert(player);
48
49    let player_props: Vec<usize> = players
50        .properties
51        .iter()
52        .filter(|(_, &owner)| owner == player)
53        .map(|(&space, _)| space)
54        .collect();
55
56    for space in player_props {
57        if let Some(creditor_player) = creditor {
58            players.properties.insert(space, creditor_player);
59        } else {
60            players.properties.remove(&space);
61        }
62    }
63    let _ = players;
64
65    let all_players = world.resource::<TurnResource>().0.players().to_vec();
66    let players_res = world.resource::<PlayersResource>();
67    let active_players: Vec<PlayerId> = all_players
68        .into_iter()
69        .filter(|p| !players_res.bankrupt.contains(p))
70        .collect();
71
72    if active_players.len() == 1 {
73        events.push(Box::new(GameOverEvent {
74            winner: active_players[0],
75        }));
76    }
77}

79/// Aplica el efecto de caer en una casilla determinada.
80/// Devuelve `true` si el jugador debe tomar una decisión (e.g. comprar propiedad)
81/// antes de que el turno pueda avanzar.
82pub fn apply_land_effect(
83    space: &SpaceKind,
84    space_idx: usize,
85    player: PlayerId,
86    roll: u32,
87    is_doubles: bool,
88    world: &mut BevyWorld,
89    events: &mut Vec<Box<dyn Event>>,
90) -> bool {
91    let board_spaces = world.resource::<BoardResource>().spaces.clone();
92
93    match space {
94        SpaceKind::Property {
95            ref group, rent, ..
96        } => {
97            let owner = world
98                .resource::<PlayersResource>()
99                .properties
100                .get(&space_idx)
101                .copied();
102            match owner {
103                None => {
104                    world
105                        .resource_mut::<PendingPurchaseResource>()
106                        .pending
107                        .insert(
108                            player,
109                            PendingDecision {
110                                space: space_idx,
111                                is_doubles,
112                            },
113                        );
114                    true
115                }
116                Some(owner) if owner == player => false,
117                Some(owner) => {
118                    let mut players = world.resource_mut::<PlayersResource>();
119                    let is_full_group = players.owns_full_group(&board_spaces, owner, group);
120                    let actual_rent = if is_full_group { rent * 2 } else { *rent };
121
122                    let payer_money =
123                        players.money.get(&player).copied().unwrap_or(0) - actual_rent as i64;
124                    players.money.insert(player, payer_money);
125                    *players.money.entry(owner).or_insert(0) += actual_rent as i64;
126                    let _ = players;
127
128                    events.push(Box::new(RentPaidEvent {
129                        payer: player,
130                        owner,
131                        space: space_idx,
132                        amount: actual_rent,
133                    }));
134
135                    if payer_money < 0 {
136                        handle_bankruptcy(player, Some(owner), world, events);
137                    }
138                    false
139                }
140            }
141        }
142        SpaceKind::Utility { .. } => {
143            let owner = world
144                .resource::<PlayersResource>()
145                .properties
146                .get(&space_idx)
147                .copied();
148            match owner {
149                None => {
150                    world
151                        .resource_mut::<PendingPurchaseResource>()
152                        .pending
153                        .insert(
154                            player,
155                            PendingDecision {
156                                space: space_idx,
157                                is_doubles,
158                            },
159                        );
160                    true
161                }
162                Some(owner) if owner == player => false,
163                Some(owner) => {
164                    let count = world
165                        .resource::<PlayersResource>()
166                        .count_utilities(&board_spaces, owner);
167                    let multiplier = if count >= 2 { 10 } else { 4 };
168                    let actual_rent = multiplier * roll;
169
170                    let mut players = world.resource_mut::<PlayersResource>();
171                    let payer_money =
172                        players.money.get(&player).copied().unwrap_or(0) - actual_rent as i64;
173                    players.money.insert(player, payer_money);
174                    *players.money.entry(owner).or_insert(0) += actual_rent as i64;
175                    let _ = players;
176
177                    events.push(Box::new(UtilityRentPaidEvent {
178                        player,
179                        owner,
180                        multiplier,
181                        dice_total: roll,
182                        amount: actual_rent,
183                    }));
184
185                    if payer_money < 0 {
186                        handle_bankruptcy(player, Some(owner), world, events);
187                    }
188                    false
189                }
190            }
191        }
192        SpaceKind::Railroad { .. } => {
193            let owner = world
194                .resource::<PlayersResource>()
195                .properties
196                .get(&space_idx)
197                .copied();
198            match owner {
199                None => {
200                    world
201                        .resource_mut::<PendingPurchaseResource>()
202                        .pending
203                        .insert(
204                            player,
205                            PendingDecision {
206                                space: space_idx,
207                                is_doubles,
208                            },
209                        );
210                    true
211                }
212                Some(owner) if owner == player => false,
213                Some(owner) => {
214                    let count = world
215                        .resource::<PlayersResource>()
216                        .count_railroads(&board_spaces, owner);
217                    let actual_rent = match count {
218                        1 => 25,
219                        2 => 50,
220                        3 => 100,
221                        _ => 200,
222                    };
223
224                    let mut players = world.resource_mut::<PlayersResource>();
225                    let payer_money =
226                        players.money.get(&player).copied().unwrap_or(0) - actual_rent as i64;
227                    players.money.insert(player, payer_money);
228                    *players.money.entry(owner).or_insert(0) += actual_rent as i64;
229                    let _ = players;
230
231                    events.push(Box::new(RailroadRentPaidEvent {
232                        player,
233                        owner,
234                        railroads_owned: count,
235                        amount: actual_rent,
236                    }));
237
238                    if payer_money < 0 {
239                        handle_bankruptcy(player, Some(owner), world, events);
240                    }
241                    false
242                }
243            }
244        }
245        SpaceKind::Tax { ref name, amount } => {
246            let mut players = world.resource_mut::<PlayersResource>();
247            let money = players.money.get(&player).copied().unwrap_or(0) - *amount as i64;
248            players.money.insert(player, money);
249            let _ = players;
250
251            events.push(Box::new(TaxPaidEvent {
252                player,
253                space: space_idx,
254                name: name.clone(),
255                amount: *amount,
256            }));
257
258            if money < 0 {
259                handle_bankruptcy(player, None, world, events);
260            }
261            false
262        }
263        SpaceKind::GoToJail => {
264            let mut players = world.resource_mut::<PlayersResource>();
265            players.position.insert(player, JAIL_SPACE);
266            let _ = players;
267
268            world
269                .resource_mut::<JailResource>()
270                .jailed
271                .insert(player, 3);
272            events.push(Box::new(JailedEvent { player }));
273            false
274        }
275        SpaceKind::Chance => {
276            let card_idx = {
277                let mut rng = world.resource_mut::<RandomResource>();
278                (rng.0.roll_die(10) - 1) as usize
279            };
280            let card = ChanceCard::ALL[card_idx % ChanceCard::ALL.len()];
281
282            events.push(Box::new(ChanceCardDrawnEvent {
283                player,
284                card,
285                description: card.description().to_string(),

287
288            match card {
289                ChanceCard::AdvanceToGo => {
290                    let mut players = world.resource_mut::<PlayersResource>();
291                    players.position.insert(player, 0);
292                    let money = players.money.get(&player).copied().unwrap_or(0) + GO_BONUS;
293                    players.money.insert(player, money);
294                    false
295                }
296                ChanceCard::AdvanceToIllinois => {
297                    let mut players = world.resource_mut::<PlayersResource>();
298                    let old_pos = players.position.get(&player).copied().unwrap_or(0);
299                    let mut money = players.money.get(&player).copied().unwrap_or(0);
300                    if old_pos > 24 {
301                        money += GO_BONUS;
302                    }
303                    players.position.insert(player, 24);
304                    players.money.insert(player, money);
305                    let _ = players;

308                    apply_land_effect(&illinois, 24, player, roll, is_doubles, world, events)

310                ChanceCard::AdvanceToStCharles => {
311                    let mut players = world.resource_mut::<PlayersResource>();
312                    let old_pos = players.position.get(&player).copied().unwrap_or(0);
313                    let mut money = players.money.get(&player).copied().unwrap_or(0);
314                    if old_pos > 11 {

324                ChanceCard::AdvanceToUtility => {
325                    let old_pos = world
326                        .resource::<PlayersResource>()
327                        .position
328                        .get(&player)
329                        .copied()
330                        .unwrap_or(0);
331                    let target_util = if !(12..=28).contains(&old_pos) {

344
345                    let util_space = board_spaces[target_util].clone();
346                    apply_land_effect(
347                        &util_space,
348                        target_util,
349                        player,
350                        roll,

354                    )
355                }
356                ChanceCard::AdvanceToRailroad => {
357                    let old_pos = world
358                        .resource::<PlayersResource>()
359                        .position
360                        .get(&player)
361                        .copied()
362                        .unwrap_or(0);
363                    let target_rr = if old_pos < 5 {
364                        5
365                    } else if old_pos < 15 {
366                        15
367                    } else if old_pos < 25 {
368                        25
369                    } else if old_pos < 35 {
370                        35
371                    } else {
372                        5
373                    };
374                    let mut players = world.resource_mut::<PlayersResource>();
375                    let mut money = players.money.get(&player).copied().unwrap_or(0);
376                    if old_pos >= 35 {
377                        money += GO_BONUS;

379                    players.position.insert(player, target_rr);

382

514pub fn build_monopoly_rules() -> RuleEngine {
515    let mut rules = RuleEngine::new();
516
517    rules.register::<PayJailFeeCommand, _>(|cmd, world| {
518        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
519            return Err(invalid("no es tu turno"));
520        }
521
522        let is_jailed = world
523            .resource::<JailResource>()
524            .jailed
525            .contains_key(&cmd.player);
526        if !is_jailed {
527            return Err(invalid("no estás en la cárcel"));
528        }
529
530        let mut players = world.resource_mut::<PlayersResource>();
531        let money = *players.money.get(&cmd.player).unwrap_or(&0);
532        if money < 50 {
533            return Err(invalid("no tenés suficiente dinero para pagar la fianza"));
534        }
535        players.money.insert(cmd.player, money - 50);
536        let _ = players;
537
538        world
539            .resource_mut::<JailResource>()
540            .jailed
541            .remove(&cmd.player);
542
543        Ok(vec![])
544    });
545
546    rules.register::<AuctionBidCommand, _>(|cmd, world| {
547        let auction_active = world.resource::<AuctionResource>().state.is_some();
548        if !auction_active {
549            return Err(invalid("no hay ninguna subasta activa"));
550        }
551
552        if cmd.amount > 0 {
553            let player_money = world
554                .resource::<PlayersResource>()
555                .money
556                .get(&cmd.player)
557                .copied()
558                .unwrap_or(0);
559            if player_money < cmd.amount as i64 {
560                return Err(invalid(
561                    "no tenés suficiente dinero para realizar esta puja",
562                ));
563            }
564
565            let mut auction = world.resource_mut::<AuctionResource>();
566            let state = auction.state.as_mut().unwrap();
567            if cmd.amount <= state.current_bid {
568                return Err(invalid("la puja debe ser mayor a la puja actual"));
569            }
570            state.current_bid = cmd.amount;
571            state.highest_bidder = Some(cmd.player);
572        } else {
573            let mut auction = world.resource_mut::<AuctionResource>();
574            let state = auction.state.as_mut().unwrap();
575            state.passed.insert(cmd.player);
576        }
577
578        let all_players = world.resource::<TurnResource>().0.players().to_vec();
579        let bankrupt = world.resource::<PlayersResource>().bankrupt.clone();
580        let active_players: Vec<PlayerId> = all_players
581            .into_iter()
582            .filter(|p| !bankrupt.contains(p))
583            .collect();
584
585        let state = world.resource::<AuctionResource>().state.as_ref().unwrap();
586        let active_bidders: Vec<PlayerId> = active_players
587            .iter()
588            .copied()
589            .filter(|p| !state.passed.contains(p))
590            .collect();
591
592        let mut events: Vec<Box<dyn Event>> = Vec::new();
593
594        if active_bidders.len() <= 1 {
595            let state = world
596                .resource_mut::<AuctionResource>()
597                .state
598                .take()
599                .unwrap();
600            if let Some(winner) = state.highest_bidder {
601                let mut players = world.resource_mut::<PlayersResource>();
602                let money = *players.money.get(&winner).unwrap_or(&0);
603                players
604                    .money
605                    .insert(winner, money - state.current_bid as i64);
606                players.properties.insert(state.space, winner);
607                let _ = players;
608
609                events.push(Box::new(AuctionWonEvent {
610                    player: winner,
611                    space: state.space,
612                    price: state.current_bid,
613                }));
614            }
615            advance_turn_skip_bankrupt(world);
616        }
617
618        Ok(events)
619    });
620
621    rules.register::<RollAndMoveCommand, _>(|cmd, world| {
622        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
623            return Err(invalid("no es tu turno"));
624        }
625        if world
626            .resource::<PlayersResource>()
627            .bankrupt
628            .contains(&cmd.player)
629        {
630            return Err(invalid("estás en quiebra y no podés jugar"));
631        }
632        if world
633            .resource::<PendingPurchaseResource>()
634            .pending
635            .contains_key(&cmd.player)
636        {
637            return Err(invalid(
638                "tenés una decisión de compra pendiente, resolvela antes de tirar",
639            ));
640        }
641        if world.resource::<AuctionResource>().state.is_some() {
642            return Err(invalid("hay una subasta activa en curso"));
643        }
644
645        let (d1, d2) = {
646            let mut rng = world.resource_mut::<RandomResource>();
647            (rng.0.roll_die(6), rng.0.roll_die(6))
648        };
649        let roll = d1 + d2;
650        let is_doubles = d1 == d2;
651
652        let is_jailed = world
653            .resource::<JailResource>()
654            .jailed
655            .contains_key(&cmd.player);
656
657        let mut events: Vec<Box<dyn Event>> = Vec::new();
658
659        if is_jailed {
660            if is_doubles {
661                world
662                    .resource_mut::<JailResource>()
663                    .jailed
664                    .remove(&cmd.player);
665            } else {
666                let attempts = *world
667                    .resource::<JailResource>()
668                    .jailed
669                    .get(&cmd.player)
670                    .unwrap();
671                if attempts <= 1 {
672                    world
673                        .resource_mut::<JailResource>()
674                        .jailed
675                        .remove(&cmd.player);
676                    let mut players = world.resource_mut::<PlayersResource>();
677                    let money = *players.money.get(&cmd.player).unwrap_or(&0) - 50;
678                    players.money.insert(cmd.player, money);
679                    let _ = players;
680
681                    if money < 0 {
682                        handle_bankruptcy(cmd.player, None, world, &mut events);
683                        advance_turn_skip_bankrupt(world);
684                        return Ok(events);
685                    }
686                } else {
687                    world
688                        .resource_mut::<JailResource>()
689                        .jailed
690                        .insert(cmd.player, attempts - 1);
691                    events.push(Box::new(PlayerMovedEvent {
692                        player: cmd.player,
693                        roll,
694                        d1,
695                        d2,
696                        is_doubles: false,
697                        new_position: JAIL_SPACE,
698                        passed_go: false,
699                    }));
700                    advance_turn_skip_bankrupt(world);
701                    return Ok(events);
702                }
703            }
704        }
705
706        let board_spaces = world.resource::<BoardResource>().spaces.clone();
707        let board_size = board_spaces.len();
708
709        let mut players = world.resource_mut::<PlayersResource>();
710        let old_pos = *players.position.get(&cmd.player).unwrap_or(&0);
711        let raw = old_pos + roll as usize;
712        let mut new_pos = raw % board_size;
713        let passed_go = raw >= board_size;
714
715        let mut money = *players.money.get(&cmd.player).unwrap_or(&0);
716        if passed_go {
717            money += GO_BONUS;
718        }
719
720        let mut jailed = false;
721        let landed_pos = new_pos;
722        if new_pos == GO_TO_JAIL_SPACE {
723            new_pos = JAIL_SPACE;
724            jailed = true;
725        }
726
727        players.position.insert(cmd.player, new_pos);
728        players.money.insert(cmd.player, money);
729        let _ = players;
730
731        if jailed {
732            world
733                .resource_mut::<JailResource>()
734                .jailed
735                .insert(cmd.player, 3);
736        }
737
738        events.push(Box::new(PlayerMovedEvent {
739            player: cmd.player,
740            roll,
741            d1,
742            d2,
743            is_doubles,
744            new_position: landed_pos,
745            passed_go,
746        }));
747
748        if jailed {
749            events.push(Box::new(JailedEvent { player: cmd.player }));
750        }
751
752        let mut awaiting_decision = false;
753
754        if !jailed {
755            let space = board_spaces[new_pos].clone();
756            awaiting_decision = apply_land_effect(
757                &space,
758                new_pos,
759                cmd.player,
760                roll,
761                is_doubles,
762                world,
763                &mut events,
764            );
765        }
766
767        let is_bankrupt = world
768            .resource::<PlayersResource>()
769            .bankrupt
770            .contains(&cmd.player);
771        if !awaiting_decision && (!is_doubles || is_bankrupt) {
772            advance_turn_skip_bankrupt(world);
773        }
774
775        Ok(events)
776    });
777
778    rules.register::<BuyPropertyCommand, _>(|cmd, world| {
779        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
780            return Err(invalid("no es tu turno"));
781        }
782
783        let pending_decision = *world
784            .resource::<PendingPurchaseResource>()
785            .pending
786            .get(&cmd.player)
787            .ok_or_else(|| invalid("no tenés ninguna decisión de compra pendiente"))?;
788
789        let space = pending_decision.space;
790        let is_doubles = pending_decision.is_doubles;
791
792        let mut events: Vec<Box<dyn Event>> = Vec::new();
793
794        if cmd.buy {
795            let price = match &world.resource::<BoardResource>().spaces[space] {
796                SpaceKind::Property { price, .. } => *price,
797                SpaceKind::Utility { price, .. } => *price,
798                SpaceKind::Railroad { price, .. } => *price,
799                _ => return Err(invalid("esa casilla no es una propiedad comprable")),
800            };
801
802            let mut players = world.resource_mut::<PlayersResource>();
803            let money = *players.money.get(&cmd.player).unwrap_or(&0);
804            if money < price as i64 {
805                return Err(invalid(
806                    "no tenés suficiente dinero para comprar esa propiedad",
807                ));
808            }
809            players.money.insert(cmd.player, money - price as i64);
810            players.properties.insert(space, cmd.player);
811            let _ = players;
812
813            events.push(Box::new(PropertyPurchasedEvent {
814                player: cmd.player,
815                space,
816                price,
817            }));
818
819            world
820                .resource_mut::<PendingPurchaseResource>()
821                .pending
822                .remove(&cmd.player);
823
824            if !is_doubles {
825                advance_turn_skip_bankrupt(world);
826            }
827        } else {
828            world
829                .resource_mut::<PendingPurchaseResource>()
830                .pending
831                .remove(&cmd.player);
832
833            world.resource_mut::<AuctionResource>().state = Some(AuctionState {
834                space,
835                current_bid: 0,
836                highest_bidder: None,
837                passed: HashSet::new(),
838            });
839
840            events.push(Box::new(AuctionStartedEvent { space }));
841        }
842
843        Ok(events)
844    });
