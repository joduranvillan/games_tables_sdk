121/// Tablero estándar de 40 casillas de Monopoly:
122/// - 28 propiedades (22 calles + 4 estaciones + 2 suministros)
123/// - GO, Cárcel (visita), Parking Gratuito, Ve a la Cárcel
124/// - 6 casillas de Suerte/Arca Comunitaria (3 cada una)
125/// - Impuesto sobre la Renta (200$), Impuesto de Lujo (100$)
126pub fn board_layout() -> Vec<SpaceKind> {
127    fn property(name: &str, group: &str, price: u32, rent: u32) -> SpaceKind {
128        SpaceKind::Property {
129            name: name.to_string(),
130            group: group.to_string(),
131            price,
132            rent,
133        }
134    }
135
136    fn utility(name: &str, price: u32) -> SpaceKind {
137        SpaceKind::Utility {
138            name: name.to_string(),
139            price,
140        }
141    }
142
143    fn railroad(name: &str, price: u32) -> SpaceKind {
144        SpaceKind::Railroad {
145            name: name.to_string(),
146            price,
147        }
148    }
149
150    fn tax(name: &str, amount: u32) -> SpaceKind {
151        SpaceKind::Tax {
152            name: name.to_string(),
153            amount,
154        }
155    }
156
157    vec![
158        // 0: GO
159        SpaceKind::Go,
160        // 1..3: Marrón
161        property("Av. Mediterráneo", "Marrón", 60, 2),

164        // 4: Impuesto sobre la Renta
