124/// Tablero estándar de 40 casillas de Monopoly:
125/// - 28 propiedades (22 calles + 4 estaciones + 2 suministros)
126/// - GO, Cárcel (visita), Parking Gratuito, Ve a la Cárcel
127/// - 6 casillas de Suerte/Arca Comunitaria (3 cada una)
128/// - Impuesto sobre la Renta (200$), Impuesto de Lujo (100$)
129pub fn board_layout() -> Vec<SpaceKind> {
130    fn property(name: &str, group: &str, price: u32, rent: u32) -> SpaceKind {
131        SpaceKind::Property {
132            name: name.to_string(),
133            group: group.to_string(),
134            price,
135            rent,
136        }
137    }
138
139    fn utility(name: &str, price: u32) -> SpaceKind {
140        SpaceKind::Utility {
141            name: name.to_string(),
142            price,
143        }
144    }
145
146    fn railroad(name: &str, price: u32) -> SpaceKind {
147        SpaceKind::Railroad {
148            name: name.to_string(),
149            price,
150        }
151    }
152
153    fn tax(name: &str, amount: u32) -> SpaceKind {
154        SpaceKind::Tax {
155            name: name.to_string(),
156            amount,
157        }
158    }
159
160    vec![
161        // 0: GO
162        SpaceKind::Go,
163        // 1..3: Marrón
164        property("Av. Mediterráneo", "Marrón", 60, 2),
165        SpaceKind::CommunityChest,
166        property("Av. Báltica", "Marrón", 60, 4),
167        // 4: Impuesto sobre la Renta
168        tax("Impuesto sobre la Renta", 200),
169        // 5: Ferrocarril Reading
170        railroad("Ferrocarril Reading", 200),
171        // 6..9: Celeste
172        property("Av. Oriental", "Celeste", 100, 6),
173        SpaceKind::Chance,
174        property("Av. Vermont", "Celeste", 100, 6),
175        property("Av. Connecticut", "Celeste", 120, 8),
176        // 10: Cárcel / Solo Visita
177        SpaceKind::Jail,
178        // 11..14: Rosa + Compañía Eléctrica
179        property("Plaza San Carlos", "Rosa", 140, 10),
180        utility("Compañía Eléctrica", 150),
181        property("Av. Estados", "Rosa", 140, 10),
182        property("Av. Virginia", "Rosa", 160, 12),
183        // 15: Ferrocarril de Pennsylvania
184        railroad("Ferrocarril de Pennsylvania", 200),
185        // 16..19: Naranja
186        property("Plaza St. James", "Naranja", 180, 14),
187        SpaceKind::CommunityChest,
188        property("Av. Tennessee", "Naranja", 180, 14),
189        property("Av. Nueva York", "Naranja", 200, 16),
190        // 20: Parking Gratuito
191        SpaceKind::FreeParking,
192        // 21..24: Rojo
193        property("Av. Kentucky", "Rojo", 220, 18),
194        SpaceKind::Chance,
195        property("Av. Indiana", "Rojo", 220, 18),
196        property("Av. Illinois", "Rojo", 240, 20),
197        // 25: Ferrocarril B. & O.
198        railroad("Ferrocarril B. & O.", 200),
199        // 26..29: Amarillo + Planta de Agua
200        property("Av. Atlántico", "Amarillo", 260, 22),
201        property("Av. Ventnor", "Amarillo", 260, 22),
202        utility("Planta de Agua", 150),
203        property("Jardines Marvin", "Amarillo", 280, 24),
204        // 30: Ve a la Cárcel
205        SpaceKind::GoToJail,
206        // 31..34: Verde
207        property("Av. Pacífico", "Verde", 300, 26),
208        property("Av. Carolina del Norte", "Verde", 300, 26),
209        SpaceKind::CommunityChest,
210        property("Av. Pennsylvania", "Verde", 320, 28),
211        // 35: Ferrocarril Short Line
212        railroad("Ferrocarril Short Line", 200),
213        // 36..39: Suerte, Azul Oscuro, Impuesto de Lujo
214        SpaceKind::Chance,
215        property("Plaza Park", "Azul Oscuro", 350, 35),
216        tax("Impuesto de Lujo", 100),
217        property("Paseo Tablado", "Azul Oscuro", 400, 50),
218    ]
219}
