4//! Código QR de invitación a una sala.
5//!
6//! Una aclaración de diseño importante: Nearby Connections (Android) y
7//! MultipeerConnectivity (iOS) YA hacen descubrimiento de dispositivos
8//! cercanos solos - no hace falta ningún código para que un teléfono
9//! "encuentre" a otro. El QR acá resuelve un problema distinto: cuando
10//! hay más de una partida siendo anunciada cerca (dos mesas jugando en el
11//! mismo bar, por ejemplo), ¿a cuál se une el jugador? Y además, que un
12//! jugador no técnico no tenga que tipear nada.
13//!
14//! Por eso el QR NO codifica direcciones de red, MACs de Bluetooth ni
15//! IDs de conexión de Nearby Connections/Multipeer - esos son manejados
16//! por esas APIs y son efímeros. El QR codifica solo un `JoinPayload`
17//! chico: un `join_code` legible (también se muestra como texto, por si
18//! alguien no puede escanear) que el cliente usa para filtrar, entre los
19//! dispositivos que ya descubrió por Bluetooth/Wi-Fi, cuál es el correcto.
20
21use qrcode::render::svg;
22use qrcode::QrCode;
23use serde::{Deserialize, Serialize};
24use thiserror::Error;
25
26use open_table_core::ids::SessionId;
27use open_table_core::random::RandomService;
28
29const CODE_ALPHABET: &[u8] = b"ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; // sin 0/O/1/I: se confunden a simple vista
30const CODE_LENGTH: usize = 6;
31
32#[derive(Debug, Error)]
33pub enum QrError {
34    #[error("no se pudo serializar el payload de invitación: {0}")]
35    Serialization(#[from] serde_json::Error),
36    #[error("no se pudo generar el código QR: {0}")]
37    Encoding(#[from] qrcode::types::QrError),
38}
39
40/// Lo que efectivamente viaja adentro del QR (y también se puede mostrar
41/// como texto plano para quien no pueda escanear).
42#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
43pub struct JoinPayload {
44    pub session_id: SessionId,
45    pub join_code: String,
46    pub host_name: String,
47    pub game_type: String,
48}
49
50/// Payload extendido para partidas en red local (LAN / Hotspot offline) que
51/// incluye dirección IP y puerto TCP además de los metadatos de la sala.
52#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
53pub struct LanJoinPayload {
54    pub session_id: SessionId,
55    pub join_code: String,
56    pub host_name: String,
57    pub game_type: String,
58    pub ip: String,
59    pub port: u16,
60}
61
62/// Genera un código corto, legible, para mostrar junto al QR. Usa
63/// `RandomService` (el mismo que ya usa el motor para dados/barajado) en
64/// vez de una fuente de aleatoriedad nueva, para no introducir una
65/// segunda noción de "azar" en el SDK.
66pub fn generate_join_code(rng: &mut RandomService) -> String {
67    (0..CODE_LENGTH)
68        .map(|_| {
69            let idx = rng.range(0, CODE_ALPHABET.len() as u32 - 1) as usize;
70            CODE_ALPHABET[idx] as char
71        })
72        .collect()
73}
74
75/// Genera el QR como un string SVG, listo para que el bridge de Godot lo
76/// cargue como textura (Godot puede rasterizar SVG directamente, o este
77/// string se puede convertir a PNG del lado de Godot si hace falta).
78pub fn generate_join_qr_svg(payload: &JoinPayload) -> Result<String, QrError> {
79    let json = serde_json::to_string(payload)?;
80    let code = QrCode::new(json.as_bytes())?;
81
82    let mut svg = code
83        .render::<svg::Color>()
84        .min_dimensions(240, 240)
85        .quiet_zone(true)
86        .build();
87
88    svg.push_str(&format!("<!--OPENTABLE_QR_JSON:{json}-->"));
89    Ok(svg)
90}

92/// Genera un código QR SVG con los datos completos de conexión LAN offline
93/// (`ip`, `port`, `game_type`, `join_code`, `host_name`).
94pub fn generate_lan_join_qr_svg(payload: &LanJoinPayload) -> Result<String, QrError> {
95    let json = serde_json::to_string(payload)?;
96    let code = QrCode::new(json.as_bytes())?;
97
98    let mut svg = code
99        .render::<svg::Color>()
100        .min_dimensions(240, 240)
101        .quiet_zone(true)
102        .build();
103
104    svg.push_str(&format!("<!--OPENTABLE_QR_JSON:{json}-->"));
105    Ok(svg)
106}

108/// El lado del cliente: decodifica lo que salió de escanear el QR (o de
109/// tipear el `join_code` a mano como fallback).
110pub fn parse_join_payload(json: &str) -> Result<JoinPayload, QrError> {
111    Ok(serde_json::from_str(json)?)
112}
113
114/// Decodifica un payload LAN completo desde JSON.
115pub fn parse_lan_join_payload(json: &str) -> Result<LanJoinPayload, QrError> {
116    Ok(serde_json::from_str(json)?)
117}

119/// Extrae el JSON de invitación desde un archivo/string SVG generado por
120/// `open_table_qr` o desde un texto JSON directo.
121pub fn extract_qr_payload_json(input: &str) -> Option<String> {
122    let trimmed = input.trim();
123    if let Some(start_idx) = trimmed.find("<!--OPENTABLE_QR_JSON:") {
124        let rest = &trimmed[start_idx + "<!--OPENTABLE_QR_JSON:".len()..];
125        if let Some(end_idx) = rest.find("-->") {
126            return Some(rest[..end_idx].trim().to_string());
127        }
128    }
129    if trimmed.starts_with('{') && trimmed.ends_with('}') {
130        return Some(trimmed.to_string());
131    }
132    None
133}
