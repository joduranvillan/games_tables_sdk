1//! Código QR de invitación a una sala.
2//!
3//! Una aclaración de diseño importante: Nearby Connections (Android) y
4//! MultipeerConnectivity (iOS) YA hacen descubrimiento de dispositivos
5//! cercanos solos - no hace falta ningún código para que un teléfono
6//! "encuentre" a otro. El QR acá resuelve un problema distinto: cuando
7//! hay más de una partida siendo anunciada cerca (dos mesas jugando en el
8//! mismo bar, por ejemplo), ¿a cuál se une el jugador? Y además, que un
9//! jugador no técnico no tenga que tipear nada.
10//!
11//! Por eso el QR NO codifica direcciones de red, MACs de Bluetooth ni
12//! IDs de conexión de Nearby Connections/Multipeer - esos son manejados
13//! por esas APIs y son efímeros. El QR codifica solo un `JoinPayload`
14//! chico: un `join_code` legible (también se muestra como texto, por si
15//! alguien no puede escanear) que el cliente usa para filtrar, entre los
16//! dispositivos que ya descubrió por Bluetooth/Wi-Fi, cuál es el correcto.
17
18use qrcode::render::svg;
19use qrcode::QrCode;
20use serde::{Deserialize, Serialize};
21use thiserror::Error;
22
23use open_table_core::ids::SessionId;
24use open_table_core::random::RandomService;
25
26const CODE_ALPHABET: &[u8] = b"ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; // sin 0/O/1/I: se confunden a simple vista
27const CODE_LENGTH: usize = 6;
28
29#[derive(Debug, Error)]
30pub enum QrError {
31    #[error("no se pudo serializar el payload de invitación: {0}")]
32    Serialization(#[from] serde_json::Error),
33    #[error("no se pudo generar el código QR: {0}")]
34    Encoding(#[from] qrcode::types::QrError),
35}
36
37/// Lo que efectivamente viaja adentro del QR (y también se puede mostrar
38/// como texto plano para quien no pueda escanear).
39#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
40pub struct JoinPayload {
41    pub session_id: SessionId,
42    pub join_code: String,
43    pub host_name: String,
44    pub game_type: String,
45}
46
47/// Genera un código corto, legible, para mostrar junto al QR. Usa
48/// `RandomService` (el mismo que ya usa el motor para dados/barajado) en
49/// vez de una fuente de aleatoriedad nueva, para no introducir una
50/// segunda noción de "azar" en el SDK.
51pub fn generate_join_code(rng: &mut RandomService) -> String {
52    (0..CODE_LENGTH)
53        .map(|_| {
54            let idx = rng.range(0, CODE_ALPHABET.len() as u32 - 1) as usize;
55            CODE_ALPHABET[idx] as char
56        })
57        .collect()
58}
59
60/// Genera el QR como un string SVG, listo para que el bridge de Godot lo
61/// cargue como textura (Godot puede rasterizar SVG directamente, o este
62/// string se puede convertir a PNG del lado de Godot si hace falta).
63pub fn generate_join_qr_svg(payload: &JoinPayload) -> Result<String, QrError> {
64    let json = serde_json::to_string(payload)?;
65    let code = QrCode::new(json.as_bytes())?;
66
67    let svg = code
68        .render::<svg::Color>()
69        .min_dimensions(240, 240)
70        .quiet_zone(true)
71        .build();
72
73    Ok(svg)
74}
75
76/// El lado del cliente: decodifica lo que salió de escanear el QR (o de
77/// tipear el `join_code` a mano como fallback).
78pub fn parse_join_payload(json: &str) -> Result<JoinPayload, QrError> {
79    Ok(serde_json::from_str(json)?)
