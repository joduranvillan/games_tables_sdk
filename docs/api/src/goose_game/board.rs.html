10pub const GARDEN_SQUARE: u8 = 63;
11pub const GOOSE_SQUARES: [u8; 13] = [5, 9, 14, 18, 23, 27, 32, 36, 41, 45, 50, 54, 59];
12pub const BRIDGE_A: u8 = 6;
13pub const BRIDGE_B: u8 = 12;
14pub const INN_SQUARE: u8 = 19;
15pub const DICE_A: u8 = 26;
16pub const DICE_B: u8 = 53;
17pub const WELL_SQUARE: u8 = 31;
18pub const LABYRINTH_SQUARE: u8 = 42;
19pub const LABYRINTH_TARGET: u8 = 30;
20pub const PRISON_SQUARE: u8 = 52;
21pub const SKULL_SQUARE: u8 = 58;
22pub const SKULL_TARGET: u8 = 1;
23
24#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
25pub enum SpecialSquareKind {
26    Normal,
27    Goose,
28    Bridge,
29    Inn,
30    Dice,
31    Well,
32    Labyrinth,
33    Prison,
34    Skull,
35    Garden,
36}

38#[derive(Debug, Clone, PartialEq, Eq)]
39pub struct LandingResolution {
40    pub raw_square: u8,
41    pub final_square: u8,
42    pub bonus_turn: bool,
43    pub wait_turns: u8,
44    pub kind: SpecialSquareKind,
45    pub description: String,
46}
47
48pub fn is_goose_square(sq: u8) -> bool {
49    GOOSE_SQUARES.contains(&sq)
50}
51
52pub fn next_goose_square(sq: u8) -> u8 {
53    for &g in &GOOSE_SQUARES {
54        if g > sq {
55            return g;
56        }
57    }
58    GARDEN_SQUARE
59}

61/// Calcula la casilla tras aplicar el avance y el rebote exacto en la casilla 63.
62pub fn compute_raw_square(from: u8, roll: u8) -> u8 {
63    let target = from.saturating_add(roll);
64    if target <= GARDEN_SQUARE {
65        target
66    } else {
67        let excess = target - GARDEN_SQUARE;
68        GARDEN_SQUARE.saturating_sub(excess)
69    }
70}

72/// Resuelve las reglas clásicas de las 63 casillas de El Juego de la Oca.
73pub fn resolve_landing(from: u8, roll: u8) -> LandingResolution {
74    let raw_square = compute_raw_square(from, roll);
75
76    if raw_square == GARDEN_SQUARE {
77        return LandingResolution {
78            raw_square,
79            final_square: GARDEN_SQUARE,
80            bonus_turn: false,
81            wait_turns: 0,
82            kind: SpecialSquareKind::Garden,
83            description: "¡Jardín de la Oca (63)! Victoria exacta.".to_string(),
84        };
85    }
86
87    if is_goose_square(raw_square) {
88        let dest = next_goose_square(raw_square);
89        let is_win = dest == GARDEN_SQUARE;
90        return LandingResolution {
91            raw_square,
92            final_square: dest,
93            bonus_turn: !is_win,
94            wait_turns: 0,
95            kind: if is_win {
96                SpecialSquareKind::Garden
97            } else {
98                SpecialSquareKind::Goose
99            },
100            description: if is_win {
101                format!("¡De Oca ({raw_square}) al Jardín de la Oca (63)! Victoria.")
102            } else {
103                format!("¡De oca ({raw_square}) a oca ({dest}) y tiro porque me toca!")
104            },
105        };
106    }
107
108    match raw_square {
109        BRIDGE_A => LandingResolution {
110            raw_square,
111            final_square: BRIDGE_B,
112            bonus_turn: true,
113            wait_turns: 0,
114            kind: SpecialSquareKind::Bridge,
115            description:
116                "¡De puente (6) a puente (12) y tiro porque me lleva la corriente!".to_string(),
117        },
118        BRIDGE_B => LandingResolution {
119            raw_square,
120            final_square: BRIDGE_A,
121            bonus_turn: true,
122            wait_turns: 0,
123            kind: SpecialSquareKind::Bridge,
124            description:
125                "¡De puente (12) a puente (6) y tiro porque me lleva la corriente!".to_string(),
126        },
127        INN_SQUARE => LandingResolution {
128            raw_square,
129            final_square: INN_SQUARE,
130            bonus_turn: false,
131            wait_turns: 1,
132            kind: SpecialSquareKind::Inn,
133            description: "La Posada (19): descansas 1 turno.".to_string(),
134        },
135        DICE_A => LandingResolution {
136            raw_square,
137            final_square: DICE_B,
138            bonus_turn: true,
139            wait_turns: 0,
140            kind: SpecialSquareKind::Dice,
141            description: "¡De dado (26) a dado (53) y tiro porque me ha tocado!".to_string(),
142        },
143        DICE_B => LandingResolution {
144            raw_square,
145            final_square: DICE_A,
146            bonus_turn: true,
147            wait_turns: 0,
148            kind: SpecialSquareKind::Dice,
149            description: "¡De dado (53) a dado (26) y tiro porque me ha tocado!".to_string(),
150        },
151        WELL_SQUARE => LandingResolution {
152            raw_square,
153            final_square: WELL_SQUARE,
154            bonus_turn: false,
155            wait_turns: 2,
156            kind: SpecialSquareKind::Well,
157            description: "El Pozo (31): esperas 2 turnos (o hasta que otro jugador te rescate)."
158                .to_string(),
159        },
160        LABYRINTH_SQUARE => LandingResolution {
161            raw_square,
162            final_square: LABYRINTH_TARGET,
163            bonus_turn: false,
164            wait_turns: 0,
165            kind: SpecialSquareKind::Labyrinth,
166            description: "El Laberinto (42): te pierdes y retrocedes a la casilla 30.".to_string(),
167        },
168        PRISON_SQUARE => LandingResolution {
169            raw_square,
170            final_square: PRISON_SQUARE,
171            bonus_turn: false,
172            wait_turns: 3,
173            kind: SpecialSquareKind::Prison,
174            description: "La Cárcel (52): retenido durante 3 turnos.".to_string(),
175        },
176        SKULL_SQUARE => LandingResolution {
177            raw_square,
178            final_square: SKULL_TARGET,
179            bonus_turn: false,
180            wait_turns: 0,
181            kind: SpecialSquareKind::Skull,
182            description: "La Calavera (58): vuelves a empezar desde la casilla 1.".to_string(),
183        },
184        _ => LandingResolution {
185            raw_square,
186            final_square: raw_square,
187            bonus_turn: false,
188            wait_turns: 0,
189            kind: SpecialSquareKind::Normal,
190            description: format!("Avanza a la casilla {raw_square}."),
191        },
192    }
193}

195#[derive(GameResource, Debug, Clone)]
196pub struct GooseBoardResource {
197    pub positions: HashMap<PlayerId, u8>,
198    pub wait_turns: HashMap<PlayerId, u8>,
199    pub winner: Option<PlayerId>,
200    pub last_effect: String,
201}
202
203impl GooseBoardResource {
204    pub fn new(players: &[PlayerId]) -> Self {
205        let mut positions = HashMap::new();
206        let mut wait_turns = HashMap::new();
207        for &p in players {
208            positions.insert(p, 0);
209            wait_turns.insert(p, 0);
210        }
211        Self {
212            positions,
213            wait_turns,
214            winner: None,
215            last_effect: String::new(),
216        }
217    }
218
219    pub fn position_of(&self, player: PlayerId) -> u8 {
220        self.positions.get(&player).copied().unwrap_or(0)
221    }
222
223    pub fn wait_turns_of(&self, player: PlayerId) -> u8 {
224        self.wait_turns.get(&player).copied().unwrap_or(0)
225    }
226}
