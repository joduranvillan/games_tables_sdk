11#[derive(Debug, Default, Clone, Copy)]
12pub struct GooseMoveGenerator;
13
14impl GooseMoveGenerator {
15    pub fn legal_moves(&self, session: &Session, player: PlayerId) -> Vec<Box<dyn Command>> {
16        self.generate(session.world(), player)
17    }
18}
19
20impl LegalMoveGenerator for GooseMoveGenerator {
21    fn generate(&self, world: &BevyWorld, player: PlayerId) -> Vec<Box<dyn Command>> {
22        if world
23            .get_resource::<TurnResource>()
24            .is_none_or(|t| !t.0.is_current(player))
25        {
26            return Vec::new();
27        }
28        if world
29            .get_resource::<GooseBoardResource>()
30            .and_then(|b| b.winner)
31            .is_some()
32        {
33            return Vec::new();
34        }
35        let has_pending = world
36            .get_resource::<PendingRollResource>()
37            .and_then(|p| p.pending.get(&player))
38            .is_some();
39        if has_pending {
40            vec![Box::new(MoveTokenCommand { player })]
41        } else {
42            vec![Box::new(RollDiceCommand { player })]
43        }
44    }
45}
46
47/// Selecciona el movimiento de avance para `player` cuando tiene una tirada pendiente.
48pub fn goose_ai_pick_move(session: &Session, player: PlayerId) -> Option<MoveTokenCommand> {
49    if session.current_player() != player {
50        return None;
51    }
52
53    if session
54        .world()
55        .get_resource::<GooseBoardResource>()
56        .and_then(|b| b.winner)
57        .is_some()
58    {
59        return None;
60    }
61
62    let _roll = session
63        .world()
64        .get_resource::<PendingRollResource>()
65        .and_then(|p| p.pending.get(&player).copied())?;
66
67    Some(MoveTokenCommand { player })
68}
69
70pub fn best_move(
71    session: &Session,
72    player: PlayerId,
73    _difficulty: AiDifficulty,
74) -> Option<MoveTokenCommand> {
75    goose_ai_pick_move(session, player)
76}
77
78pub fn select_ai_move(
79    session: &Session,
80    player: PlayerId,
81    _difficulty: GooseAiDifficulty,
82) -> Option<Box<dyn Command>> {
83    GooseMoveGenerator
84        .legal_moves(session, player)
85        .into_iter()
86        .next()
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92    use crate::new_goose_session;
93
94    const P1: PlayerId = PlayerId(1);
95    const P2: PlayerId = PlayerId(2);
96
97    #[test]
98    fn test_goose_ai_pick_move_with_pending_roll() {
99        let mut session = new_goose_session(SessionId(1), 42, vec![P1, P2]);
100        assert!(goose_ai_pick_move(&session, P1).is_none());
101
102        session
103            .world_mut()
104            .resource_mut::<PendingRollResource>()
105            .pending
106            .insert(P1, 4);
107
108        let mv = goose_ai_pick_move(&session, P1);
109        assert_eq!(mv, Some(MoveTokenCommand { player: P1 }));
110    }
