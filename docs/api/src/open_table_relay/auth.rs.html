1//! Validación y emisión de tokens JWT (`HS256`) para el servidor Relay.
2//!
3//! Lee `JWT_SECRET` desde las variables de entorno:
4//! - Si no está presente, la autenticación queda desactivada (modo desarrollo local).
5//! - Si está presente, el cliente debe enviar `{"op": "auth", "token": "<jwt>"}` en los primeros
6//!   5 segundos tras conectar.
7//! - Valida estrictamente la firma `HMAC-SHA256` y el campo `exp`. Tokens expirados son
8//!   rechazados con `{"op": "auth_error", "reason": "expired"}`.
9
10use hmac::{Hmac, KeyInit, Mac};
11use serde::{Deserialize, Serialize};
12use sha2::{Digest, Sha256};
13use std::time::{SystemTime, UNIX_EPOCH};
14
15type HmacSha256 = Hmac<Sha256>;

17/// Tiempo máximo permitido para recibir el primer mensaje `{"op": "auth"}` cuando `JWT_SECRET` está activo.
18pub const AUTH_HANDSHAKE_TIMEOUT_SECS: u64 = 5;
19
20/// Tiempo de vida de los tokens JWT de invitado emitidos por `POST /auth/guest` (7 días en segundos).
21pub const GUEST_TOKEN_TTL_SECS: u64 = 7 * 24 * 60 * 60;
22
23/// Secreto de desarrollo utilizado para firmar tokens locales cuando `JWT_SECRET` no está configurado.
24pub const DEV_DEFAULT_JWT_SECRET: &str = "open_table_dev_local_secret";

33/// Respuesta JSON emitida por `POST /auth/guest`.
34#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
35pub struct GuestAuthResponse {
36    /// Token JWT firmado con `HS256`.
37    pub token: String,
38    /// Identificador canónico de invitado (`guest_<hash8>`).
39    pub player_id: String,
40    /// Timestamp UNIX (en segundos) de expiración del token (`now + 7 días`).
41    pub expires_at: u64,
42}

44/// Claims mínimos requeridos en el token JWT del cliente.
45#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
46pub struct JwtClaims {
47    /// Identificador canónico del jugador (`sub`).
48    pub sub: String,
49    /// Timestamp UNIX (en segundos) de expiración del token (`exp`).
50    pub exp: u64,
51}

53/// Errores posibles durante la validación de un JWT.
54#[derive(Debug, Clone, PartialEq, Eq)]
55pub enum AuthError {
56    /// El token ha expirado (`exp <= now`).
57    Expired,
58    /// La firma HMAC-SHA256 no coincide con `JWT_SECRET`.
59    InvalidSignature,
60    /// El formato del JWT o su cabecera/payload no es válido.
61    MalformedToken,
62}
63
64impl AuthError {
65    /// Código corto de razón para el mensaje `{"op": "auth_error", "reason": "..."}`.
66    pub fn reason(&self) -> &'static str {
67        match self {
68            Self::Expired => "expired",
69            Self::InvalidSignature => "invalid_signature",
70            Self::MalformedToken => "malformed_token",
71        }
72    }
73}

75/// Configuración de autenticación del servidor Relay.
76#[derive(Debug, Clone, Default)]
77pub struct AuthConfig {
78    /// Secreto compartido `HS256` si la autenticación está habilitada.
79    pub jwt_secret: Option<String>,
80}
81
82impl AuthConfig {
83    /// Construye `AuthConfig` leyendo la variable de entorno `JWT_SECRET`.
84    pub fn from_env() -> Self {
85        let secret = std::env::var("JWT_SECRET")
86            .ok()
87            .filter(|s| !s.trim().is_empty());
88        Self { jwt_secret: secret }
89    }
90
91    /// Crea una configuración con la autenticación obligatoria desactivada (`jwt_secret: None`).
92    pub fn disabled() -> Self {
93        Self::default()
94    }
95
96    /// Crea una configuración con secreto explícito.
97    pub fn with_secret(secret: impl Into<String>) -> Self {
98        Self {
99            jwt_secret: Some(secret.into()),
100        }
101    }
102
103    /// Indica si el servidor requiere autenticación JWT.
104    pub fn is_enabled(&self) -> bool {
105        self.jwt_secret.is_some()
106    }
107
108    /// Verifica un token JWT contra el secreto configurado usando el reloj actual del sistema.
109    pub fn verify_token(&self, token: &str) -> Result<JwtClaims, AuthError> {
110        let now = current_unix_timestamp();
111        let Some(secret) = self.jwt_secret.as_deref() else {
112            return match verify_jwt_hs256(token, DEV_DEFAULT_JWT_SECRET, now) {
113                Ok(claims) => Ok(claims),
114                Err(AuthError::Expired) => Err(AuthError::Expired),
115                Err(_) => Ok(JwtClaims {
116                    sub: "anonymous".to_string(),
117                    exp: u64::MAX,
118                }),
119            };
120        };
121        verify_jwt_hs256(token, secret, now)
122    }
123
124    /// Emite un token JWT de invitado válido por 7 días (`GUEST_TOKEN_TTL_SECS`) a partir de `device_id`.
125    ///
126    /// Si `JWT_SECRET` no está configurado en el servidor, firma un token de desarrollo válido en modo local.
127    pub fn issue_guest_token(&self, device_id: &str) -> GuestAuthResponse {
128        self.issue_guest_token_at(device_id, current_unix_timestamp())
129    }
130
131    /// Emite un token JWT de invitado válido por 7 días a partir del timestamp `now_unix_secs`.
132    pub fn issue_guest_token_at(&self, device_id: &str, now_unix_secs: u64) -> GuestAuthResponse {
133        issue_guest_token_with_secret(device_id, self.jwt_secret.as_deref(), now_unix_secs)
134    }
135}

137/// Deriva un identificador canónico `guest_<hash8>` determinista a partir del `device_id`.
138pub fn derive_guest_player_id(device_id: &str) -> String {
139    let clean = device_id.trim();
140    let input = if clean.is_empty() { "anonymous" } else { clean };
141    let mut hasher = Sha256::new();
142    hasher.update(input.as_bytes());
143    let digest = hasher.finalize();
144    format!(
145        "guest_{:02x}{:02x}{:02x}{:02x}",
146        digest[0], digest[1], digest[2], digest[3]
147    )
148}

150/// Emite una respuesta `GuestAuthResponse` con un JWT firmado con `HS256` y expiración de 7 días.
151///
152/// Si `secret` es `None` o vacío, utiliza `DEV_DEFAULT_JWT_SECRET` para producir un token de desarrollo válido.
153pub fn issue_guest_token_with_secret(
154    device_id: &str,
155    secret: Option<&str>,
156    now_unix_secs: u64,
157) -> GuestAuthResponse {
158    let player_id = derive_guest_player_id(device_id);
159    let expires_at = now_unix_secs.saturating_add(GUEST_TOKEN_TTL_SECS);
160    let signing_secret = secret
161        .filter(|s| !s.trim().is_empty())
162        .unwrap_or(DEV_DEFAULT_JWT_SECRET);
163    let claims = JwtClaims {
164        sub: player_id.clone(),
165        exp: expires_at,
166    };
167    let token = encode_jwt_hs256(&claims, signing_secret);
168    GuestAuthResponse {
169        token,
170        player_id,
171        expires_at,
172    }
173}

175/// Devuelve el timestamp UNIX actual en segundos.
176pub fn current_unix_timestamp() -> u64 {
177    SystemTime::now()
178        .duration_since(UNIX_EPOCH)
179        .unwrap_or_default()
180        .as_secs()
181}

183/// Genera un token JWT firmado con `HS256` (`HMAC-SHA256`).
184pub fn encode_jwt_hs256(claims: &JwtClaims, secret: &str) -> String {
185    let header_json = r#"{"alg":"HS256","typ":"JWT"}"#;
186    let payload_json = serde_json::to_string(claims).unwrap_or_else(|_| "{}".to_string());
187
188    let header_b64 = base64url_encode(header_json.as_bytes());
189    let payload_b64 = base64url_encode(payload_json.as_bytes());
190    let signing_input = format!("{header_b64}.{payload_b64}");
191
192    let sig = compute_hmac_sha256(signing_input.as_bytes(), secret.as_bytes());
193    let sig_b64 = base64url_encode(&sig);
194    format!("{signing_input}.{sig_b64}")
195}

197/// Valida un token JWT `HS256` comprobando su firma y su expiración (`exp > now_unix_secs`).
198pub fn verify_jwt_hs256(
199    token: &str,
200    secret: &str,
201    now_unix_secs: u64,
202) -> Result<JwtClaims, AuthError> {
203    let parts: Vec<&str> = token.trim().split('.').collect();
204    if parts.len() != 3 {
205        return Err(AuthError::MalformedToken);
206    }
207
208    let header_bytes = base64url_decode(parts[0]).ok_or(AuthError::MalformedToken)?;
209    let header_val: serde_json::Value =
210        serde_json::from_slice(&header_bytes).map_err(|_| AuthError::MalformedToken)?;
211    if header_val.get("alg").and_then(|v| v.as_str()) != Some("HS256") {
212        return Err(AuthError::MalformedToken);
213    }
214
215    let signing_input = format!("{}.{}", parts[0], parts[1]);
216    let expected_sig = compute_hmac_sha256(signing_input.as_bytes(), secret.as_bytes());
217    let provided_sig = base64url_decode(parts[2]).ok_or(AuthError::MalformedToken)?;
218
219    if !constant_time_eq(&expected_sig, &provided_sig) {
220        return Err(AuthError::InvalidSignature);
221    }
222
223    let payload_bytes = base64url_decode(parts[1]).ok_or(AuthError::MalformedToken)?;
224    let claims: JwtClaims =
225        serde_json::from_slice(&payload_bytes).map_err(|_| AuthError::MalformedToken)?;
226
227    if claims.sub.trim().is_empty() {
228        return Err(AuthError::MalformedToken);
229    }
230
231    if claims.exp <= now_unix_secs {
232        return Err(AuthError::Expired);
233    }
234
235    Ok(claims)
236}

238fn compute_hmac_sha256(data: &[u8], key: &[u8]) -> Vec<u8> {
239    let mut mac = HmacSha256::new_from_slice(key).expect("HMAC accepts keys of any size");
240    mac.update(data);
241    mac.finalize().into_bytes().to_vec()
242}

244fn constant_time_eq(a: &[u8], b: &[u8]) -> bool {
245    if a.len() != b.len() {
246        return false;
247    }
248    let mut diff = 0u8;
249    for (&x, &y) in a.iter().zip(b.iter()) {
250        diff |= x ^ y;
251    }
252    diff == 0
253}

255const B64_URL_TABLE: &[u8; 64] =
256    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
257
258fn base64url_encode(input: &[u8]) -> String {
259    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
260    let mut i = 0;
261    while i + 3 <= input.len() {
262        let n = ((input[i] as u32) << 16) | ((input[i + 1] as u32) << 8) | (input[i + 2] as u32);
263        out.push(B64_URL_TABLE[((n >> 18) & 0x3F) as usize] as char);
264        out.push(B64_URL_TABLE[((n >> 12) & 0x3F) as usize] as char);
265        out.push(B64_URL_TABLE[((n >> 6) & 0x3F) as usize] as char);
266        out.push(B64_URL_TABLE[(n & 0x3F) as usize] as char);
267        i += 3;
268    }
269    let rem = input.len() - i;
270    if rem == 1 {
271        let n = (input[i] as u32) << 16;
272        out.push(B64_URL_TABLE[((n >> 18) & 0x3F) as usize] as char);
273        out.push(B64_URL_TABLE[((n >> 12) & 0x3F) as usize] as char);
274    } else if rem == 2 {
275        let n = ((input[i] as u32) << 16) | ((input[i + 1] as u32) << 8);
276        out.push(B64_URL_TABLE[((n >> 18) & 0x3F) as usize] as char);
277        out.push(B64_URL_TABLE[((n >> 12) & 0x3F) as usize] as char);
278        out.push(B64_URL_TABLE[((n >> 6) & 0x3F) as usize] as char);
279    }
280    out
281}

283fn base64url_decode(input: &str) -> Option<Vec<u8>> {
284    let clean = input.trim_end_matches('=');
285    let mut buf = Vec::with_capacity(clean.len() * 3 / 4);
286    let mut accum = 0u32;
287    let mut bits = 0u32;
288
289    for b in clean.bytes() {
290        let val = match b {
291            b'A'..=b'Z' => b - b'A',
292            b'a'..=b'z' => b - b'a' + 26,
293            b'0'..=b'9' => b - b'0' + 52,
294            b'-' | b'+' => 62,
295            b'_' | b'/' => 63,
296            _ => return None,
297        } as u32;
298        accum = (accum << 6) | val;
299        bits += 6;
300        if bits >= 8 {
301            bits -= 8;
302            buf.push(((accum >> bits) & 0xFF) as u8);
303        }
304    }
305    Some(buf)
306}

313    fn test_auth_jwt_valid_and_canonical_sub() {
314        let secret = "super_secret_test_key";
315        let now = 1_700_000_000;
316        let claims = JwtClaims {
317            sub: "grandmaster_42".to_string(),
318            exp: now + 3600,
319        };
320        let token = encode_jwt_hs256(&claims, secret);
321        let decoded = verify_jwt_hs256(&token, secret, now).unwrap();
322        assert_eq!(decoded.sub, "grandmaster_42");
