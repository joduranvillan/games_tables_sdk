1//! Validación Anti-Cheat autoritativa de movimientos en el servidor Relay para `chess` y `checkers`.
2//!
3//! Cuando una sala se crea con `"game_id": "chess"` o `"game_id": "checkers"`, el Relay
4//! mantiene una instancia local de `open_table_engine::session::Session` y verifica cada movimiento
5//! antes de retransmitirlo a los demás participantes.
6
7use open_table_core::command::Command;
8use open_table_core::ids::{PlayerId, SessionId};
9use open_table_engine::session::Session;
10use serde_json::Value;
11use std::fmt;
12
13/// Estado de validación Anti-Cheat asociado a una sala activa.
14pub struct AntiCheatSession {
15    game_id: String,
16    session: Session,
17}
18
19impl fmt::Debug for AntiCheatSession {
20    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
21        f.debug_struct("AntiCheatSession")
22            .field("game_id", &self.game_id)
23            .field("current_tick", &self.session.current_tick())
24            .finish()
25    }
26}
27
28impl AntiCheatSession {
29    /// Crea un validador Anti-Cheat si `game_id` es `"chess"` o `"checkers"`.
30    /// Para cualquier otro juego devuelve `None` (retransmisión directa sin validación).
31    pub fn for_game(game_id: Option<&str>) -> Option<Self> {
32        let gid = game_id?.trim().to_ascii_lowercase();
33        let p1 = PlayerId(1);
34        let p2 = PlayerId(2);
35        match gid.as_str() {
36            "chess" => Some(Self {
37                game_id: "chess".to_string(),
38                session: chess_game::new_chess_session(SessionId(1), 42, p1, p2),
39            }),
40            "checkers" => Some(Self {
41                game_id: "checkers".to_string(),
42                session: checkers_game::new_checkers_session(SessionId(1), 42, p1, p2),
43            }),
44            _ => None,
45        }
46    }
47
48    /// Deserializa y aplica un comando sobre la `Session` local del juego.
49    /// Devuelve `Ok(())` si el comando es legal y fue aplicado, o `Err("illegal_move")` si es inválido.
50    pub fn validate_and_apply(
51        &mut self,
52        sender_room_player_id: u32,
53        data: &Value,
54    ) -> Result<(), &'static str> {
55        let cmd_val: Value = match data {
56            Value::String(s) => {
57                let trimmed = s.trim();
58                if !trimmed.starts_with('{') {
59                    // Mensajes de control no estructurados como JSON de comando
60                    return Ok(());
61                }
62                serde_json::from_str(trimmed).map_err(|_| "illegal_move")?
63            }
64            Value::Object(_) => data.clone(),
65            _ => return Err("illegal_move"),
66        };
67
68        let sender = cmd_val
69            .get("player")
70            .and_then(|v| v.as_u64())
71            .and_then(|v| u32::try_from(v).ok())
72            .map(PlayerId)
73            .unwrap_or(PlayerId(sender_room_player_id));
74
75        let cmd_box: Box<dyn Command> = match self.game_id.as_str() {
76            "chess" => parse_chess_command(&cmd_val, sender).ok_or("illegal_move")?,
77            "checkers" => parse_checkers_command(&cmd_val, sender).ok_or("illegal_move")?,
78            _ => return Ok(()),
79        };
80
81        self.session
82            .process_command(sender, cmd_box)
83            .map(|_| ())
84            .map_err(|_| "illegal_move")
85    }
86}

88fn parse_chess_command(val: &Value, default_player: PlayerId) -> Option<Box<dyn Command>> {
89    if let Ok(cmd) = serde_json::from_value::<chess_game::MovePieceCommand>(val.clone()) {
90        return Some(Box::new(cmd));
91    }
92
93    let from = parse_chess_pos(val.get("from")?)?;
94    let to = parse_chess_pos(val.get("to")?)?;
95    let player = val
96        .get("player")
97        .and_then(|v| v.as_u64())
98        .and_then(|v| u32::try_from(v).ok())
99        .map(PlayerId)
100        .unwrap_or(default_player);
101
102    Some(Box::new(chess_game::MovePieceCommand {
103        player,
104        from,
105        to,
106        promote_to: None,
107    }))
108}
109
110fn parse_chess_pos(v: &Value) -> Option<chess_game::Position> {
111    if let Ok(pos) = serde_json::from_value::<chess_game::Position>(v.clone()) {
112        return Some(pos);
113    }
114    if let Some(arr) = v.as_array() {
115        if arr.len() == 2 {
116            let r = u8::try_from(arr[0].as_u64()?).ok()?;
117            let c = u8::try_from(arr[1].as_u64()?).ok()?;
118            return Some(chess_game::Position::new(r, c));
119        }
120    }
121    None
122}

124fn parse_checkers_command(val: &Value, default_player: PlayerId) -> Option<Box<dyn Command>> {
125    if let Ok(cmd) = serde_json::from_value::<checkers_game::MovePieceCommand>(val.clone()) {
126        return Some(Box::new(cmd));
127    }
128
129    let from = parse_checkers_pos(val.get("from")?)?;
130    let to = parse_checkers_pos(val.get("to")?)?;
131    let player = val
132        .get("player")
133        .and_then(|v| v.as_u64())
134        .and_then(|v| u32::try_from(v).ok())
135        .map(PlayerId)
136        .unwrap_or(default_player);
137
138    Some(Box::new(checkers_game::MovePieceCommand {
139        player,
140        from,
141        to,
142    }))
143}

145fn parse_checkers_pos(v: &Value) -> Option<checkers_game::Position> {
146    if let Ok(pos) = serde_json::from_value::<checkers_game::Position>(v.clone()) {
147        return Some(pos);
148    }
149    if let Some(arr) = v.as_array() {

153            return Some(checkers_game::Position::new(r, c));
154        }
155    }
156    None
157}

159#[cfg(test)]
160mod tests {
161    use super::*;
162    use serde_json::json;
163
164    #[test]
165    fn test_anticheat_chess_legal_and_illegal_moves() {
166        let mut ac = AntiCheatSession::for_game(Some("chess")).unwrap();
167
168        // Movimiento legal de apertura para Blancas (P1): peón e2 (1,4) -> e4 (3,4)
169        let legal = json!({
170            "player": 1,
171            "from": { "row": 1, "col": 4 },
172            "to": { "row": 3, "col": 4 }
173        });
174        assert!(ac.validate_and_apply(1, &legal).is_ok());
175
176        // Movimiento ilegal para Negras (P2): intentar mover rey e8 (7,4) -> e5 (4,4)
177        let illegal = json!({
178            "player": 2,
179            "from": { "row": 7, "col": 4 },
180            "to": { "row": 4, "col": 4 }
181        });
182        assert_eq!(ac.validate_and_apply(2, &illegal), Err("illegal_move"));
183    }
184
185    #[test]
186    fn test_anticheat_checkers_legal_and_illegal_moves() {
187        let mut ac = AntiCheatSession::for_game(Some("checkers")).unwrap();
188
189        // Movimiento ilegal en Damas: saltar a una casilla ocupada o fuera de alcance (0,1) -> (5,5)
190        let illegal = json!({
191            "player": 1,
192            "from": { "row": 0, "col": 1 },
193            "to": { "row": 5, "col": 5 }
194        });
195        assert_eq!(ac.validate_and_apply(1, &illegal), Err("illegal_move"));
196
197        // Movimiento diagonal simple legal para P1 (top): (2,1) -> (3,0)
198        let legal = json!({
199            "player": 1,
200            "from": { "row": 2, "col": 1 },
201            "to": { "row": 3, "col": 0 }
202        });
203        assert!(ac.validate_and_apply(1, &legal).is_ok());
204    }
