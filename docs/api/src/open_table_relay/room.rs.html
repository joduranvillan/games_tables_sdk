1//! Gestión de salas y generación de códigos legibles para `open_table_relay`.
2//!
3//! Administra salas públicas/privadas en memoria y asigna códigos alfanuméricos
4//! de 4 caracteres sin caracteres ambiguos (`O`, `0`, `I`, `1`).
5
6use crate::anticheat::AntiCheatSession;
7use serde_json::Value;
8use std::collections::HashMap;
9use tokio::sync::mpsc;
10
11/// Alfabeto legible de 32 caracteres: excluye `O`, `0`, `I`, `1` para evitar confusiones visuales.
12pub const ROOM_CODE_ALPHABET: &[u8; 32] = b"ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
13
14/// Longitud estándar de los códigos de sala.
15pub const ROOM_CODE_LEN: usize = 4;
16
17/// Canal de envío hacia un cliente WebSocket conectado.
18pub type ClientSender = mpsc::UnboundedSender<String>;
19
20/// Identificador único de conexión interna asignado por el servidor.
21pub type ConnectionId = u64;
22
23/// Identificador de jugador dentro de una sala (1 = host, 2..N = invitados, 0 = espectador).
24pub type RoomPlayerId = u32;
25
26/// Límite máximo de espectadores permitidos por sala.
27pub const MAX_SPECTATORS_PER_ROOM: usize = 10;
28
29/// Rol de un participante dentro de una sala Relay.
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub enum ParticipantRole {
32    /// Jugador activo con permisos de escritura/retransmisión.
33    Player,
34    /// Espectador de solo lectura (sin permisos para emitir `op: "relay"`).
35    Spectator,
36}
37
38/// Información de un participante conectado a una sala.
39#[derive(Debug, Clone)]
40pub struct Participant {
41    /// Identificador único de la conexión.
42    pub conn_id: ConnectionId,
43    /// Identificador de jugador dentro de la sala (1-indexed para jugadores, 0 para espectadores).
44    pub player_id: RoomPlayerId,
45    /// Rol del participante en la sala (`Player` o `Spectator`).
46    pub role: ParticipantRole,
47    /// Canal para enviar mensajes JSON serializados al cliente.
48    pub sender: ClientSender,
49}

51/// Representa una sala activa en el servidor Relay.
52#[derive(Debug)]
53pub struct Room {
54    /// Código único de 4 caracteres de la sala.
55    pub code: String,
56    /// Identificador opcional del juego (ej. `"chess"`, `"go"`).
57    pub game_id: Option<String>,
58    /// Indica si la sala es pública o privada.
59    pub is_public: bool,
60    /// Contador incremental para asignar `player_id` a nuevos participantes.
61    next_player_id: RoomPlayerId,
62    /// Participantes activos indexados por `ConnectionId`.
63    pub participants: HashMap<ConnectionId, Participant>,
64    /// Sesión autoritativa opcional para validación Anti-Cheat (`chess`, `checkers`).
65    pub anticheat: Option<AntiCheatSession>,
66}
67
68impl Room {
69    /// Crea una nueva sala con el anfitrión inicial (`player_id = 1`).
70    pub fn new(
71        code: String,
72        host_conn_id: ConnectionId,
73        host_sender: ClientSender,
74        game_id: Option<String>,
75        is_public: bool,
76    ) -> Self {
77        let mut participants = HashMap::new();
78        participants.insert(
79            host_conn_id,
80            Participant {
81                conn_id: host_conn_id,
82                player_id: 1,
83                role: ParticipantRole::Player,
84                sender: host_sender,
85            },
86        );
87        let anticheat = AntiCheatSession::for_game(game_id.as_deref());
88        Self {
89            code,
90            game_id,
91            is_public,
92            next_player_id: 2,
93            participants,
94            anticheat,
95        }
96    }
97
98    /// Añade un participante jugador a la sala y devuelve el `player_id` asignado.
99    pub fn add_participant(&mut self, conn_id: ConnectionId, sender: ClientSender) -> RoomPlayerId {
100        if let Some(existing) = self.participants.get(&conn_id) {
101            if existing.role == ParticipantRole::Player {
102                return existing.player_id;
103            }
104        }
105        let assigned_id = self.next_player_id;
106        self.next_player_id += 1;
107        self.participants.insert(
108            conn_id,
109            Participant {
110                conn_id,
111                player_id: assigned_id,
112                role: ParticipantRole::Player,
113                sender,
114            },
115        );
116        assigned_id
117    }
118
119    /// Añade un espectador (solo lectura) a la sala y devuelve el `player_count` actual.
120    /// Falla con `RoomError::SpectatorLimit` si ya hay 10 espectadores en la sala.
121    pub fn add_spectator(
122        &mut self,
123        conn_id: ConnectionId,
124        sender: ClientSender,
125    ) -> Result<usize, RoomError> {
126        if let Some(existing) = self.participants.get(&conn_id) {
127            if existing.role == ParticipantRole::Spectator {
128                return Ok(self.player_count());
129            }
130        }
131        if self.spectator_count() >= MAX_SPECTATORS_PER_ROOM {
132            return Err(RoomError::SpectatorLimit);
133        }
134        self.participants.insert(
135            conn_id,
136            Participant {
137                conn_id,
138                player_id: 0,
139                role: ParticipantRole::Spectator,
140                sender,
141            },
142        );
143        Ok(self.player_count())
144    }
145
146    /// Cantidad de jugadores activos (no espectadores) en la sala.
147    pub fn player_count(&self) -> usize {
148        self.participants
149            .values()
150            .filter(|p| p.role == ParticipantRole::Player)
151            .count()
152    }
153
154    /// Cantidad de espectadores activos en la sala.
155    pub fn spectator_count(&self) -> usize {
156        self.participants
157            .values()
158            .filter(|p| p.role == ParticipantRole::Spectator)
159            .count()
160    }
161
162    /// Elimina un participante de la sala y devuelve su `player_id` si estaba presente.
163    pub fn remove_participant(&mut self, conn_id: ConnectionId) -> Option<RoomPlayerId> {
164        self.participants.remove(&conn_id).map(|p| p.player_id)
165    }
166
167    /// Envía un mensaje JSON a todos los participantes (jugadores y espectadores) excepto `exclude_conn`.
168    pub fn broadcast_except(&self, exclude_conn: ConnectionId, message: &str) {
169        for (&cid, participant) in &self.participants {
170            if cid != exclude_conn {
171                let _ = participant.sender.send(message.to_string());
172            }
173        }
174    }
175
176    /// Devuelve `true` si no quedan participantes conectados en la sala.
177    pub fn is_empty(&self) -> bool {
178        self.participants.is_empty()
179    }
180}

182/// Genera un código de sala aleatorio de 4 caracteres usando [`ROOM_CODE_ALPHABET`].
183pub fn generate_room_code() -> String {
184    let mut buf = [0u8; ROOM_CODE_LEN];
185    if getrandom::fill(&mut buf).is_err() {
186        // Fallback determinista seguro en caso de fallo del generador del SO
187        let ts = std::time::SystemTime::now()
188            .duration_since(std::time::UNIX_EPOCH)
189            .map(|d| d.as_nanos() as u64)
190            .unwrap_or(0xA5A5_5A5A);
191        for (i, byte) in buf.iter_mut().enumerate() {
192            *byte = ((ts >> (i * 8)) & 0xFF) as u8;
193        }
194    }
195    buf.iter()
196        .map(|&b| ROOM_CODE_ALPHABET[(b as usize) % ROOM_CODE_ALPHABET.len()] as char)
197        .collect()
198}
199
200/// Valida que un código de sala tenga exactamente 4 caracteres del alfabeto permitido.
201pub fn is_valid_room_code(code: &str) -> bool {
202    code.len() == ROOM_CODE_LEN
203        && code
204            .bytes()
205            .all(|b| ROOM_CODE_ALPHABET.contains(&b.to_ascii_uppercase()))
206}

208/// Gestor central de salas activas y membresía de conexiones.
209#[derive(Debug, Default)]
210pub struct RoomManager {
211    /// Salas activas por código (en mayúsculas).
212    rooms: HashMap<String, Room>,
213    /// Mapa inverso de `ConnectionId` al código de sala en el que participa.
214    conn_to_room: HashMap<ConnectionId, String>,
215}
216
217/// Errores posibles en las operaciones de sala.
218#[derive(Debug, Clone, PartialEq, Eq)]
219pub enum RoomError {
220    /// El código de sala solicitado no existe.
221    RoomNotFound,
222    /// El código de sala tiene un formato inválido.
223    InvalidRoomCode,
224    /// El cliente no se encuentra actualmente en ninguna sala.
225    NotInRoom,
226    /// El movimiento fue rechazado por el validador Anti-Cheat del motor.
227    IllegalMove,
228    /// Un espectador intentó enviar un mensaje `relay`.
229    SpectatorCannotRelay,
230    /// Se superó el límite de 10 espectadores en la sala.
231    SpectatorLimit,
232}
233
234impl RoomManager {
235    /// Crea un nuevo gestor de salas vacío.
236    pub fn new() -> Self {
237        Self::default()
238    }

240    /// Crea una nueva sala con un código único de 4 caracteres y registra al anfitrión.
241    pub fn create_room(
242        &mut self,
243        host_conn_id: ConnectionId,
244        host_sender: ClientSender,
245        game_id: Option<String>,
246        is_public: bool,
247    ) -> String {
248        // Si el cliente ya estaba en otra sala, lo sacamos primero limpiamente
249        let _ = self.leave_room(host_conn_id);
250
251        let code = loop {
252            let candidate = generate_room_code();
253            if !self.rooms.contains_key(&candidate) {
254                break candidate;
255            }
256        };
257
258        let room = Room::new(code.clone(), host_conn_id, host_sender, game_id, is_public);
259        self.rooms.insert(code.clone(), room);
260        self.conn_to_room.insert(host_conn_id, code.clone());
261        code
262    }
263
264    /// Une un cliente a una sala existente por su código.
265    ///
266    /// Devuelve `(room_code_normalizado, player_id_asignado)`.
267    pub fn join_room(
268        &mut self,
269        conn_id: ConnectionId,
270        sender: ClientSender,
271        raw_code: &str,
272    ) -> Result<(String, RoomPlayerId), RoomError> {
273        let normalized = raw_code.trim().to_ascii_uppercase();
274        if !is_valid_room_code(&normalized) {
275            return Err(RoomError::InvalidRoomCode);
276        }
277
278        if !self.rooms.contains_key(&normalized) {
279            return Err(RoomError::RoomNotFound);
280        }
281
282        // Si estaba en otra sala distinta, salir primero
283        if let Some(prev_code) = self.conn_to_room.get(&conn_id) {
284            if prev_code != &normalized {
285                let _ = self.leave_room(conn_id);
286            }
287        }
288
289        let room = self
290            .rooms
291            .get_mut(&normalized)
292            .ok_or(RoomError::RoomNotFound)?;
293        let player_id = room.add_participant(conn_id, sender);
294        self.conn_to_room.insert(conn_id, normalized.clone());
295
296        Ok((normalized, player_id))
297    }
298
299    /// Une un cliente como espectador de solo lectura a una sala existente por su código.
300    ///
301    /// Devuelve `(room_code_normalizado, player_count)`.
302    pub fn join_spectator(
303        &mut self,
304        conn_id: ConnectionId,
305        sender: ClientSender,
306        raw_code: &str,
307    ) -> Result<(String, usize), RoomError> {
308        let normalized = raw_code.trim().to_ascii_uppercase();
309        if !is_valid_room_code(&normalized) {
310            return Err(RoomError::InvalidRoomCode);
311        }
312
313        if !self.rooms.contains_key(&normalized) {
314            return Err(RoomError::RoomNotFound);
315        }
316
317        if let Some(prev_code) = self.conn_to_room.get(&conn_id) {
318            if prev_code != &normalized {
319                let _ = self.leave_room(conn_id);
320            }
321        }
322
323        let room = self
324            .rooms
325            .get_mut(&normalized)
326            .ok_or(RoomError::RoomNotFound)?;
327        let player_count = room.add_spectator(conn_id, sender)?;
328        self.conn_to_room.insert(conn_id, normalized.clone());
329
330        Ok((normalized, player_count))
331    }

333    /// Retransmite un mensaje JSON a los demás miembros de la sala donde está `conn_id`.
334    ///
335    /// Devuelve el `(room_code, player_id)` del emisor.
336    pub fn relay_to_peers(
337        &self,
338        conn_id: ConnectionId,
339        outbound_json: &str,
340    ) -> Result<(String, RoomPlayerId), RoomError> {
341        let code = self
342            .conn_to_room
343            .get(&conn_id)
344            .ok_or(RoomError::NotInRoom)?;
345        let room = self.rooms.get(code).ok_or(RoomError::RoomNotFound)?;
346        let participant = room
347            .participants
348            .get(&conn_id)
349            .ok_or(RoomError::NotInRoom)?;
350        if participant.role == ParticipantRole::Spectator {
351            return Err(RoomError::SpectatorCannotRelay);
352        }
353        let sender_player_id = participant.player_id;
354
355        room.broadcast_except(conn_id, outbound_json);
356        Ok((code.clone(), sender_player_id))
357    }
358
359    /// Valida el payload con `AntiCheatSession` (si aplica a la sala) y retransmite a los demás participantes.
360    pub fn validate_and_relay(
361        &mut self,
362        conn_id: ConnectionId,
363        payload: &Value,
364        outbound_json: &str,
365    ) -> Result<(String, RoomPlayerId), RoomError> {
366        let code = self
367            .conn_to_room
368            .get(&conn_id)
369            .cloned()
370            .ok_or(RoomError::NotInRoom)?;
371        let room = self.rooms.get_mut(&code).ok_or(RoomError::RoomNotFound)?;
372        let participant = room
373            .participants
374            .get(&conn_id)
375            .ok_or(RoomError::NotInRoom)?;
376        if participant.role == ParticipantRole::Spectator {
377            return Err(RoomError::SpectatorCannotRelay);
378        }
379        let sender_player_id = participant.player_id;
380
381        if let Some(ac) = room.anticheat.as_mut() {
382            if ac.validate_and_apply(sender_player_id, payload).is_err() {
383                return Err(RoomError::IllegalMove);
384            }
385        }
386
387        room.broadcast_except(conn_id, outbound_json);
388        Ok((code, sender_player_id))
389    }

391    /// Obtiene el `player_id` y código de sala de un cliente conectado.
392    pub fn participant_info(&self, conn_id: ConnectionId) -> Option<(String, RoomPlayerId)> {
393        let code = self.conn_to_room.get(&conn_id)?;
394        let room = self.rooms.get(code)?;
395        let participant = room.participants.get(&conn_id)?;
396        Some((code.clone(), participant.player_id))
397    }

399    /// Retira a un cliente de su sala actual (por `{"op": "leave"}` o desconexión)
400    /// y notifica a los participantes restantes.
401    ///
402    /// Si la sala queda vacía, se elimina de memoria automáticamente.
403    pub fn leave_room(&mut self, conn_id: ConnectionId) -> Option<(String, RoomPlayerId)> {
404        let code = self.conn_to_room.remove(&conn_id)?;
405        let mut should_remove_room = false;
406        let mut removed_player_id = None;
407
408        if let Some(room) = self.rooms.get_mut(&code) {
409            if let Some(pid) = room.remove_participant(conn_id) {
410                removed_player_id = Some(pid);
411                if pid > 0 {
412                    let notification = serde_json::json!({
413                        "op": "peer_left",
414                        "room_code": code,
415                        "player_id": pid,
416                    })
417                    .to_string();
418                    room.broadcast_except(conn_id, &notification);
419                }
420            }
421            should_remove_room = room.is_empty();
422        }
423
424        if should_remove_room {
425            self.rooms.remove(&code);
426        }
427
428        removed_player_id.map(|pid| (code, pid))
429    }

431    /// Cantidad de salas activas actualmente.
432    pub fn active_room_count(&self) -> usize {
433        self.rooms.len()
434    }

442    fn test_room_code_generation_format_and_no_ambiguous_chars() {
443        for _ in 0..200 {

454    #[test]
455    fn test_room_lifecycle_create_join_relay_leave() {
456        let mut manager = RoomManager::new();
457        let (tx1, mut rx1) = mpsc::unbounded_channel();
458        let (tx2, mut rx2) = mpsc::unbounded_channel();
459
460        let code = manager.create_room(100, tx1, Some("chess".to_string()), true);
461        assert_eq!(manager.active_room_count(), 1);
462
463        let (joined_code, player_id) = manager.join_room(200, tx2, &code).unwrap();
464        assert_eq!(joined_code, code);
465        assert_eq!(player_id, 2);
466
467        // Cliente 1 envía relay -> solo Cliente 2 lo recibe
468        manager
469            .relay_to_peers(100, r#"{"op":"relay","data":"e2e4"}"#)
470            .unwrap();
471        assert_eq!(rx2.try_recv().unwrap(), r#"{"op":"relay","data":"e2e4"}"#);
472        assert!(rx1.try_recv().is_err());
473
474        // Cliente 2 abandona la sala -> Cliente 1 recibe notificación peer_left
475        let (_, left_pid) = manager.leave_room(200).unwrap();
476        assert_eq!(left_pid, 2);
477        let msg = rx1.try_recv().unwrap();
478        let parsed: serde_json::Value = serde_json::from_str(&msg).unwrap();
479        assert_eq!(parsed["op"], "peer_left");
480        assert_eq!(parsed["player_id"], 2);
481        assert_eq!(manager.active_room_count(), 1);
482
483        // Cliente 1 abandona -> la sala se elimina por quedar vacía
484        manager.leave_room(100);
485        assert_eq!(manager.active_room_count(), 0);
486    }

488    #[test]
489    fn test_room_spectator_join_relay_and_limit() {
490        let mut manager = RoomManager::new();
491        let (tx_host, _rx_host) = mpsc::unbounded_channel();
492        let (tx_guest, _rx_guest) = mpsc::unbounded_channel();
493
494        let code = manager.create_room(1, tx_host, Some("chess".to_string()), true);
495        let (_, guest_pid) = manager.join_room(2, tx_guest, &code).unwrap();
496        assert_eq!(guest_pid, 2);
497
498        // Unir hasta 10 espectadores
499        let mut spec_rxs = Vec::new();
500        for i in 0..MAX_SPECTATORS_PER_ROOM {
501            let (tx_spec, rx_spec) = mpsc::unbounded_channel();
502            let conn_id = 1000 + (i as u64);
503            let (joined_code, player_count) =
504                manager.join_spectator(conn_id, tx_spec, &code).unwrap();
505            assert_eq!(joined_code, code);
506            assert_eq!(player_count, 2);
507            spec_rxs.push(rx_spec);
508        }
509
510        // El 11.º espectador supera el límite de 10 y recibe SpectatorLimit
511        let (tx_spec_overflow, _rx_spec_overflow) = mpsc::unbounded_channel();
512        let overflow_res = manager.join_spectator(2000, tx_spec_overflow, &code);
513        assert_eq!(overflow_res, Err(RoomError::SpectatorLimit));
514
515        // El host envía relay -> los espectadores lo reciben
516        manager
517            .relay_to_peers(1, r#"{"op":"relay","data":"e2e4"}"#)
518            .unwrap();
519        for rx_spec in &mut spec_rxs {
520            assert_eq!(
521                rx_spec.try_recv().unwrap(),
522                r#"{"op":"relay","data":"e2e4"}"#
523            );
524        }
525
526        // Un espectador intenta hacer relay -> recibe SpectatorCannotRelay
527        let illegal_msg = r#"{"op":"relay","data":"illegal"}"#;
528        let spec_relay_res = manager.relay_to_peers(1000, illegal_msg);
529        assert_eq!(spec_relay_res, Err(RoomError::SpectatorCannotRelay));
530    }
