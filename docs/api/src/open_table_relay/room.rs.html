1//! Gestión de salas y generación de códigos legibles para `open_table_relay`.
2//!
3//! Administra salas públicas/privadas en memoria y asigna códigos alfanuméricos
4//! de 4 caracteres sin caracteres ambiguos (`O`, `0`, `I`, `1`).
5
6use std::collections::HashMap;
7use tokio::sync::mpsc;
8
9/// Alfabeto legible de 32 caracteres: excluye `O`, `0`, `I`, `1` para evitar confusiones visuales.
10pub const ROOM_CODE_ALPHABET: &[u8; 32] = b"ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
11
12/// Longitud estándar de los códigos de sala.
13pub const ROOM_CODE_LEN: usize = 4;
14
15/// Canal de envío hacia un cliente WebSocket conectado.
16pub type ClientSender = mpsc::UnboundedSender<String>;
17
18/// Identificador único de conexión interna asignado por el servidor.
19pub type ConnectionId = u64;
20
21/// Identificador de jugador dentro de una sala (1 = host, 2..N = invitados).
22pub type RoomPlayerId = u32;
23
24/// Información de un participante conectado a una sala.
25#[derive(Debug, Clone)]
26pub struct Participant {
27    /// Identificador único de la conexión.
28    pub conn_id: ConnectionId,
29    /// Identificador de jugador dentro de la sala (1-indexed).
30    pub player_id: RoomPlayerId,
31    /// Canal para enviar mensajes JSON serializados al cliente.
32    pub sender: ClientSender,
33}
34
35/// Representa una sala activa en el servidor Relay.
36#[derive(Debug)]
37pub struct Room {
38    /// Código único de 4 caracteres de la sala.
39    pub code: String,
40    /// Identificador opcional del juego (ej. `"chess"`, `"go"`).
41    pub game_id: Option<String>,
42    /// Indica si la sala es pública o privada.
43    pub is_public: bool,
44    /// Contador incremental para asignar `player_id` a nuevos participantes.
45    next_player_id: RoomPlayerId,
46    /// Participantes activos indexados por `ConnectionId`.
47    pub participants: HashMap<ConnectionId, Participant>,
48}
49
50impl Room {
51    /// Crea una nueva sala con el anfitrión inicial (`player_id = 1`).
52    pub fn new(
53        code: String,
54        host_conn_id: ConnectionId,
55        host_sender: ClientSender,
56        game_id: Option<String>,
57        is_public: bool,
58    ) -> Self {
59        let mut participants = HashMap::new();
60        participants.insert(
61            host_conn_id,
62            Participant {
63                conn_id: host_conn_id,
64                player_id: 1,
65                sender: host_sender,
66            },
67        );
68        Self {
69            code,
70            game_id,
71            is_public,
72            next_player_id: 2,
73            participants,
74        }
75    }
76
77    /// Añade un participante a la sala y devuelve el `player_id` asignado.
78    pub fn add_participant(&mut self, conn_id: ConnectionId, sender: ClientSender) -> RoomPlayerId {
79        if let Some(existing) = self.participants.get(&conn_id) {
80            return existing.player_id;
81        }
82        let assigned_id = self.next_player_id;
83        self.next_player_id += 1;
84        self.participants.insert(
85            conn_id,
86            Participant {
87                conn_id,
88                player_id: assigned_id,
89                sender,
90            },
91        );
92        assigned_id
93    }
94
95    /// Elimina un participante de la sala y devuelve su `player_id` si estaba presente.
96    pub fn remove_participant(&mut self, conn_id: ConnectionId) -> Option<RoomPlayerId> {
97        self.participants.remove(&conn_id).map(|p| p.player_id)
98    }
99
100    /// Envía un mensaje JSON a todos los participantes de la sala excepto `exclude_conn`.
101    pub fn broadcast_except(&self, exclude_conn: ConnectionId, message: &str) {
102        for (&cid, participant) in &self.participants {
103            if cid != exclude_conn {
104                let _ = participant.sender.send(message.to_string());
105            }
106        }
107    }
108
109    /// Devuelve `true` si no quedan participantes conectados en la sala.
110    pub fn is_empty(&self) -> bool {
111        self.participants.is_empty()
112    }
113}
114
115/// Genera un código de sala aleatorio de 4 caracteres usando [`ROOM_CODE_ALPHABET`].
116pub fn generate_room_code() -> String {
117    let mut buf = [0u8; ROOM_CODE_LEN];
118    if getrandom::fill(&mut buf).is_err() {
119        // Fallback determinista seguro en caso de fallo del generador del SO
120        let ts = std::time::SystemTime::now()
121            .duration_since(std::time::UNIX_EPOCH)
122            .map(|d| d.as_nanos() as u64)
123            .unwrap_or(0xA5A5_5A5A);
124        for (i, byte) in buf.iter_mut().enumerate() {
125            *byte = ((ts >> (i * 8)) & 0xFF) as u8;
126        }
127    }
128    buf.iter()
129        .map(|&b| ROOM_CODE_ALPHABET[(b as usize) % ROOM_CODE_ALPHABET.len()] as char)
130        .collect()
131}
132
133/// Valida que un código de sala tenga exactamente 4 caracteres del alfabeto permitido.
134pub fn is_valid_room_code(code: &str) -> bool {
135    code.len() == ROOM_CODE_LEN
136        && code
137            .bytes()
138            .all(|b| ROOM_CODE_ALPHABET.contains(&b.to_ascii_uppercase()))
139}

141/// Gestor central de salas activas y membresía de conexiones.
142#[derive(Debug, Default)]
143pub struct RoomManager {
144    /// Salas activas por código (en mayúsculas).
145    rooms: HashMap<String, Room>,
146    /// Mapa inverso de `ConnectionId` al código de sala en el que participa.
147    conn_to_room: HashMap<ConnectionId, String>,
148}
149
150/// Errores posibles en las operaciones de sala.
151#[derive(Debug, Clone, PartialEq, Eq)]
152pub enum RoomError {
153    /// El código de sala solicitado no existe.
154    RoomNotFound,
155    /// El código de sala tiene un formato inválido.
156    InvalidRoomCode,
157    /// El cliente no se encuentra actualmente en ninguna sala.
158    NotInRoom,
159}
160
161impl RoomManager {
162    /// Crea un nuevo gestor de salas vacío.
163    pub fn new() -> Self {
164        Self::default()
165    }
166
167    /// Crea una nueva sala con un código único de 4 caracteres y registra al anfitrión.
168    pub fn create_room(
169        &mut self,
170        host_conn_id: ConnectionId,
171        host_sender: ClientSender,
172        game_id: Option<String>,
173        is_public: bool,
174    ) -> String {
175        // Si el cliente ya estaba en otra sala, lo sacamos primero limpiamente
176        let _ = self.leave_room(host_conn_id);
177
178        let code = loop {
179            let candidate = generate_room_code();
180            if !self.rooms.contains_key(&candidate) {
181                break candidate;
182            }
183        };
184
185        let room = Room::new(code.clone(), host_conn_id, host_sender, game_id, is_public);
186        self.rooms.insert(code.clone(), room);
187        self.conn_to_room.insert(host_conn_id, code.clone());
188        code
189    }
190
191    /// Une un cliente a una sala existente por su código.
192    ///
193    /// Devuelve `(room_code_normalizado, player_id_asignado)`.
194    pub fn join_room(
195        &mut self,
196        conn_id: ConnectionId,
197        sender: ClientSender,
198        raw_code: &str,
199    ) -> Result<(String, RoomPlayerId), RoomError> {
200        let normalized = raw_code.trim().to_ascii_uppercase();
201        if !is_valid_room_code(&normalized) {
202            return Err(RoomError::InvalidRoomCode);
203        }
204
205        if !self.rooms.contains_key(&normalized) {
206            return Err(RoomError::RoomNotFound);
207        }
208
209        // Si estaba en otra sala distinta, salir primero
210        if let Some(prev_code) = self.conn_to_room.get(&conn_id) {
211            if prev_code != &normalized {
212                let _ = self.leave_room(conn_id);
213            }
214        }
215
216        let room = self
217            .rooms
218            .get_mut(&normalized)
219            .ok_or(RoomError::RoomNotFound)?;
220        let player_id = room.add_participant(conn_id, sender);
221        self.conn_to_room.insert(conn_id, normalized.clone());
222
223        Ok((normalized, player_id))
224    }
225
226    /// Retransmite un mensaje JSON a los demás miembros de la sala donde está `conn_id`.
227    ///
228    /// Devuelve el `(room_code, player_id)` del emisor.
229    pub fn relay_to_peers(
230        &self,
231        conn_id: ConnectionId,
232        outbound_json: &str,
233    ) -> Result<(String, RoomPlayerId), RoomError> {
234        let code = self
235            .conn_to_room
236            .get(&conn_id)
237            .ok_or(RoomError::NotInRoom)?;
238        let room = self.rooms.get(code).ok_or(RoomError::RoomNotFound)?;
239        let sender_player_id = room
240            .participants
241            .get(&conn_id)
242            .map(|p| p.player_id)
243            .ok_or(RoomError::NotInRoom)?;
244
245        room.broadcast_except(conn_id, outbound_json);
246        Ok((code.clone(), sender_player_id))
247    }
248
249    /// Obtiene el `player_id` y código de sala de un cliente conectado.
250    pub fn participant_info(&self, conn_id: ConnectionId) -> Option<(String, RoomPlayerId)> {
251        let code = self.conn_to_room.get(&conn_id)?;
252        let room = self.rooms.get(code)?;
253        let participant = room.participants.get(&conn_id)?;
254        Some((code.clone(), participant.player_id))
255    }
256
257    /// Retira a un cliente de su sala actual (por `{"op": "leave"}` o desconexión)
258    /// y notifica a los participantes restantes.
259    ///
260    /// Si la sala queda vacía, se elimina de memoria automáticamente.
261    pub fn leave_room(&mut self, conn_id: ConnectionId) -> Option<(String, RoomPlayerId)> {
262        let code = self.conn_to_room.remove(&conn_id)?;
263        let mut should_remove_room = false;
264        let mut removed_player_id = None;
265
266        if let Some(room) = self.rooms.get_mut(&code) {
267            if let Some(pid) = room.remove_participant(conn_id) {
268                removed_player_id = Some(pid);
269                let notification = serde_json::json!({
270                    "op": "peer_left",
271                    "room_code": code,
272                    "player_id": pid,
273                })
274                .to_string();
275                room.broadcast_except(conn_id, &notification);
276            }
277            should_remove_room = room.is_empty();
278        }
279
280        if should_remove_room {
281            self.rooms.remove(&code);
282        }
283
284        removed_player_id.map(|pid| (code, pid))
285    }
286
287    /// Cantidad de salas activas actualmente.
288    pub fn active_room_count(&self) -> usize {
289        self.rooms.len()
290    }
291}

293#[cfg(test)]
294mod tests {
295    use super::*;

297    #[test]
298    fn test_room_code_generation_format_and_no_ambiguous_chars() {
299        for _ in 0..200 {
300            let code = generate_room_code();
301            assert_eq!(code.len(), 4);
302            assert!(is_valid_room_code(&code));
303            assert!(!code.contains('O'));
304            assert!(!code.contains('0'));
305            assert!(!code.contains('I'));
306            assert!(!code.contains('1'));
307        }
308    }
309
310    #[test]
311    fn test_room_lifecycle_create_join_relay_leave() {
312        let mut manager = RoomManager::new();
313        let (tx1, mut rx1) = mpsc::unbounded_channel();
314        let (tx2, mut rx2) = mpsc::unbounded_channel();
315
316        let code = manager.create_room(100, tx1, Some("chess".to_string()), true);
317        assert_eq!(manager.active_room_count(), 1);
318
319        let (joined_code, player_id) = manager.join_room(200, tx2, &code).unwrap();
320        assert_eq!(joined_code, code);
321        assert_eq!(player_id, 2);
322
323        // Cliente 1 envía relay -> solo Cliente 2 lo recibe
324        manager
325            .relay_to_peers(100, r#"{"op":"relay","data":"e2e4"}"#)
326            .unwrap();
327        assert_eq!(rx2.try_recv().unwrap(), r#"{"op":"relay","data":"e2e4"}"#);
328        assert!(rx1.try_recv().is_err());
329
330        // Cliente 2 abandona la sala -> Cliente 1 recibe notificación peer_left
331        let (_, left_pid) = manager.leave_room(200).unwrap();
332        assert_eq!(left_pid, 2);
333        let msg = rx1.try_recv().unwrap();
334        let parsed: serde_json::Value = serde_json::from_str(&msg).unwrap();
335        assert_eq!(parsed["op"], "peer_left");
336        assert_eq!(parsed["player_id"], 2);
337        assert_eq!(manager.active_room_count(), 1);
338
339        // Cliente 1 abandona -> la sala se elimina por quedar vacía
340        manager.leave_room(100);
341        assert_eq!(manager.active_room_count(), 0);
342    }
