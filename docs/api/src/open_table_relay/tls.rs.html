1//! Configuración y carga de certificados PEM para WebSocket Secure (`wss://`).
2//!
3//! Lee `TLS_CERT_PATH` y `TLS_KEY_PATH` del entorno. Si no están definidas,
4//! el servidor Relay arranca en modo plain WS (`ws://`) para desarrollo local.
5
6use std::fs;
7use std::io::{self, ErrorKind};
8use std::path::{Path, PathBuf};
9
10/// Configuración TLS cargada desde archivos PEM.
11#[derive(Debug, Clone, PartialEq, Eq)]
12pub struct TlsConfig {
13    /// Ruta al certificado PEM.
14    pub cert_path: PathBuf,
15    /// Ruta a la clave privada PEM.
16    pub key_path: PathBuf,
17    /// Bloques de certificados PEM validados.
18    pub cert_pem_blocks: Vec<String>,
19    /// Bloque de clave privada PEM validado.
20    pub key_pem_block: String,
21}
22
23impl TlsConfig {
24    /// Carga la configuración TLS desde las variables de entorno `TLS_CERT_PATH` y `TLS_KEY_PATH`.
25    /// Devuelve `Ok(None)` si alguna de las dos variables no está definida (modo plain WS).
26    pub fn from_env() -> io::Result<Option<Self>> {
27        let cert_var = std::env::var("TLS_CERT_PATH").ok();
28        let key_var = std::env::var("TLS_KEY_PATH").ok();
29
30        match (cert_var, key_var) {
31            (Some(cert), Some(key)) if !cert.trim().is_empty() && !key.trim().is_empty() => {
32                Self::from_paths(cert, key).map(Some)
33            }
34            _ => Ok(None),
35        }
36    }
37
38    /// Carga y valida los archivos PEM de certificado y clave privada desde disco.
39    pub fn from_paths(cert_path: impl AsRef<Path>, key_path: impl AsRef<Path>) -> io::Result<Self> {
40        let c_path = cert_path.as_ref().to_path_buf();
41        let k_path = key_path.as_ref().to_path_buf();
42
43        let cert_raw = fs::read_to_string(&c_path)?;
44        let key_raw = fs::read_to_string(&k_path)?;
45
46        let cert_pem_blocks = parse_pem_certificates(&cert_raw)?;
47        let key_pem_block = parse_pem_private_key(&key_raw)?;
48
49        Ok(Self {
50            cert_path: c_path,
51            key_path: k_path,
52            cert_pem_blocks,
53            key_pem_block,
54        })
55    }
56}
57
58/// Extrae y valida bloques `-----BEGIN CERTIFICATE-----` desde un contenido PEM.
59pub fn parse_pem_certificates(pem_str: &str) -> io::Result<Vec<String>> {
60    let mut certs = Vec::new();
61    let begin_marker = "-----BEGIN CERTIFICATE-----";
62    let end_marker = "-----END CERTIFICATE-----";
63
64    let mut rest = pem_str;
65    while let Some(start) = rest.find(begin_marker) {
66        let after_start = &rest[start..];
67        let Some(end_rel) = after_start.find(end_marker) else {
68            return Err(io::Error::new(
69                ErrorKind::InvalidData,
70                "unclosed PEM certificate block",
71            ));
72        };
73        let block = &after_start[..end_rel + end_marker.len()];
74        certs.push(block.to_string());
75        rest = &after_start[end_rel + end_marker.len()..];
76    }
77
78    if certs.is_empty() {
79        return Err(io::Error::new(
80            ErrorKind::InvalidData,
81            "no PEM certificates found",
82        ));
83    }
84    Ok(certs)
85}

87/// Extrae y valida un bloque de clave privada (`PRIVATE KEY` o `RSA PRIVATE KEY`) desde PEM.
88pub fn parse_pem_private_key(pem_str: &str) -> io::Result<String> {
89    for (begin, end) in [
90        ("-----BEGIN PRIVATE KEY-----", "-----END PRIVATE KEY-----"),
91        (
92            "-----BEGIN RSA PRIVATE KEY-----",
93            "-----END RSA PRIVATE KEY-----",
94        ),
95        (
96            "-----BEGIN EC PRIVATE KEY-----",
97            "-----END EC PRIVATE KEY-----",
98        ),
99    ] {
100        if let Some(start) = pem_str.find(begin) {
101            let after_start = &pem_str[start..];
102            if let Some(end_rel) = after_start.find(end) {
103                return Ok(after_start[..end_rel + end.len()].to_string());
104            }
105        }
106    }
107
108    Err(io::Error::new(
109        ErrorKind::InvalidData,
110        "no valid PEM private key found",
111    ))
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117
118    #[test]
119    fn test_parse_pem_certificates_and_private_key() {
120        let cert_pem = "-----BEGIN CERTIFICATE-----\nMIIB...\n-----END CERTIFICATE-----\n";
121        let key_pem = "-----BEGIN PRIVATE KEY-----\nMIIE...\n-----END PRIVATE KEY-----\n";
122
123        let certs = parse_pem_certificates(cert_pem).unwrap();
124        assert_eq!(certs.len(), 1);
125
126        let key = parse_pem_private_key(key_pem).unwrap();
127        assert!(key.contains("BEGIN PRIVATE KEY"));
128    }
