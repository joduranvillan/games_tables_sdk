1//! Servidor HTTP ligero en el puerto 8081 para consultar rankings ELO y métricas Prometheus.
2//!
3//! Endpoints soportados:
4//! - `GET /rankings/:game_id` -> JSON con el top-50 de jugadores por ELO para ese juego.
5//! - `GET /player/:player_id` -> JSON con todos los ratings del jugador por juego.
6//! - `GET /metrics` -> Métricas operacionales en formato Prometheus (`text/plain; version=0.0.4`).
7
8use crate::elo::EloStore;
9use crate::metrics::global_metrics;
10use std::sync::Arc;
11use tokio::io::{AsyncReadExt, AsyncWriteExt};
12use tokio::net::{TcpListener, TcpStream};
13use tokio::sync::Mutex;
14
15/// Ejecuta el servidor HTTP de consulta de rankings ELO sobre el `TcpListener` indicado.
16pub async fn run_http_api_server(
17    listener: TcpListener,
18    elo_store: Arc<Mutex<EloStore>>,
19) -> std::io::Result<()> {
20    loop {
21        let (stream, _) = listener.accept().await?;
22        let store_clone = Arc::clone(&elo_store);
23        tokio::spawn(async move {
24            let _ = handle_http_connection(stream, store_clone).await;
25        });
26    }
27}
28
29async fn handle_http_connection(
30    mut stream: TcpStream,
31    elo_store: Arc<Mutex<EloStore>>,
32) -> std::io::Result<()> {
33    let mut buf = [0u8; 4096];
34    let n = stream.read(&mut buf).await?;
35    if n == 0 {
36        return Ok(());
37    }
38
39    let request_str = String::from_utf8_lossy(&buf[..n]);
40    let first_line = request_str.lines().next().unwrap_or("");
41    let mut parts = first_line.split_whitespace();
42    let method = parts.next().unwrap_or("");
43    let raw_path = parts.next().unwrap_or("");
44    let path = raw_path.split('?').next().unwrap_or("");
45
46    let mut content_type = "application/json";
47    let (status_line, body_payload) = if method == "GET" {
48        if path == "/metrics" {
49            content_type = "text/plain; version=0.0.4; charset=utf-8";
50            ("HTTP/1.1 200 OK", global_metrics().render_prometheus_text())
51        } else if let Some(game_id) = path.strip_prefix("/rankings/") {
52            let clean_game = game_id.trim_matches('/');
53            let store = elo_store.lock().await;
54            let rankings = store.get_rankings(clean_game, 50);
55            let body = serde_json::to_string(&rankings).unwrap_or_else(|_| "[]".to_string());
56            ("HTTP/1.1 200 OK", body)
57        } else if let Some(player_id) = path.strip_prefix("/player/") {
58            let clean_player = player_id.trim_matches('/');
59            let store = elo_store.lock().await;
60            let ratings = store.get_player_ratings(clean_player);
61            let body = serde_json::to_string(&ratings).unwrap_or_else(|_| "[]".to_string());
62            ("HTTP/1.1 200 OK", body)
63        } else {
64            (
65                "HTTP/1.1 404 Not Found",
66                r#"{"error":"not found"}"#.to_string(),
67            )
68        }
69    } else {
70        (
71            "HTTP/1.1 405 Method Not Allowed",
72            r#"{"error":"method not allowed"}"#.to_string(),
73        )
74    };
75
76    let response = format!(
77        "{status_line}\r\nContent-Type: {content_type}\r\nAccess-Control-Allow-Origin: *\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body_payload}",
78        body_payload.len()
79    );
80    stream.write_all(response.as_bytes()).await?;
81    stream.flush().await
82}
