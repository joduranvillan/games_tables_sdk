4//! Microservidor standalone de salas, retransmisión WebSocket y servicio ELO para OpenTable SDK.
5//!
6//! Protocolo JSON WebSocket (puerto 8080):
7//! - `{"op": "create"}` -> `{"op": "created", "room_code": "XKCD"}`
8//! - `{"op": "join", "room_code": "XKCD"}` -> `{"op": "joined", "player_id": 2}`
9//! - `{"op": "relay", "data": "..."}` -> retransmite el payload a los demás participantes
10//! - `{"op": "report_result", "winner": "player1", "loser": "player2", "game_id": "chess"}` -> actualiza ELO (K=32)
11//! - `{"op": "leave"}` / desconexión -> notifica a los participantes restantes (`peer_left`)
12//!
13//! API HTTP (puerto 8081):
14//! - `GET /rankings/:game_id` -> JSON con top-50 jugadores por ELO
15//! - `GET /player/:player_id` -> JSON con todos los ratings del jugador por juego

74/// Solicitud entrante enviada por un cliente WebSocket al servidor Relay.
75#[derive(Debug, Deserialize)]
76#[serde(tag = "op", rename_all = "snake_case")]
77pub enum RelayRequest {
78    /// Autentica la conexión mediante un token JWT (`HS256`).
79    Auth {
80        /// Token JWT firmado con `JWT_SECRET`.
81        token: String,
82    },
83    /// Crea una nueva sala y registra al cliente como anfitrión (`player_id = 1`).
84    #[serde(alias = "create_room")]
85    Create {
86        /// Identificador opcional del juego.
87        #[serde(default)]
88        game_id: Option<String>,
89        /// Indica si la sala es pública (por defecto `true`).
90        #[serde(default = "default_true")]
91        is_public: bool,
92        /// Tiempo máximo por turno en segundos (`120` por defecto; presets: `30`, `60`, `120`, `0` en salas privadas).
93        #[serde(default)]
94        turn_timeout_secs: Option<u64>,
95    },
96    /// Encola al jugador en una partida de matchmaking público (`turn_timeout_secs = 60`).
97    #[serde(alias = "matchmaking")]
98    Queue {
99        /// Identificador opcional del juego.
100        #[serde(default)]
101        game_id: Option<String>,
102    },
103    /// Une al cliente a una sala existente mediante su código.
104    Join {
105        /// Código de la sala.
106        room_code: String,
107    },
108    /// Reconecta a un jugador desconectado a su asiento original dentro de la ventana de gracia de 90 segundos.
109    #[serde(alias = "rejoin")]
110    RejoinRoom {
111        /// Código de la sala activa.
112        room_code: String,
113        /// Identificador canónico opcional del jugador (si se omite, se emplea el del JWT autenticado).
114        #[serde(default)]
115        player_id: Option<String>,
116    },
117    /// Une al cliente como espectador de solo lectura a una sala existente.
118    JoinSpectator {
119        /// Código de la sala.
120        room_code: String,
121    },
122    /// Retransmite `data` a todos los demás participantes de la sala actual.
123    Relay {
124        /// Payload arbitrario a retransmitir.
125        data: Value,
126    },
127    /// Reporta el resultado de una partida para actualizar el ELO persistente (K=32).
128    ReportResult {
129        /// Identificador del jugador ganador.
130        winner: String,
131        /// Identificador del jugador perdedor.
132        loser: String,
133        /// Identificador del juego (ej. `"chess"`).
134        game_id: String,
135    },
136    /// Crea un nuevo torneo de eliminación simple (4, 8 o 16 jugadores).
137    CreateTournament {
138        /// Identificador del juego.
139        game_id: String,
140        /// Capacidad máxima de participantes (`4`, `8` o `16`).
141        max_players: usize,
142    },
143    /// Une a un participante a un torneo existente.
144    JoinTournament {
145        /// Identificador del torneo (ej. `"T-AB12"`).
146        tournament_id: String,
147        /// Identificador opcional del jugador.
148        #[serde(default)]
149        player_id: Option<String>,
150    },
151    /// Reporta el ganador de un enfrentamiento de torneo y avanza a la siguiente ronda.
152    ReportTournamentMatch {
153        /// Identificador del torneo.
154        tournament_id: String,
155        /// Índice de ronda.
156        round: usize,
157        /// Índice de enfrentamiento dentro de la ronda.
158        match_index: usize,
159        /// Identificador del jugador ganador.
160        winner: String,
161    },
162    /// Abandona voluntariamente la sala actual.
163    Leave,
164    /// Se rinde en la partida activa y adjudica inmediatamente la victoria al oponente.
165    Resign,
166    /// Ping de mantenimiento de conexión.
167    Ping,
168}

171    true

335/// Espera una señal de parada del sistema (`SIGINT` / `Ctrl+C` o `SIGTERM` en contenedores Docker/Unix).
336pub async fn wait_for_shutdown_signal() {
337    #[cfg(unix)]
338    {
339        use tokio::signal::unix::{signal, SignalKind};
340        match signal(SignalKind::terminate()) {
341            Ok(mut sigterm) => {
342                tokio::select! {
343                    _ = tokio::signal::ctrl_c() => {},
344                    _ = sigterm.recv() => {},
345                }
346            }
347            Err(_) => {
348                let _ = tokio::signal::ctrl_c().await;
349            }
350        }
351    }
352    #[cfg(not(unix))]
353    {
354        let _ = tokio::signal::ctrl_c().await;
355    }
356}

358/// Ejecuta el protocolo de apagado ordenado (Graceful Shutdown):
359/// 1. Envía `{"op": "server_shutting_down"}` a todos los clientes conectados.
360/// 2. Ejecuta flush completo y sincronización de almacenamiento de SQLite (`wal_checkpoint_and_flush`).
361pub async fn perform_graceful_shutdown(state: &RelaySharedState) -> std::io::Result<()> {
362    if let Ok(shutdown_json) = serde_json::to_string(&RelayResponse::ServerShuttingDown) {
363        let mgr = state.manager.lock().await;
364        mgr.broadcast_all_rooms(&shutdown_json);
365    }
366    tokio::time::sleep(Duration::from_millis(25)).await;
367    let mut store = state.elo_store.lock().await;
368    store.wal_checkpoint_and_flush()
369}

416/// Inicia el servidor Relay a partir de un [`RelaySharedState`] preconfigurado y ejecuta
417/// apagado ordenado cuando se resuelve la señal `shutdown_signal`.
418pub async fn run_relay_server_with_state_until<F>(
419    listener: TcpListener,
420    state: RelaySharedState,
421    shutdown_signal: F,
422) -> std::io::Result<()>
423where
424    F: std::future::Future<Output = ()>,
425{
426    let next_conn_id = Arc::new(AtomicU64::new(1));
427    let accept_state = state.clone();
428    let ticker_state = state.clone();
429    let gc_state = state.clone();
430
431    let accept_loop = async move {
432        loop {
433            let (stream, peer_addr) = listener.accept().await?;
434            let conn_id = next_conn_id.fetch_add(1, Ordering::Relaxed);
435            let state_clone = accept_state.clone();
436
437            tokio::spawn(async move {
438                handle_connection(stream, peer_addr.ip(), conn_id, state_clone).await;
439            });
440        }
441        #[allow(unreachable_code)]
442        Ok::<(), std::io::Error>(())
443    };
444
445    // Tarea periódica cada 1s que verifica timeouts de turno y desconexiones expiradas
446    let periodic_watchdog = async move {
447        let mut interval = tokio::time::interval(Duration::from_secs(1));
448        loop {
449            interval.tick().await;
450            let outcomes = {
451                let mut mgr = ticker_state.manager.lock().await;
452                let mut out = mgr.expire_turn_timeouts();
453                out.extend(mgr.expire_disconnected_seats());
454                global_metrics().set_rooms_active(mgr.active_room_count());
455                out
456            };
457            for outcome in outcomes {
458                persist_and_broadcast_elo_outcome(outcome, &ticker_state).await;
459            }
460        }
461    };
462
463    // Recolector periódico cada 60s para salas inactivas (>30m) y cada 5m para RateLimitManager
464    let periodic_collector = async move {
465        let mut room_gc_interval =
466            tokio::time::interval(Duration::from_secs(room::ROOM_PRUNE_INTERVAL_SECS));
467        let mut rate_limit_gc_interval = tokio::time::interval(Duration::from_secs(
468            rate_limit::RATE_LIMIT_CLEANUP_INTERVAL_SECS,
469        ));
470        loop {
471            tokio::select! {
472                _ = room_gc_interval.tick() => {
473                    let mut mgr = gc_state.manager.lock().await;
474                    mgr.prune_stale_rooms(Duration::from_secs(room::DEFAULT_MAX_ROOM_IDLE_SECS));
475                    global_metrics().set_rooms_active(mgr.active_room_count());
476                }
477                _ = rate_limit_gc_interval.tick() => {
478                    let mut limiter = gc_state.rate_limiter.lock().await;
479                    limiter.prune_expired_now();
480                }
481            }
482        }
483    };
484
485    tokio::select! {
486        res = accept_loop => res,
487        _ = periodic_watchdog => Ok(()),
488        _ = periodic_collector => Ok(()),
489        _ = shutdown_signal => {
490            perform_graceful_shutdown(&state).await?;
491            Ok(())
492        }
493    }
494}

496async fn handle_connection(
497    stream: TcpStream,
498    peer_ip: IpAddr,
499    conn_id: ConnectionId,
500    state: RelaySharedState,
501) {
502    {
503        let mut limiter = state.rate_limiter.lock().await;
504        if limiter.is_ip_banned(peer_ip) {
505            return;
506        }
507    }
508
509    let ws_stream = match tokio_tungstenite::accept_async(stream).await {
510        Ok(ws) => ws,
511        Err(_) => return,
512    };
513
514    global_metrics().inc_connections_active();
515
516    let (mut ws_sink, mut ws_source) = ws_stream.split();
517    let (tx, mut rx) = mpsc::unbounded_channel::<String>();
518
519    {
520        let mut mgr = state.manager.lock().await;
521        mgr.register_client(conn_id, tx.clone());
522    }
523
524    let writer_task = tokio::spawn(async move {
525        while let Some(msg) = rx.recv().await {
526            if ws_sink.send(Message::Text(msg.into())).await.is_err() {
527                break;
528            }
529        }
530    });
531
532    let mut canonical_player_id: Option<String> = None;
533
534    if state.auth_config.is_enabled() {
535        let first_frame = tokio::time::timeout(
536            Duration::from_secs(AUTH_HANDSHAKE_TIMEOUT_SECS),
537            ws_source.next(),
538        )
539        .await;
540
541        let Some(Some(Ok(Message::Text(first_text)))) = first_frame.ok() else {
542            send_response(
543                &tx,
544                &RelayResponse::AuthError {
545                    reason: "auth_timeout".to_string(),
546                },
547            );
548            tokio::time::sleep(Duration::from_millis(20)).await;
549            writer_task.abort();
550            return;
551        };
552
553        let Ok(RelayRequest::Auth { token }) = serde_json::from_str::<RelayRequest>(&first_text)
554        else {
555            send_response(
556                &tx,
557                &RelayResponse::AuthError {
558                    reason: "auth_required".to_string(),
559                },
560            );
561            tokio::time::sleep(Duration::from_millis(20)).await;
562            writer_task.abort();
563            return;
564        };
565
566        match state.auth_config.verify_token(&token) {
567            Ok(claims) => {
568                canonical_player_id = Some(claims.sub.clone());
569                send_response(
570                    &tx,
571                    &RelayResponse::AuthOk {
572                        player_id: claims.sub,
573                    },
574                );
575            }
576            Err(err) => {
577                send_response(
578                    &tx,
579                    &RelayResponse::AuthError {
580                        reason: err.reason().to_string(),
581                    },
582                );
583                tokio::time::sleep(Duration::from_millis(20)).await;
584                writer_task.abort();
585                return;
586            }
587        }
588    }
589
590    while let Some(Ok(msg)) = ws_source.next().await {
591        match msg {
592            Message::Text(text) => {
593                let current_room = {
594                    let mgr = state.manager.lock().await;
595                    mgr.participant_info(conn_id).map(|(code, _)| code)
596                };
597                let decision = {
598                    let mut limiter = state.rate_limiter.lock().await;
599                    limiter.check_message(peer_ip, current_room.as_deref())
600                };
601                if let RateLimitDecision::Limited {
602                    retry_after_ms,
603                    disconnect,
604                } = decision
605                {
606                    send_response(&tx, &RelayResponse::RateLimited { retry_after_ms });
607                    if disconnect {
608                        tokio::time::sleep(Duration::from_millis(20)).await;
609                        break;
610                    }
611                    continue;
612                }
613
614                process_text_message(&text, conn_id, &mut canonical_player_id, &tx, &state).await;
615            }
616            Message::Ping(payload) => {
617                let _ = tx.send(
618                    serde_json::to_string(&RelayResponse::Pong)
619                        .unwrap_or_else(|_| r#"{"op":"pong"}"#.to_string()),
620                );
621                let _ = payload;
622            }
623            Message::Close(_) => {
624                break;
625            }
626            _ => {}
627        }
628    }
629
630    let maybe_grace = {
631        let mut mgr = state.manager.lock().await;
632        mgr.unregister_client(conn_id);
633        let disconnected = mgr.disconnect_participant(conn_id);
634        global_metrics().set_rooms_active(mgr.active_room_count());
635        disconnected.map(|_| mgr.rejoin_grace_duration())
636    };
637
638    if let Some(grace_duration) = maybe_grace {
639        let state_for_timer = state.clone();
640        tokio::spawn(async move {
641            tokio::time::sleep(grace_duration).await;
642            let outcomes = {
643                let mut mgr = state_for_timer.manager.lock().await;
644                let expired = mgr.expire_disconnected_seats();
645                global_metrics().set_rooms_active(mgr.active_room_count());
646                expired
647            };
648            for outcome in outcomes {
649                persist_and_broadcast_elo_outcome(outcome, &state_for_timer).await;
650            }
651        });
652    }
653
654    global_metrics().dec_connections_active();
655    writer_task.abort();
656}

658async fn process_text_message(
659    raw_text: &str,
660    conn_id: ConnectionId,
661    canonical_player_id: &mut Option<String>,
662    tx: &ClientSender,
663    state: &RelaySharedState,
664) {
665    let parsed: Result<RelayRequest, _> = serde_json::from_str(raw_text);
666    let request = match parsed {
667        Ok(req) => req,
668        Err(err) => {
669            send_response(
670                tx,
671                &RelayResponse::error_message(format!("invalid json request: {err}")),
672            );
673            return;
674        }
675    };
676
677    match request {
678        RelayRequest::Auth { token } => match state.auth_config.verify_token(&token) {
679            Ok(claims) => {
680                *canonical_player_id = Some(claims.sub.clone());
681                {
682                    let mut mgr = state.manager.lock().await;
683                    mgr.set_authenticated_player_id(conn_id, canonical_player_id.clone());
684                }
685                send_response(
686                    tx,
687                    &RelayResponse::AuthOk {
688                        player_id: claims.sub,
689                    },
690                );
691            }
692            Err(err) => {
693                send_response(
694                    tx,
695                    &RelayResponse::AuthError {
696                        reason: err.reason().to_string(),
697                    },
698                );
699            }
700        },
701        RelayRequest::Create {
702            game_id,
703            is_public,
704            turn_timeout_secs,
705        } => {
706            global_metrics().inc_messages_total(game_id.as_deref());
707            let mut mgr = state.manager.lock().await;
708            let room_code = mgr.create_room_with_timeout(
709                conn_id,
710                tx.clone(),
711                game_id,
712                is_public,
713                turn_timeout_secs,
714            );
715            mgr.set_authenticated_player_id(conn_id, canonical_player_id.clone());
716            let seed = mgr.room_seed(&room_code).unwrap_or(42);
717            global_metrics().set_rooms_active(mgr.active_room_count());
718            send_response(tx, &RelayResponse::Created { room_code, seed });
719        }
720        RelayRequest::Queue { game_id } => {
721            global_metrics().inc_messages_total(game_id.as_deref());
722            let mut mgr = state.manager.lock().await;
723            let (room_code, player_id, is_new) =
724                mgr.queue_matchmaking(conn_id, tx.clone(), game_id);
725            mgr.set_authenticated_player_id(conn_id, canonical_player_id.clone());
726            let seed = mgr.room_seed(&room_code).unwrap_or(42);
727            global_metrics().set_rooms_active(mgr.active_room_count());
728            if is_new {
729                send_response(tx, &RelayResponse::Created { room_code, seed });
730            } else {
731                send_response(tx, &RelayResponse::Joined { player_id, seed });
732                if let Ok(peer_notice) =
733                    serde_json::to_string(&RelayResponse::PeerJoined { player_id })
734                {
735                    let _ = mgr.relay_to_peers(conn_id, &peer_notice);
736                }
737            }
738        }
739        RelayRequest::Join { room_code } => {
740            let mut mgr = state.manager.lock().await;
741            match mgr.join_room(conn_id, tx.clone(), &room_code) {
742                Ok((normalized_code, player_id)) => {
743                    mgr.set_authenticated_player_id(conn_id, canonical_player_id.clone());
744                    let seed = mgr.room_seed(&normalized_code).unwrap_or(42);
745                    send_response(tx, &RelayResponse::Joined { player_id, seed });
746                    if let Ok(peer_notice) =
747                        serde_json::to_string(&RelayResponse::PeerJoined { player_id })
748                    {
749                        let _ = mgr.relay_to_peers(conn_id, &peer_notice);
750                    }
751                }
752                Err(RoomError::InvalidRoomCode) => {
753                    send_response(
754                        tx,
755                        &RelayResponse::error_message("invalid room code format"),
756                    );
757                }
758                Err(RoomError::RoomNotFound) => {
759                    send_response(tx, &RelayResponse::error_message("room not found"));
760                }
761                Err(_) => {
762                    send_response(tx, &RelayResponse::error_message("not in room"));
763                }
764            }
765        }
766        RelayRequest::RejoinRoom {
767            room_code,
768            player_id,
769        } => {
770            if let (Some(auth_id), Some(req_id)) = (&canonical_player_id, &player_id) {
771                let trimmed = req_id.trim();
772                if !trimmed.is_empty() && trimmed != auth_id {
773                    send_response(tx, &RelayResponse::error_reason("seat_not_found"));
774                    return;
775                }
776            }
777            let effective_id = canonical_player_id.clone().or(player_id);
778            let mut expired_outcomes = Vec::new();
779            let rejoin_res = {
780                let mut mgr = state.manager.lock().await;
781                let res = mgr.rejoin_room(conn_id, tx.clone(), &room_code, effective_id.as_deref());
782                if matches!(res, Err(RoomError::RejoinExpired)) {
783                    expired_outcomes = mgr.expire_disconnected_seats();
784                }
785                res
786            };
787            for outcome in expired_outcomes {
788                persist_and_broadcast_elo_outcome(outcome, state).await;
789            }
790            match rejoin_res {
791                Ok(outcome) => {
792                    if canonical_player_id.is_none() && effective_id.is_some() {
793                        *canonical_player_id = effective_id;
794                    }
795                    send_response(
796                        tx,
797                        &RelayResponse::Rejoined {
798                            room_code: outcome.room_code,
799                            player_id: outcome.player_id,
800                            seed: outcome.seed,
801                            commands: outcome.commands,
802                        },
803                    );
804                }
805                Err(RoomError::RejoinExpired) => {
806                    send_response(tx, &RelayResponse::error_reason("rejoin_expired"));
807                }
808                Err(RoomError::InvalidRoomCode) => {
809                    send_response(
810                        tx,
811                        &RelayResponse::error_message("invalid room code format"),
812                    );
813                }
814                Err(RoomError::RoomNotFound) => {
815                    send_response(tx, &RelayResponse::error_message("room not found"));
816                }
817                Err(_) => {
818                    send_response(tx, &RelayResponse::error_reason("seat_not_found"));
819                }
820            }
821        }
822        RelayRequest::JoinSpectator { room_code } => {
823            let mut mgr = state.manager.lock().await;
824            match mgr.join_spectator(conn_id, tx.clone(), &room_code) {
825                Ok((_normalized_code, player_count)) => {
826                    send_response(tx, &RelayResponse::Spectating { player_count });
827                }
828                Err(RoomError::SpectatorLimit) => {
829                    send_response(tx, &RelayResponse::error_reason("spectator_limit"));
830                }
831                Err(RoomError::InvalidRoomCode) => {
832                    send_response(
833                        tx,
834                        &RelayResponse::error_message("invalid room code format"),
835                    );
836                }
837                Err(RoomError::RoomNotFound) => {
838                    send_response(tx, &RelayResponse::error_message("room not found"));
839                }
840                Err(_) => {
841                    send_response(tx, &RelayResponse::error_message("not in room"));
842                }
843            }
844        }
845        RelayRequest::Relay { data } => {
846            let mut maybe_outcome = None;
847            {
848                let mut mgr = state.manager.lock().await;
849                let from_player = mgr.participant_info(conn_id).map(|(_, pid)| pid);
850                let outbound = RelayResponse::Relay {
851                    data: data.clone(),
852                    from: from_player,
853                };
854                if let Ok(serialized) = serde_json::to_string(&outbound) {
855                    match mgr.validate_and_relay(conn_id, &data, &serialized) {
856                        Ok(_) => {
857                            maybe_outcome = mgr.claim_authoritative_outcome(conn_id);
858                        }
859                        Err(RoomError::SpectatorCannotRelay) => {
860                            send_response(
861                                tx,
862                                &RelayResponse::error_reason("spectator_cannot_relay"),
863                            );
864                        }
865                        Err(RoomError::IllegalMove) => {
866                            send_response(
867                                tx,
868                                &RelayResponse::MoveRejected {
869                                    reason: "illegal_move".to_string(),
870                                },
871                            );
872                        }
873                        Err(_) => {
874                            send_response(
875                                tx,
876                                &RelayResponse::error_message("client is not in a room"),
877                            );
878                        }
879                    }
880                }
881            }
882            if let Some(outcome) = maybe_outcome {
883                persist_and_broadcast_elo_outcome(outcome, state).await;
884            }
885        }
886        RelayRequest::Resign => {
887            let resign_res = {
888                let mut mgr = state.manager.lock().await;
889                mgr.resign(conn_id)
890            };
891            match resign_res {
892                Ok(Some(outcome)) => {
893                    persist_and_broadcast_elo_outcome(outcome, state).await;
894                }
895                Ok(None) => {}
896                Err(RoomError::SpectatorCannotRelay) => {
897                    send_response(tx, &RelayResponse::error_reason("spectator_cannot_relay"));
898                }
899                Err(_) => {
900                    send_response(tx, &RelayResponse::error_message("client is not in a room"));
901                }
902            }
903        }
904        RelayRequest::ReportResult {
905            winner,
906            loser,
907            game_id,
908        } => {
909            let report_check = {
910                let mut mgr = state.manager.lock().await;
911                mgr.check_manual_report_result(conn_id)
912            };
913            match report_check {
914                ReportResultCheck::MatchNotFinished => {
915                    send_response(tx, &RelayResponse::error_reason("match_not_finished"));
916                    return;
917                }
918                ReportResultCheck::AlreadyReported => {
919                    return;
920                }
921                ReportResultCheck::Allowed => {}
922            }
923            global_metrics().inc_messages_total(Some(&game_id));
924            let mut store = state.elo_store.lock().await;
925            match store.report_result(&winner, &loser, &game_id) {
926                Ok((w_rating, l_rating)) => {
927                    global_metrics().inc_elo_updates_total();
928                    send_response(
929                        tx,
930                        &RelayResponse::ResultReported {
931                            game_id,
932                            winner,
933                            winner_elo: w_rating.elo,
934                            loser,
935                            loser_elo: l_rating.elo,
936                        },
937                    );
938                }
939                Err(err) => {
940                    send_response(
941                        tx,
942                        &RelayResponse::error_message(format!(
943                            "failed to persist ELO result: {err}"
944                        )),
945                    );
946                }
947            }
948        }
949        RelayRequest::CreateTournament {
950            game_id,
951            max_players,
952        } => {
953            let mut tm = state.tournaments.lock().await;
954            match tm.create_tournament(game_id, max_players) {
955                Ok(tournament_id) => {
956                    send_response(tx, &RelayResponse::TournamentCreated { tournament_id });
957                }
958                Err(err) => {
959                    send_response(
960                        tx,
961                        &RelayResponse::error_message(format!(
962                            "invalid tournament configuration: {err:?}"
963                        )),
964                    );
965                }
966            }
967        }
968        RelayRequest::JoinTournament {
969            tournament_id,
970            player_id,
971        } => {
972            let mut tm = state.tournaments.lock().await;
973            let pid = canonical_player_id
974                .clone()
975                .or(player_id)
976                .unwrap_or_else(|| format!("player_{conn_id}"));
977            match tm.join_tournament(&tournament_id, pid) {
978                Ok((tournament, _started)) => {
979                    send_response(tx, &RelayResponse::TournamentUpdated { tournament });
980                }
981                Err(err) => {
982                    send_response(
983                        tx,
984                        &RelayResponse::error_message(format!(
985                            "failed to join tournament: {err:?}"
986                        )),
987                    );
988                }
989            }
990        }
991        RelayRequest::ReportTournamentMatch {
992            tournament_id,
993            round,
994            match_index,
995            winner,
996        } => {
997            let mut tm = state.tournaments.lock().await;
998            match tm.report_match(&tournament_id, round, match_index, &winner) {
999                Ok(tournament) => {
1000                    send_response(tx, &RelayResponse::TournamentUpdated { tournament });
1001                }
1002                Err(err) => {
1003                    send_response(
1004                        tx,
1005                        &RelayResponse::error_message(format!(
1006                            "failed to report tournament match: {err:?}"
1007                        )),
1008                    );
1009                }
1010            }
1011        }
1012        RelayRequest::Leave => {
1013            let mut mgr = state.manager.lock().await;
1014            if let Some((room_code, _player_id)) = mgr.leave_room(conn_id) {
1015                send_response(tx, &RelayResponse::Left { room_code });
1016            } else {
1017                send_response(tx, &RelayResponse::error_message("client is not in a room"));
1018            }
1019        }
1020        RelayRequest::Ping => {
1021            send_response(tx, &RelayResponse::Pong);
1022        }
1023    }
1024}

1027    outcome: AuthoritativeMatchOutcome,
1028    state: &RelaySharedState,
1029) {
1030    global_metrics().inc_messages_total(Some(&outcome.game_id));
1031    let elo_res = {
1032        let mut store = state.elo_store.lock().await;
1033        store.report_result(&outcome.winner, &outcome.loser, &outcome.game_id)
1034    };
1035    if let Ok((w_rating, l_rating)) = elo_res {
1036        global_metrics().inc_elo_updates_total();
1037        let response = RelayResponse::EloUpdated {
1038            winner: outcome.winner,
1039            loser: outcome.loser,
1040            winner_elo: w_rating.elo,
1041            loser_elo: l_rating.elo,
1042        };
1043        if let Ok(json_str) = serde_json::to_string(&response) {
1044            let mgr = state.manager.lock().await;
1045            mgr.broadcast_to_room(&outcome.room_code, &json_str);
1046        }
1047    }
1048}
