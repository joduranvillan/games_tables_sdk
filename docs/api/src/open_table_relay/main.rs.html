1//! Microservidor standalone de salas, retransmisión WebSocket y servicio ELO para OpenTable SDK.
2//!
3//! Protocolo JSON WebSocket (puerto 8080):
4//! - `{"op": "create"}` -> `{"op": "created", "room_code": "XKCD"}`
5//! - `{"op": "join", "room_code": "XKCD"}` -> `{"op": "joined", "player_id": 2}`
6//! - `{"op": "relay", "data": "..."}` -> retransmite el payload a los demás participantes
7//! - `{"op": "report_result", "winner": "player1", "loser": "player2", "game_id": "chess"}` -> actualiza ELO (K=32)
8//! - `{"op": "leave"}` / desconexión -> notifica a los participantes restantes (`peer_left`)
9//!
10//! API HTTP (puerto 8081):
11//! - `GET /rankings/:game_id` -> JSON con top-50 jugadores por ELO
12//! - `GET /player/:player_id` -> JSON con todos los ratings del jugador por juego
13

34/// Solicitud entrante enviada por un cliente WebSocket al servidor Relay.
35#[derive(Debug, Deserialize)]
36#[serde(tag = "op", rename_all = "snake_case")]
37pub enum RelayRequest {
38    /// Crea una nueva sala y registra al cliente como anfitrión (`player_id = 1`).
39    Create {
40        /// Identificador opcional del juego.
41        #[serde(default)]
42        game_id: Option<String>,
43        /// Indica si la sala es pública (por defecto `true`).
44        #[serde(default = "default_true")]
45        is_public: bool,
46    },
47    /// Une al cliente a una sala existente mediante su código.
48    Join {
49        /// Código de la sala.
50        room_code: String,
51    },
52    /// Retransmite `data` a todos los demás participantes de la sala actual.
53    Relay {
54        /// Payload arbitrario a retransmitir.
55        data: Value,
56    },
57    /// Reporta el resultado de una partida para actualizar el ELO persistente (K=32).
58    ReportResult {
59        /// Identificador del jugador ganador.
60        winner: String,
61        /// Identificador del jugador perdedor.
62        loser: String,
63        /// Identificador del juego (ej. `"chess"`).
64        game_id: String,
65    },
66    /// Crea un nuevo torneo de eliminación simple (4, 8 o 16 jugadores).
67    CreateTournament {
68        /// Identificador del juego.
69        game_id: String,
70        /// Capacidad máxima de participantes (`4`, `8` o `16`).
71        max_players: usize,
72    },
73    /// Une a un participante a un torneo existente.
74    JoinTournament {
75        /// Identificador del torneo (ej. `"T-AB12"`).
76        tournament_id: String,
77        /// Identificador opcional del jugador.
78        #[serde(default)]
79        player_id: Option<String>,
80    },
81    /// Reporta el ganador de un enfrentamiento de torneo y avanza a la siguiente ronda.
82    ReportTournamentMatch {
83        /// Identificador del torneo.
84        tournament_id: String,
85        /// Índice de ronda.
86        round: usize,
87        /// Índice de enfrentamiento dentro de la ronda.
88        match_index: usize,
89        /// Identificador del jugador ganador.
90        winner: String,
91    },
92    /// Abandona voluntariamente la sala actual.
93    Leave,
94    /// Ping de mantenimiento de conexión.
95    Ping,
96}
97
98fn default_true() -> bool {
99    true
100}

102/// Respuesta o evento enviado por el servidor Relay a los clientes.
103#[derive(Debug, Serialize, Deserialize, PartialEq)]
104#[serde(tag = "op", rename_all = "snake_case")]
105pub enum RelayResponse {
106    /// Confirmación de creación de sala.
107    Created {
108        /// Código asignado a la sala.
109        room_code: String,
110    },
111    /// Confirmación de unión exitosa a una sala.
112    Joined {
113        /// Identificador de jugador asignado dentro de la sala (2..N).
114        player_id: u32,
115    },
116    /// Notificación a los miembros existentes cuando un nuevo jugador entra a la sala.
117    PeerJoined {
118        /// Identificador del nuevo jugador.
119        player_id: u32,
120    },
121    /// Mensaje retransmitido desde otro participante de la sala.
122    Relay {
123        /// Payload retransmitido.
124        data: Value,
125        /// Identificador del jugador que originó el mensaje.
126        #[serde(skip_serializing_if = "Option::is_none")]
127        from: Option<u32>,
128    },
129    /// Confirmación de actualización de ELO tras reportar un resultado.
130    ResultReported {
131        /// Identificador del juego.
132        game_id: String,
133        /// Identificador del ganador.
134        winner: String,
135        /// Nuevo ELO del ganador.
136        winner_elo: i32,
137        /// Identificador del perdedor.
138        loser: String,
139        /// Nuevo ELO del perdedor.
140        loser_elo: i32,
141    },
142    /// Confirmación de creación de torneo.
143    TournamentCreated {
144        /// Identificador único del torneo (ej. `"T-AB12"`).
145        tournament_id: String,
146    },
147    /// Estado actualizado del bracket del torneo.
148    TournamentUpdated {
149        /// Estado completo del torneo y sus rondas.
150        tournament: Tournament,
151    },
152    /// Confirmación al cliente que solicitó salir de la sala.
153    Left {
154        /// Código de la sala abandonada.
155        room_code: String,
156    },
157    /// Notificación a los participantes restantes cuando alguien se desconecta o sale.
158    PeerLeft {
159        /// Código de la sala.
160        room_code: String,
161        /// Identificador del jugador que salió.
162        player_id: u32,
163    },
164    /// Respuesta a `{"op": "ping"}`.
165    Pong,
166    /// Error de protocolo o de estado de sala.
167    Error {
168        /// Descripción legible del error.
169        message: String,
170    },
171}

173/// Inicia el servidor Relay sobre un `TcpListener` ya enlazado usando la base de datos por defecto.
174pub async fn run_relay_server(listener: TcpListener) -> std::io::Result<()> {
175    let db_path =
176        std::env::var("OPEN_TABLE_RATINGS_DB").unwrap_or_else(|_| "./data/ratings.db".to_string());
177    let elo_store = Arc::new(Mutex::new(EloStore::new(db_path)?));
178    run_relay_server_with_store(listener, elo_store).await
179}

181/// Inicia el servidor Relay compartiendo una instancia de `EloStore` con el servidor HTTP.
182pub async fn run_relay_server_with_store(
183    listener: TcpListener,
184    elo_store: Arc<Mutex<EloStore>>,
185) -> std::io::Result<()> {
186    let manager = Arc::new(Mutex::new(RoomManager::new()));
187    let tournaments = Arc::new(Mutex::new(TournamentManager::new()));
188    let next_conn_id = Arc::new(AtomicU64::new(1));
189
190    loop {
191        let (stream, _peer_addr) = listener.accept().await?;
192        let conn_id = next_conn_id.fetch_add(1, Ordering::Relaxed);
193        let manager_clone = Arc::clone(&manager);
194        let store_clone = Arc::clone(&elo_store);
195        let tournaments_clone = Arc::clone(&tournaments);
196
197        tokio::spawn(async move {
198            handle_connection(stream, conn_id, manager_clone, store_clone, tournaments_clone).await;
199        });
200    }
201}

203async fn handle_connection(
204    stream: TcpStream,
205    conn_id: ConnectionId,
206    manager: Arc<Mutex<RoomManager>>,
207    elo_store: Arc<Mutex<EloStore>>,
208    tournaments: Arc<Mutex<TournamentManager>>,
209) {
210    let ws_stream = match tokio_tungstenite::accept_async(stream).await {
211        Ok(ws) => ws,
212        Err(_) => return,
213    };
214
215    let (mut ws_sink, mut ws_source) = ws_stream.split();
216    let (tx, mut rx) = mpsc::unbounded_channel::<String>();
217
218    let writer_task = tokio::spawn(async move {
219        while let Some(msg) = rx.recv().await {
220            if ws_sink.send(Message::Text(msg.into())).await.is_err() {
221                break;
222            }
223        }
224    });
225
226    while let Some(Ok(msg)) = ws_source.next().await {
227        match msg {
228            Message::Text(text) => {
229                process_text_message(&text, conn_id, &tx, &manager, &elo_store, &tournaments).await;
230            }
231            Message::Ping(payload) => {
232                let _ = tx.send(
233                    serde_json::to_string(&RelayResponse::Pong)
234                        .unwrap_or_else(|_| r#"{"op":"pong"}"#.to_string()),
235                );
236                let _ = payload;
237            }
238            Message::Close(_) => {
239                break;
240            }
241            _ => {}
242        }
243    }
244
245    {
246        let mut mgr = manager.lock().await;
247        mgr.leave_room(conn_id);
248    }
249
250    writer_task.abort();
251}

253async fn process_text_message(
254    raw_text: &str,
255    conn_id: ConnectionId,
256    tx: &ClientSender,
257    manager: &Arc<Mutex<RoomManager>>,
258    elo_store: &Arc<Mutex<EloStore>>,
259    tournaments: &Arc<Mutex<TournamentManager>>,
260) {
261    let parsed: Result<RelayRequest, _> = serde_json::from_str(raw_text);
262    let request = match parsed {
263        Ok(req) => req,
264        Err(err) => {
265            send_response(
266                tx,
267                &RelayResponse::Error {
268                    message: format!("invalid json request: {err}"),
269                },
270            );
271            return;
272        }
273    };
274
275    match request {
276        RelayRequest::Create { game_id, is_public } => {
277            let mut mgr = manager.lock().await;
278            let room_code = mgr.create_room(conn_id, tx.clone(), game_id, is_public);
279            send_response(tx, &RelayResponse::Created { room_code });
280        }
281        RelayRequest::Join { room_code } => {
282            let mut mgr = manager.lock().await;
283            match mgr.join_room(conn_id, tx.clone(), &room_code) {
284                Ok((normalized_code, player_id)) => {
285                    send_response(tx, &RelayResponse::Joined { player_id });
286                    if let Ok(peer_notice) =
287                        serde_json::to_string(&RelayResponse::PeerJoined { player_id })
288                    {
289                        let _ = mgr.relay_to_peers(conn_id, &peer_notice);
290                    }
291                    let _ = normalized_code;
292                }
293                Err(RoomError::InvalidRoomCode) => {
294                    send_response(
295                        tx,
296                        &RelayResponse::Error {
297                            message: "invalid room code format".to_string(),
298                        },
299                    );
300                }
301                Err(RoomError::RoomNotFound) => {
302                    send_response(
303                        tx,
304                        &RelayResponse::Error {
305                            message: "room not found".to_string(),
306                        },
307                    );
308                }
309                Err(RoomError::NotInRoom) => {
310                    send_response(
311                        tx,
312                        &RelayResponse::Error {
313                            message: "not in room".to_string(),
314                        },
315                    );
316                }
317            }
318        }
319        RelayRequest::Relay { data } => {
320            let mgr = manager.lock().await;
321            let from_player = mgr.participant_info(conn_id).map(|(_, pid)| pid);
322            let outbound = RelayResponse::Relay {
323                data,
324                from: from_player,
325            };
326            if let Ok(serialized) = serde_json::to_string(&outbound) {
327                if mgr.relay_to_peers(conn_id, &serialized).is_err() {
328                    send_response(
329                        tx,
330                        &RelayResponse::Error {
331                            message: "client is not in a room".to_string(),
332                        },
333                    );
334                }
335            }
336        }
337        RelayRequest::ReportResult {
338            winner,
339            loser,
340            game_id,
341        } => {
342            let mut store = elo_store.lock().await;
343            match store.report_result(&winner, &loser, &game_id) {
344                Ok((w_rating, l_rating)) => {
345                    send_response(
346                        tx,
347                        &RelayResponse::ResultReported {
348                            game_id,
349                            winner,
350                            winner_elo: w_rating.elo,
351                            loser,
352                            loser_elo: l_rating.elo,
353                        },
354                    );
355                }
356                Err(err) => {
357                    send_response(
358                        tx,
359                        &RelayResponse::Error {
360                            message: format!("failed to persist ELO result: {err}"),
361                        },
362                    );
363                }
364            }
365        }
366        RelayRequest::CreateTournament {
367            game_id,
368            max_players,
369        } => {
370            let mut tm = tournaments.lock().await;
371            match tm.create_tournament(game_id, max_players) {
372                Ok(tournament_id) => {
373                    send_response(tx, &RelayResponse::TournamentCreated { tournament_id });
374                }
375                Err(err) => {
376                    send_response(
377                        tx,
378                        &RelayResponse::Error {
379                            message: format!("invalid tournament configuration: {err:?}"),
380                        },
381                    );
382                }
383            }
384        }
385        RelayRequest::JoinTournament {
386            tournament_id,
387            player_id,
388        } => {
389            let mut tm = tournaments.lock().await;
390            let pid = player_id.unwrap_or_else(|| format!("player_{conn_id}"));
391            match tm.join_tournament(&tournament_id, pid) {
392                Ok((tournament, _started)) => {
393                    send_response(tx, &RelayResponse::TournamentUpdated { tournament });
394                }
395                Err(err) => {
396                    send_response(
397                        tx,
398                        &RelayResponse::Error {
399                            message: format!("failed to join tournament: {err:?}"),
400                        },
401                    );
402                }
403            }
404        }
405        RelayRequest::ReportTournamentMatch {
406            tournament_id,
407            round,
408            match_index,
409            winner,
410        } => {
411            let mut tm = tournaments.lock().await;
412            match tm.report_match(&tournament_id, round, match_index, &winner) {
413                Ok(tournament) => {
414                    send_response(tx, &RelayResponse::TournamentUpdated { tournament });
415                }
416                Err(err) => {
417                    send_response(
418                        tx,
419                        &RelayResponse::Error {
420                            message: format!("failed to report tournament match: {err:?}"),
421                        },
422                    );
423                }
424            }
425        }
426        RelayRequest::Leave => {
427            let mut mgr = manager.lock().await;
428            if let Some((room_code, _player_id)) = mgr.leave_room(conn_id) {
429                send_response(tx, &RelayResponse::Left { room_code });
430            } else {
431                send_response(
432                    tx,
433                    &RelayResponse::Error {
434                        message: "client is not in a room".to_string(),
435                    },
436                );
437            }
438        }
439        RelayRequest::Ping => {
440            send_response(tx, &RelayResponse::Pong);
441        }
442    }
443}

445fn send_response(tx: &ClientSender, response: &RelayResponse) {
446    if let Ok(json) = serde_json::to_string(response) {
447        let _ = tx.send(json);
448    }
449}

451#[tokio::main]
452async fn main() -> std::io::Result<()> {
453    let ws_addr: SocketAddr = std::env::var("OPEN_TABLE_RELAY_ADDR")
454        .unwrap_or_else(|_| "0.0.0.0:8080".to_string())
455        .parse()
456        .expect("OPEN_TABLE_RELAY_ADDR debe ser un SocketAddr válido (ej. 0.0.0.0:8080)");
457
458    let http_addr: SocketAddr = std::env::var("OPEN_TABLE_HTTP_ADDR")
459        .unwrap_or_else(|_| "0.0.0.0:8081".to_string())
460        .parse()
461        .expect("OPEN_TABLE_HTTP_ADDR debe ser un SocketAddr válido (ej. 0.0.0.0:8081)");
462
463    let db_path =
464        std::env::var("OPEN_TABLE_RATINGS_DB").unwrap_or_else(|_| "./data/ratings.db".to_string());
465    let elo_store = Arc::new(Mutex::new(EloStore::new(Path::new(&db_path))?));
466
467    let ws_listener = TcpListener::bind(ws_addr).await?;
468    let http_listener = TcpListener::bind(http_addr).await?;
469
470    println!("open_table_relay WebSocket listening on ws://{ws_addr}");
471    println!("open_table_relay HTTP ELO API listening on http://{http_addr}");
472
473    tokio::try_join!(
474        run_relay_server_with_store(ws_listener, Arc::clone(&elo_store)),
475        run_http_api_server(http_listener, elo_store),
476    )?;
477    Ok(())
478}
