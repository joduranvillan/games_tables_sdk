1//! Microservidor standalone de salas, retransmisión WebSocket y servicio ELO para OpenTable SDK.
2//!
3//! Protocolo JSON WebSocket (puerto 8080):
4//! - `{"op": "create"}` -> `{"op": "created", "room_code": "XKCD"}`
5//! - `{"op": "join", "room_code": "XKCD"}` -> `{"op": "joined", "player_id": 2}`
6//! - `{"op": "relay", "data": "..."}` -> retransmite el payload a los demás participantes
7//! - `{"op": "report_result", "winner": "player1", "loser": "player2", "game_id": "chess"}` -> actualiza ELO (K=32)
8//! - `{"op": "leave"}` / desconexión -> notifica a los participantes restantes (`peer_left`)
9//!
10//! API HTTP (puerto 8081):
11//! - `GET /rankings/:game_id` -> JSON con top-50 jugadores por ELO
12//! - `GET /player/:player_id` -> JSON con todos los ratings del jugador por juego

44/// Estado compartido entre todas las conexiones WebSocket del Relay.
45#[derive(Clone)]
46pub struct RelaySharedState {
47    /// Gestor de salas activas.
48    pub manager: Arc<Mutex<RoomManager>>,
49    /// Almacén persistente de ELO.
50    pub elo_store: Arc<Mutex<EloStore>>,
51    /// Gestor de torneos.
52    pub tournaments: Arc<Mutex<TournamentManager>>,
53    /// Limitador de velocidad por IP y por sala.
54    pub rate_limiter: Arc<Mutex<RelayRateLimiter>>,
55    /// Configuración de autenticación JWT.
56    pub auth_config: AuthConfig,
57    /// Configuración TLS opcional cargada desde entorno.
58    pub tls_config: Option<TlsConfig>,
59}

61/// Solicitud entrante enviada por un cliente WebSocket al servidor Relay.
62#[derive(Debug, Deserialize)]
63#[serde(tag = "op", rename_all = "snake_case")]
64pub enum RelayRequest {
65    /// Autentica la conexión mediante un token JWT (`HS256`).
66    Auth {
67        /// Token JWT firmado con `JWT_SECRET`.
68        token: String,
69    },
70    /// Crea una nueva sala y registra al cliente como anfitrión (`player_id = 1`).
71    Create {
72        /// Identificador opcional del juego.
73        #[serde(default)]
74        game_id: Option<String>,
75        /// Indica si la sala es pública (por defecto `true`).
76        #[serde(default = "default_true")]
77        is_public: bool,
78    },
79    /// Une al cliente a una sala existente mediante su código.
80    Join {
81        /// Código de la sala.
82        room_code: String,
83    },
84    /// Une al cliente como espectador de solo lectura a una sala existente.
85    JoinSpectator {
86        /// Código de la sala.
87        room_code: String,
88    },
89    /// Retransmite `data` a todos los demás participantes de la sala actual.
90    Relay {
91        /// Payload arbitrario a retransmitir.
92        data: Value,
93    },
94    /// Reporta el resultado de una partida para actualizar el ELO persistente (K=32).
95    ReportResult {
96        /// Identificador del jugador ganador.
97        winner: String,
98        /// Identificador del jugador perdedor.
99        loser: String,
100        /// Identificador del juego (ej. `"chess"`).
101        game_id: String,
102    },
103    /// Crea un nuevo torneo de eliminación simple (4, 8 o 16 jugadores).
104    CreateTournament {
105        /// Identificador del juego.
106        game_id: String,
107        /// Capacidad máxima de participantes (`4`, `8` o `16`).
108        max_players: usize,
109    },
110    /// Une a un participante a un torneo existente.
111    JoinTournament {
112        /// Identificador del torneo (ej. `"T-AB12"`).
113        tournament_id: String,
114        /// Identificador opcional del jugador.
115        #[serde(default)]
116        player_id: Option<String>,
117    },
118    /// Reporta el ganador de un enfrentamiento de torneo y avanza a la siguiente ronda.
119    ReportTournamentMatch {
120        /// Identificador del torneo.
121        tournament_id: String,
122        /// Índice de ronda.
123        round: usize,
124        /// Índice de enfrentamiento dentro de la ronda.
125        match_index: usize,
126        /// Identificador del jugador ganador.
127        winner: String,
128    },
129    /// Abandona voluntariamente la sala actual.
130    Leave,
131    /// Ping de mantenimiento de conexión.
132    Ping,
133}
134
135fn default_true() -> bool {
136    true
137}

139/// Respuesta o evento enviado por el servidor Relay a los clientes.
140#[derive(Debug, Serialize, Deserialize, PartialEq)]
141#[serde(tag = "op", rename_all = "snake_case")]
142pub enum RelayResponse {
143    /// Confirmación de autenticación JWT exitosa.
144    AuthOk {
145        /// Identificador canónico (`sub`) extraído del JWT.
146        player_id: String,
147    },
148    /// Error de autenticación JWT (`expired`, `invalid_signature`, `malformed_token`).
149    AuthError {
150        /// Motivo del rechazo.
151        reason: String,
152    },
153    /// Notificación de límite de velocidad superado.
154    RateLimited {
155        /// Tiempo recomendado antes de reintentar en milisegundos (`1000`).
156        retry_after_ms: u64,
157    },
158    /// Notificación de movimiento rechazado por el validador Anti-Cheat.
159    MoveRejected {
160        /// Motivo del rechazo (`"illegal_move"`).
161        reason: String,
162    },
163    /// Confirmación de creación de sala.
164    Created {
165        /// Código asignado a la sala.
166        room_code: String,
167    },
168    /// Confirmación de unión exitosa a una sala.
169    Joined {
170        /// Identificador de jugador asignado dentro de la sala (2..N).
171        player_id: u32,
172    },
173    /// Confirmación de unión exitosa como espectador a una sala.
174    Spectating {
175        /// Cantidad de jugadores activos en la sala.
176        player_count: usize,
177    },
178    /// Notificación a los miembros existentes cuando un nuevo jugador entra a la sala.
179    PeerJoined {
180        /// Identificador del nuevo jugador.
181        player_id: u32,
182    },
183    /// Mensaje retransmitido desde otro participante de la sala.
184    Relay {
185        /// Payload retransmitido.
186        data: Value,
187        /// Identificador del jugador que originó el mensaje.
188        #[serde(skip_serializing_if = "Option::is_none")]
189        from: Option<u32>,
190    },
191    /// Confirmación de actualización de ELO tras reportar un resultado.
192    ResultReported {
193        /// Identificador del juego.
194        game_id: String,
195        /// Identificador del ganador.
196        winner: String,
197        /// Nuevo ELO del ganador.
198        winner_elo: i32,
199        /// Identificador del perdedor.
200        loser: String,
201        /// Nuevo ELO del perdedor.
202        loser_elo: i32,
203    },
204    /// Confirmación de creación de torneo.
205    TournamentCreated {
206        /// Identificador único del torneo (ej. `"T-AB12"`).
207        tournament_id: String,
208    },
209    /// Estado actualizado del bracket del torneo.
210    TournamentUpdated {
211        /// Estado completo del torneo y sus rondas.
212        tournament: Tournament,
213    },
214    /// Confirmación al cliente que solicitó salir de la sala.
215    Left {
216        /// Código de la sala abandonada.
217        room_code: String,
218    },
219    /// Notificación a los participantes restantes cuando alguien se desconecta o sale.
220    PeerLeft {
221        /// Código de la sala.
222        room_code: String,
223        /// Identificador del jugador que salió.
224        player_id: u32,
225    },
226    /// Respuesta a `{"op": "ping"}`.
227    Pong,
228    /// Error de protocolo o de estado de sala.
229    Error {
230        /// Código estructurado del motivo del error (`"spectator_cannot_relay"`, `"spectator_limit"`, etc.).
231        #[serde(skip_serializing_if = "Option::is_none")]
232        reason: Option<String>,
233        /// Descripción legible del error.
234        #[serde(skip_serializing_if = "Option::is_none")]
235        message: Option<String>,
236    },
237}

239impl RelayResponse {
240    pub fn error_reason(reason: impl Into<String>) -> Self {
241        Self::Error {
242            reason: Some(reason.into()),
243            message: None,
244        }
245    }
246
247    pub fn error_message(message: impl Into<String>) -> Self {
248        Self::Error {
249            reason: None,
250            message: Some(message.into()),
251        }
252    }

255/// Inicia el servidor Relay sobre un `TcpListener` ya enlazado usando la base de datos por defecto.
256pub async fn run_relay_server(listener: TcpListener) -> std::io::Result<()> {
257    let db_path =
258        std::env::var("OPEN_TABLE_RATINGS_DB").unwrap_or_else(|_| "./data/ratings.db".to_string());
259    let elo_store = Arc::new(Mutex::new(EloStore::new(db_path)?));
260    run_relay_server_with_store(listener, elo_store).await
261}

263/// Inicia el servidor Relay compartiendo una instancia de `EloStore` con el servidor HTTP.
264pub async fn run_relay_server_with_store(
265    listener: TcpListener,
266    elo_store: Arc<Mutex<EloStore>>,
267) -> std::io::Result<()> {

271}

273/// Inicia el servidor Relay con configuración explícita de `AuthConfig` y `TlsConfig`.
274pub async fn run_relay_server_with_config(
275    listener: TcpListener,
276    elo_store: Arc<Mutex<EloStore>>,
277    auth_config: AuthConfig,
278    tls_config: Option<TlsConfig>,
279) -> std::io::Result<()> {
280    let state = RelaySharedState {
281        manager: Arc::new(Mutex::new(RoomManager::new())),
282        elo_store,
283        tournaments: Arc::new(Mutex::new(TournamentManager::new())),
284        rate_limiter: Arc::new(Mutex::new(RelayRateLimiter::new())),
285        auth_config,
286        tls_config,
287    };
288    let next_conn_id = Arc::new(AtomicU64::new(1));
289
290    loop {
291        let (stream, peer_addr) = listener.accept().await?;
292        let conn_id = next_conn_id.fetch_add(1, Ordering::Relaxed);
293        let state_clone = state.clone();
294
295        tokio::spawn(async move {
296            handle_connection(stream, peer_addr.ip(), conn_id, state_clone).await;
297        });
298    }
299}

301async fn handle_connection(
302    stream: TcpStream,
303    peer_ip: IpAddr,
304    conn_id: ConnectionId,
305    state: RelaySharedState,
306) {
307    {
308        let mut limiter = state.rate_limiter.lock().await;
309        if limiter.is_ip_banned(peer_ip) {
310            return;
311        }
312    }
313
314    let ws_stream = match tokio_tungstenite::accept_async(stream).await {
315        Ok(ws) => ws,
316        Err(_) => return,
317    };
318
319    global_metrics().inc_connections_active();
320
321    let (mut ws_sink, mut ws_source) = ws_stream.split();
322    let (tx, mut rx) = mpsc::unbounded_channel::<String>();
323
324    let writer_task = tokio::spawn(async move {
325        while let Some(msg) = rx.recv().await {
326            if ws_sink.send(Message::Text(msg.into())).await.is_err() {
327                break;
328            }
329        }
330    });
331
332    let mut canonical_player_id: Option<String> = None;
333
334    if state.auth_config.is_enabled() {
335        let first_frame = tokio::time::timeout(
336            Duration::from_secs(AUTH_HANDSHAKE_TIMEOUT_SECS),
337            ws_source.next(),
338        )
339        .await;
340
341        let Some(Some(Ok(Message::Text(first_text)))) = first_frame.ok() else {
342            send_response(
343                &tx,
344                &RelayResponse::AuthError {
345                    reason: "auth_timeout".to_string(),
346                },
347            );
348            tokio::time::sleep(Duration::from_millis(20)).await;
349            writer_task.abort();
350            return;
351        };
352
353        let Ok(RelayRequest::Auth { token }) = serde_json::from_str::<RelayRequest>(&first_text)
354        else {
355            send_response(
356                &tx,
357                &RelayResponse::AuthError {
358                    reason: "auth_required".to_string(),
359                },
360            );
361            tokio::time::sleep(Duration::from_millis(20)).await;
362            writer_task.abort();
363            return;
364        };
365
366        match state.auth_config.verify_token(&token) {
367            Ok(claims) => {
368                canonical_player_id = Some(claims.sub.clone());
369                send_response(
370                    &tx,
371                    &RelayResponse::AuthOk {
372                        player_id: claims.sub,
373                    },
374                );
375            }
376            Err(err) => {
377                send_response(
378                    &tx,
379                    &RelayResponse::AuthError {
380                        reason: err.reason().to_string(),
381                    },
382                );
383                tokio::time::sleep(Duration::from_millis(20)).await;
384                writer_task.abort();
385                return;
386            }
387        }
388    }
389
390    while let Some(Ok(msg)) = ws_source.next().await {
391        match msg {
392            Message::Text(text) => {
393                let current_room = {
394                    let mgr = state.manager.lock().await;
395                    mgr.participant_info(conn_id).map(|(code, _)| code)
396                };
397                let decision = {
398                    let mut limiter = state.rate_limiter.lock().await;
399                    limiter.check_message(peer_ip, current_room.as_deref())
400                };
401                if let RateLimitDecision::Limited {
402                    retry_after_ms,
403                    disconnect,
404                } = decision
405                {
406                    send_response(&tx, &RelayResponse::RateLimited { retry_after_ms });
407                    if disconnect {
408                        tokio::time::sleep(Duration::from_millis(20)).await;
409                        break;
410                    }
411                    continue;
412                }
413
414                process_text_message(&text, conn_id, &mut canonical_player_id, &tx, &state).await;
415            }
416            Message::Ping(payload) => {
417                let _ = tx.send(
418                    serde_json::to_string(&RelayResponse::Pong)
419                        .unwrap_or_else(|_| r#"{"op":"pong"}"#.to_string()),
420                );
421                let _ = payload;
422            }
423            Message::Close(_) => {
424                break;
425            }
426            _ => {}
427        }
428    }
429
430    {
431        let mut mgr = state.manager.lock().await;
432        mgr.leave_room(conn_id);
433        global_metrics().set_rooms_active(mgr.active_room_count());
434    }
435
436    global_metrics().dec_connections_active();
437    writer_task.abort();
438}

440async fn process_text_message(
441    raw_text: &str,
442    conn_id: ConnectionId,
443    canonical_player_id: &mut Option<String>,
444    tx: &ClientSender,
445    state: &RelaySharedState,
446) {
447    let parsed: Result<RelayRequest, _> = serde_json::from_str(raw_text);
448    let request = match parsed {
449        Ok(req) => req,
450        Err(err) => {
451            send_response(
452                tx,
453                &RelayResponse::error_message(format!("invalid json request: {err}")),
454            );
455            return;
456        }
457    };
458
459    match request {
460        RelayRequest::Auth { token } => match state.auth_config.verify_token(&token) {
461            Ok(claims) => {
462                *canonical_player_id = Some(claims.sub.clone());
463                send_response(
464                    tx,
465                    &RelayResponse::AuthOk {
466                        player_id: claims.sub,
467                    },
468                );
469            }
470            Err(err) => {
471                send_response(
472                    tx,
473                    &RelayResponse::AuthError {
474                        reason: err.reason().to_string(),
475                    },
476                );
477            }
478        },
479        RelayRequest::Create { game_id, is_public } => {
480            global_metrics().inc_messages_total(game_id.as_deref());
481            let mut mgr = state.manager.lock().await;
482            let room_code = mgr.create_room(conn_id, tx.clone(), game_id, is_public);
483            global_metrics().set_rooms_active(mgr.active_room_count());
484            send_response(tx, &RelayResponse::Created { room_code });
485        }
486        RelayRequest::Join { room_code } => {
487            let mut mgr = state.manager.lock().await;
488            match mgr.join_room(conn_id, tx.clone(), &room_code) {
489                Ok((normalized_code, player_id)) => {
490                    send_response(tx, &RelayResponse::Joined { player_id });
491                    if let Ok(peer_notice) =
492                        serde_json::to_string(&RelayResponse::PeerJoined { player_id })
493                    {
494                        let _ = mgr.relay_to_peers(conn_id, &peer_notice);
495                    }
496                    let _ = normalized_code;
497                }
498                Err(RoomError::InvalidRoomCode) => {
499                    send_response(tx, &RelayResponse::error_message("invalid room code format"));
500                }
501                Err(RoomError::RoomNotFound) => {
502                    send_response(tx, &RelayResponse::error_message("room not found"));
503                }
504                Err(_) => {
505                    send_response(tx, &RelayResponse::error_message("not in room"));
506                }
507            }
508        }
509        RelayRequest::JoinSpectator { room_code } => {
510            let mut mgr = state.manager.lock().await;
511            match mgr.join_spectator(conn_id, tx.clone(), &room_code) {
512                Ok((_normalized_code, player_count)) => {
513                    send_response(tx, &RelayResponse::Spectating { player_count });
514                }
515                Err(RoomError::SpectatorLimit) => {
516                    send_response(tx, &RelayResponse::error_reason("spectator_limit"));
517                }
518                Err(RoomError::InvalidRoomCode) => {
519                    send_response(tx, &RelayResponse::error_message("invalid room code format"));
520                }
521                Err(RoomError::RoomNotFound) => {
522                    send_response(tx, &RelayResponse::error_message("room not found"));
523                }
524                Err(_) => {
525                    send_response(tx, &RelayResponse::error_message("not in room"));
526                }
527            }
528        }
529        RelayRequest::Relay { data } => {
530            let mut mgr = state.manager.lock().await;
531            let from_player = mgr.participant_info(conn_id).map(|(_, pid)| pid);
532            let outbound = RelayResponse::Relay {
533                data: data.clone(),
534                from: from_player,
535            };
536            if let Ok(serialized) = serde_json::to_string(&outbound) {
537                match mgr.validate_and_relay(conn_id, &data, &serialized) {
538                    Ok(_) => {}
539                    Err(RoomError::SpectatorCannotRelay) => {
540                        send_response(tx, &RelayResponse::error_reason("spectator_cannot_relay"));
541                    }
542                    Err(RoomError::IllegalMove) => {
543                        send_response(
544                            tx,
545                            &RelayResponse::MoveRejected {
546                                reason: "illegal_move".to_string(),
547                            },
548                        );
549                    }
550                    Err(_) => {
551                        send_response(tx, &RelayResponse::error_message("client is not in a room"));
552                    }
553                }
554            }
555        }
556        RelayRequest::ReportResult {
557            winner,
558            loser,
559            game_id,
560        } => {
561            global_metrics().inc_messages_total(Some(&game_id));
562            let mut store = state.elo_store.lock().await;
563            match store.report_result(&winner, &loser, &game_id) {
564                Ok((w_rating, l_rating)) => {
565                    global_metrics().inc_elo_updates_total();
566                    send_response(
567                        tx,
568                        &RelayResponse::ResultReported {
569                            game_id,
570                            winner,
571                            winner_elo: w_rating.elo,
572                            loser,
573                            loser_elo: l_rating.elo,
574                        },
575                    );
576                }
577                Err(err) => {
578                    send_response(
579                        tx,
580                        &RelayResponse::error_message(format!(
581                            "failed to persist ELO result: {err}"
582                        )),
583                    );
584                }
585            }
586        }
587        RelayRequest::CreateTournament {
588            game_id,
589            max_players,
590        } => {
591            let mut tm = state.tournaments.lock().await;
592            match tm.create_tournament(game_id, max_players) {
593                Ok(tournament_id) => {
594                    send_response(tx, &RelayResponse::TournamentCreated { tournament_id });
595                }
596                Err(err) => {
597                    send_response(
598                        tx,
599                        &RelayResponse::error_message(format!(
600                            "invalid tournament configuration: {err:?}"
601                        )),
602                    );
603                }
604            }
605        }
606        RelayRequest::JoinTournament {
607            tournament_id,
608            player_id,
609        } => {
610            let mut tm = state.tournaments.lock().await;
611            let pid = canonical_player_id
612                .clone()
613                .or(player_id)
614                .unwrap_or_else(|| format!("player_{conn_id}"));
615            match tm.join_tournament(&tournament_id, pid) {
616                Ok((tournament, _started)) => {
617                    send_response(tx, &RelayResponse::TournamentUpdated { tournament });
618                }
619                Err(err) => {
620                    send_response(
621                        tx,
622                        &RelayResponse::error_message(format!(
623                            "failed to join tournament: {err:?}"
624                        )),
625                    );
626                }
627            }
628        }
629        RelayRequest::ReportTournamentMatch {
630            tournament_id,
631            round,
632            match_index,
633            winner,
634        } => {
635            let mut tm = state.tournaments.lock().await;
636            match tm.report_match(&tournament_id, round, match_index, &winner) {
637                Ok(tournament) => {
638                    send_response(tx, &RelayResponse::TournamentUpdated { tournament });
639                }
640                Err(err) => {
641                    send_response(
642                        tx,
643                        &RelayResponse::error_message(format!(
644                            "failed to report tournament match: {err:?}"
645                        )),
646                    );
647                }
648            }
649        }
650        RelayRequest::Leave => {
651            let mut mgr = state.manager.lock().await;
652            if let Some((room_code, _player_id)) = mgr.leave_room(conn_id) {
653                send_response(tx, &RelayResponse::Left { room_code });
654            } else {
655                send_response(tx, &RelayResponse::error_message("client is not in a room"));
656            }
657        }
658        RelayRequest::Ping => {
659            send_response(tx, &RelayResponse::Pong);
660        }
661    }
662}

664fn send_response(tx: &ClientSender, response: &RelayResponse) {
665    if let Ok(json) = serde_json::to_string(response) {
666        let _ = tx.send(json);
667    }

670#[tokio::main]
671async fn main() -> std::io::Result<()> {
672    let _ = IpAddr::V4(Ipv4Addr::LOCALHOST);
673    let ws_addr: SocketAddr = std::env::var("OPEN_TABLE_RELAY_ADDR")
674        .unwrap_or_else(|_| "0.0.0.0:8080".to_string())
675        .parse()
676        .expect("OPEN_TABLE_RELAY_ADDR debe ser un SocketAddr válido (ej. 0.0.0.0:8080)");
677
678    let http_addr: SocketAddr = std::env::var("OPEN_TABLE_HTTP_ADDR")
679        .unwrap_or_else(|_| "0.0.0.0:8081".to_string())
680        .parse()
681        .expect("OPEN_TABLE_HTTP_ADDR debe ser un SocketAddr válido (ej. 0.0.0.0:8081)");
682
683    let db_path =
684        std::env::var("OPEN_TABLE_RATINGS_DB").unwrap_or_else(|_| "./data/ratings.db".to_string());
685    let elo_store = Arc::new(Mutex::new(EloStore::new(Path::new(&db_path))?));
686
687    let ws_listener = TcpListener::bind(ws_addr).await?;
688    let http_listener = TcpListener::bind(http_addr).await?;
689
690    println!("open_table_relay WebSocket listening on ws://{ws_addr}");
691    println!("open_table_relay HTTP ELO API listening on http://{http_addr}");
692
693    tokio::try_join!(
694        run_relay_server_with_store(ws_listener, Arc::clone(&elo_store)),
695        run_http_api_server(http_listener, elo_store),
696    )?;
697    Ok(())
698}
