8/// Selecciona el mejor movimiento para `player` según las heurísticas de Ludo.
9pub fn ludo_ai_pick_move(session: &Session, player: PlayerId) -> Option<MoveTokenCommand> {
10    if session.current_player() != player {
11        return None;
12    }
13
14    let pending_roll = session
15        .world()
16        .get_resource::<PendingRollResource>()
17        .and_then(|p| p.pending.get(&player).copied())?;
18
19    let tokens_res = session.world().get_resource::<TokensResource>()?;
20    let player_tokens = tokens_res.tokens.get(&player)?;
21
22    let min_unfinished_steps = player_tokens
23        .iter()
24        .copied()
25        .filter(|&s| s < MAX_STEPS)
26        .min()
27        .unwrap_or(0);
28
29    let mut best_token: Option<usize> = None;
30    let mut best_score: i32 = i32::MIN;
31
32    for (token_index, &current_steps) in player_tokens.iter().enumerate() {
33        if !tokens_res.is_valid_move_for_token(player, token_index, pending_roll) {
34            continue;
35        }
36
37        let target_steps = if current_steps == 0 {
38            1
39        } else {
40            current_steps + pending_roll
41        };
42
43        let pasos_avanzados = if current_steps == 0 { 1 } else { pending_roll };
44        let mut score: i32 = 0;
45
46        // +20 si saca una ficha de la base (steps 0 -> 1)
47        if current_steps == 0 && target_steps == 1 {
48            score += 20;
49        }
50
51        // +15 si llega exactamente a la meta (steps == WIN_STEPS)
52        if target_steps == WIN_STEPS {
53            score += 15;
54        }
55
56        // +10 si captura una ficha rival (en casilla compartida que no sea segura)
57        if let Some(landing_pos) = tokens_res.board_position(player, target_steps) {
58            if !tokens_res.is_safe_square(landing_pos) {
59                let captures_rival = tokens_res.tokens.iter().any(|(&other, other_tokens)| {
60                    other != player
61                        && other_tokens
62                            .iter()
63                            .any(|&s| tokens_res.board_position(other, s) == Some(landing_pos))
64                });
65                if captures_rival {
66                    score += 10;
67                }
68            }
69        }
70
71        // -10 si se mueve a una casilla ocupada por 2 fichas rivales (bloqueo rival)
72        if let Some(landing_pos) = tokens_res.board_position(player, target_steps) {
73            if tokens_res.has_rival_blockade_at(landing_pos, player) {
74                score -= 10;
75            }
76        }
77
78        // +5 si avanza la ficha más atrasada
79        if current_steps == min_unfinished_steps {
80            score += 5;
81        }
82
83        // +3 * pasos_avanzados
84        score += 3 * (pasos_avanzados as i32);
85
86        // Si hay dado con valor 6, priorizar sacar fichas de base
87        if pending_roll == 6 && current_steps == 0 && target_steps == 1 {
88            score += 10;
89        }
90
91        // En caso de empate, se queda con el de menor token_index (estricto `>`)
92        if score > best_score {
93            best_score = score;
94            best_token = Some(token_index);
95        }
96    }
97
98    best_token.map(|token_index| MoveTokenCommand {
99        player,
100        token_index,
101    })
102}

104/// Selecciona el mejor movimiento para `player` en Ludo parametrizado por `AiDifficulty`.
105pub fn best_move(
106    session: &Session,
107    player: PlayerId,
108    difficulty: AiDifficulty,
109) -> Option<MoveTokenCommand> {
110    let _depth = difficulty.max_depth();
111    ludo_ai_pick_move(session, player)
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117    use crate::{new_ludo_session, PendingRollResource, TokensResource};
118
119    const P1: PlayerId = PlayerId(1);
120    const P2: PlayerId = PlayerId(2);
121
122    #[test]
123    fn test_ludo_ai_saca_ficha_con_seis() {
124        let mut session = new_ludo_session(SessionId(1), 1, vec![P1, P2]);
125        {
126            let mut pending = session.world_mut().resource_mut::<PendingRollResource>();
127            pending.pending.insert(P1, 6);
128        }
129
130        let mv = ludo_ai_pick_move(&session, P1);
131        assert!(mv.is_some(), "La IA debe poder mover con un 6");
132        let cmd = mv.unwrap();
133        assert_eq!(cmd.player, P1);
134        assert_eq!(
135            cmd.token_index, 0,
136            "La IA debe elegir sacar la primera ficha de la base (token 0)"
137        );
138    }
139
140    #[test]
141    fn test_ludo_ai_captura_cuando_puede() {
142        let mut session = new_ludo_session(SessionId(2), 1, vec![P1, P2]);
143        {
144            let mut tokens = session.world_mut().resource_mut::<TokensResource>();
145            // P1: token 0 en steps=10, token 1 en steps=5
146            tokens.tokens.get_mut(&P1).unwrap()[0] = 10;
147            tokens.tokens.get_mut(&P1).unwrap()[1] = 5;
148
149            // P2: token 0 en steps=40 tal que posición absoluta coincide con P1 en steps=14
150            // P1 start_offset = 0 => (0 + 14 - 1)%52 = 13
151            // P2 start_offset = 26 => (26 + steps - 1)%52 = 13 => 25 + steps = 65 => steps = 40
152            tokens.tokens.get_mut(&P2).unwrap()[0] = 40;
153        }
154        {
155            let mut pending = session.world_mut().resource_mut::<PendingRollResource>();
156            pending.pending.insert(P1, 4);
157        }
158
159        let mv = ludo_ai_pick_move(&session, P1);
160        assert!(mv.is_some(), "La IA debe seleccionar un movimiento");
161        let cmd = mv.unwrap();
162        assert_eq!(cmd.player, P1);
163        assert_eq!(
164            cmd.token_index, 0,
165            "La IA debe priorizar la ficha que realiza una captura sobre la más atrasada"
166        );
167    }
168}
