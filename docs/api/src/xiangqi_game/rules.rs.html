8pub fn all_legal_moves_for(world: &World, player: PlayerId) -> Vec<MovePieceCommand> {
9    let board = world.resource::<BoardResource>();
10    let mut legal_moves = Vec::new();
11
12    for (&from, piece) in &board.squares {
13        if piece.owner == player {
14            for tx in 0..9 {
15                for ty in 0..10 {
16                    let to = (tx, ty);
17                    if is_legal_move(board, from, to, player) {
18                        let mut test_board = board.clone();
19                        let p = test_board.squares.remove(&from).unwrap();
20                        test_board.squares.insert(to, p);
21                        if !is_in_check(&test_board, player) && !check_flying_general(&test_board) {
22                            legal_moves.push(MovePieceCommand { player, from, to });
23                        }
24                    }
25                }
26            }
27        }
28    }
29
30    legal_moves
31}

33pub fn build_xiangqi_rules() -> RuleEngine {
34    let mut rules = RuleEngine::new();
35
36    rules.register::<MovePieceCommand, _>(|cmd, world| {
37        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
38            return Err(EngineError::InvalidCommand {
39                reason: "Not your turn".into(),
40            });
41        }
42
43        let (captured_kind, invalid_move) = {
44            let mut board = world.resource_mut::<BoardResource>();
45
46            let piece_for_check = board.squares.get(&cmd.from).copied();
47            if let Some(piece) = piece_for_check {
48                if piece.owner == cmd.player && piece.kind == crate::board::PieceKind::Elephant {
49                    let is_red = piece.owner == board.red;
50                    if (is_red && cmd.to.1 < 5) || (!is_red && cmd.to.1 > 4) {
51                        return Err(EngineError::InvalidCommand {
52                            reason: "Elephant cannot cross the river".into(),
53                        });
54                    }
55
56                    let eye_x = (cmd.from.0 + cmd.to.0) / 2;
57                    let eye_y = (cmd.from.1 + cmd.to.1) / 2;
58                    if board.squares.contains_key(&(eye_x, eye_y)) {
59                        return Err(EngineError::InvalidCommand {
60                            reason: "Elephant's eye is blocked".into(),
61                        });
62                    }
63                } else if piece.owner == cmd.player && piece.kind == crate::board::PieceKind::Horse
64                {
65                    let dx = (cmd.to.0 as isize - cmd.from.0 as isize).abs();
66                    let dy = (cmd.to.1 as isize - cmd.from.1 as isize).abs();
67                    if (dx == 1 && dy == 2) || (dx == 2 && dy == 1) {
68                        let block_x = if dx == 2 {
69                            (cmd.from.0 + cmd.to.0) / 2
70                        } else {
71                            cmd.from.0
72                        };
73                        let block_y = if dy == 2 {
74                            (cmd.from.1 + cmd.to.1) / 2
75                        } else {
76                            cmd.from.1
77                        };
78                        if board.squares.contains_key(&(block_x, block_y)) {
79                            return Err(EngineError::InvalidCommand {
80                                reason: "Horse's leg is blocked".into(),
81                            });
82                        }
83                    }
84                } else if piece.owner == cmd.player && piece.kind == crate::board::PieceKind::Cannon
85                {
86                    let target_has_piece = board.squares.contains_key(&cmd.to);
87                    let dx = (cmd.to.0 as isize - cmd.from.0 as isize).abs();
88                    let dy = (cmd.to.1 as isize - cmd.from.1 as isize).abs();
89                    let mut count = 0;
90                    if dx != 0 && dy == 0 {
91                        let min = cmd.from.0.min(cmd.to.0);
92                        let max = cmd.from.0.max(cmd.to.0);
93                        for x in min + 1..max {
94                            if board.squares.contains_key(&(x, cmd.from.1)) {
95                                count += 1;
96                            }
97                        }
98                    } else if dy != 0 && dx == 0 {
99                        let min = cmd.from.1.min(cmd.to.1);
100                        let max = cmd.from.1.max(cmd.to.1);
101                        for y in min + 1..max {
102                            if board.squares.contains_key(&(cmd.from.0, y)) {
103                                count += 1;
104                            }
105                        }
106                    }
107                    if !target_has_piece && count > 0 {
108                        return Err(EngineError::InvalidCommand {
109                            reason: "Cannon cannot jump when not capturing".into(),
110                        });
111                    } else if target_has_piece && count != 1 {
112                        return Err(EngineError::InvalidCommand {
113                            reason: "Cannon capture requires exactly one platform piece".into(),
114                        });
115                    }
116                } else if piece.owner == cmd.player
117                    && piece.kind == crate::board::PieceKind::Advisor
118                {
119                    let is_red = piece.owner == board.red;
120                    let dx = (cmd.to.0 as isize - cmd.from.0 as isize).abs();
121                    let dy = (cmd.to.1 as isize - cmd.from.1 as isize).abs();
122                    if dx != 1 || dy != 1 {
123                        return Err(EngineError::InvalidCommand {
124                            reason: "Advisor must move diagonally one step".into(),
125                        });
126                    }
127                    if !(3..=5).contains(&cmd.to.0)
128                        || (is_red && !(7..=9).contains(&cmd.to.1))
129                        || (!is_red && cmd.to.1 > 2)
130                    {
131                        return Err(EngineError::InvalidCommand {
132                            reason: "Advisor must stay within the palace".into(),
133                        });
134                    }
135                }
136            }
137
138            if !is_legal_move(&board, cmd.from, cmd.to, cmd.player) {
139                return Err(EngineError::InvalidCommand {
140                    reason: "Illegal move for this piece".into(),
141                });
142            }
143
144            let target = board.squares.get(&cmd.to).copied();
145            let captured_kind = target.map(|t| t.kind);
146
147            let piece = board.squares.remove(&cmd.from).unwrap();
148            board.squares.insert(cmd.to, piece);
149
150            if check_flying_general(&board) || is_in_check(&board, cmd.player) {
151                board.squares.remove(&cmd.to);
152                board.squares.insert(cmd.from, piece);
153                if let Some(t) = target {
154                    board.squares.insert(cmd.to, t);
155                }
156                (captured_kind, true)
157            } else {
158                (captured_kind, false)
159            }
160        }; // board dropped here
161
162        if invalid_move {
163            return Err(EngineError::InvalidCommand {
164                reason: "Move leaves General in check or facing enemy General".into(),
165            });
166        }
167
168        world.resource_mut::<TurnResource>().0.advance();
169
170        let opponent = {
171            let board = world.resource::<BoardResource>();
172            if cmd.player == board.red {
173                board.black
174            } else {
175                board.red
176            }
177        };
178
179        let is_mate = {
180            let board = world.resource::<BoardResource>();
181            is_in_check(board, opponent) && all_legal_moves_for(world, opponent).is_empty()
182        };
183
184        let mut events: Vec<Box<dyn Event>> = vec![Box::new(PieceMovedEvent {
185            player: cmd.player,
186            from: cmd.from,
187            to: cmd.to,
188            captured: captured_kind,
189        })];
190
191        if is_mate {
192            events.push(Box::new(GameOverEvent {
193                winner: Some(cmd.player),
194            }));
195        }
196
197        Ok(events)
198    });
199
200    rules.register::<ResignCommand, _>(|cmd, world| {
201        let players = world.resource::<TurnResource>().0.players().to_vec();
202        if !players.contains(&cmd.player) {
203            return Err(EngineError::InvalidCommand {
204                reason: "Not a player in this session".into(),
205            });
206        }
207        let opponent = players
208            .into_iter()
209            .find(|&p| p != cmd.player)
210            .unwrap_or(cmd.player);
211        world.insert_resource(GameOverResource(true));
212        world.insert_resource(WinnerResource(Some(opponent)));
213        Ok(vec![Box::new(GameOverEvent {
214            winner: Some(opponent),
215        })])
216    });
217
218    rules
219}
