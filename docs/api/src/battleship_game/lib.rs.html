34pub fn new_battleship_session(
35    session_id: SessionId,
36    seed: u64,
37    player_a: PlayerId,
38    player_b: PlayerId,
39) -> Session {
40    let mut session = Session::new_with_metadata(
41        session_id,
42        "Battleship",
43        seed,
44        vec![player_a, player_b],
45        build_battleship_rules(),
46    );
47    session.world_mut().insert_resource(FleetResource::new());
48    session.world_mut().insert_resource(ShotsResource::new());
49    session.world_mut().insert_resource(RematchResource::new());
50    session
51        .world_mut()
52        .insert_resource(BattleshipConfigResource::default());
53    session
54}
