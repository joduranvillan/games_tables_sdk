4//! IDs fundamentales. Todos son newtypes sobre enteros: sin punteros, sin
5//! referencias, para que cualquier `Command`/`Event` que los contenga sea
6//! trivialmente serializable y determinista entre plataformas.
7
8use serde::{Deserialize, Serialize};
9
10/// Identifica a un jugador dentro de una `Session`. Asignado por el host al
11/// aceptar la conexión, nunca por el propio cliente.
12#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord, Serialize, Deserialize)]
13pub struct PlayerId(pub u32);
14
15/// Identifica a un espectador dentro de una `Session`.
16#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord, Serialize, Deserialize)]
17pub struct SpectatorId(pub u32);
18
19/// Identifica una partida/sala completa.
20#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
21pub struct SessionId(pub u64);
22
23/// Identifica una entidad dentro del ECS (jugador, propiedad, carta, ficha...).
24/// Definido acá y no en el motor para que los `Command`/`Event` puedan
25/// referenciar entidades sin depender del crate de ECS.
26#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
27pub struct EntityId(pub u64);
28
29/// Tick lógico de simulación. Nunca tiempo de reloj real (ver `time.rs`).
30#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize)]
31pub struct Tick(pub u64);
32
33impl Tick {
34    pub const ZERO: Tick = Tick(0);
35
36    pub fn next(self) -> Tick {
37        Tick(self.0 + 1)
38    }
39}
40
41impl From<u64> for Tick {
42    fn from(val: u64) -> Self {
43        Tick(val)
44    }
45}
