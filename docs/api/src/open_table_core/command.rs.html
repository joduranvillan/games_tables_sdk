9/// Un `Command` es la intención de un jugador ("quiero tirar el dado",
10/// "quiero jugar esta carta"). Es solo datos: nunca contiene lógica.
11///
12/// Se define como trait object serializable (`#[typetag::serde]`) en lugar
13/// de una enum cerrada a propósito: así, el crate de un juego nuevo (ej.
14/// `chess_plugin`) puede declarar sus propios comandos concretos sin que
15/// `open_table_core` necesite conocerlos ni recompilarse.
16#[typetag::serde(tag = "command_type")]
17pub trait Command: std::fmt::Debug + Send + Sync {
18    /// Clona el comando envuelto en un Box. Necesario para clonar `Box<dyn Command>`.
19    fn clone_box(&self) -> Box<dyn Command>;
20
21    /// Jugador que el propio comando DICE que lo emite. Esto es una
22    /// afirmación del payload, no una verificación — por eso `Session`
23    /// (en `open_table_engine`) exige que coincida con el `sender` real
24    /// que vino de la conexión de red antes de ejecutar nada. Ver
25    /// `Session::process_command`.
26    fn issuer(&self) -> Option<PlayerId> {
27        None
28    }
29
30    /// Necesario para que el rule engine (en `open_table_engine`) pueda
31    /// hacer downcast a la struct concreta sin que `open_table_core` la
32    /// conozca. `#[derive(GameCommand)]` de `open_table_macros` lo genera
33    /// solo.
34    fn as_any(&self) -> &dyn std::any::Any;
35}
36
37impl Clone for Box<dyn Command> {
38    fn clone(&self) -> Self {
39        self.as_ref().clone_box()
40    }
41}
42
43/// Sobre de red para un comando. A propósito NO tiene un campo `sender`:
44/// quién mandó el comando lo determina la capa de `Transport` (qué
45/// conexión trajo estos bytes), nunca algo que viaja dentro del propio
46/// mensaje — un campo `sender` acá sería trivial de falsificar para
47/// cualquier cliente. Ver `Session::process_command` para la validación
48/// real (contrasta `Command::issuer()` contra el sender verificado).
49#[derive(Debug, Serialize, Deserialize)]
50pub struct CommandEnvelope {
51    pub tick: Tick,
52    pub payload: Box<dyn Command>,
53}
