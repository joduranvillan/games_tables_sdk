4use rand_chacha::ChaCha8Rng;
5use rand_core::{RngCore, SeedableRng};
6use serde::{Deserialize, Serialize};
7
8/// Única fuente de aleatoriedad permitida en todo el engine. Nunca
9/// `rand::thread_rng()`: con la misma seed, en cualquier plataforma,
10/// produce exactamente la misma secuencia (requisito para replay y para
11/// que el host pueda auditar/reproducir una partida).
12#[derive(Debug, Clone, Serialize, Deserialize)]
13pub struct RandomService {
14    rng: ChaCha8Rng,
15    seed: u64,
16}
17
18impl RandomService {
19    pub fn from_seed(seed: u64) -> Self {
20        Self {
21            rng: ChaCha8Rng::seed_from_u64(seed),
22            seed,
23        }
24    }
25
26    pub fn seed(&self) -> u64 {
27        self.seed
28    }
29
30    pub fn next_u32(&mut self) -> u32 {
31        self.rng.next_u32()
32    }
33
34    /// Entero uniforme en `[min, max]` inclusive.
35    pub fn range(&mut self, min: u32, max: u32) -> u32 {
36        debug_assert!(max >= min);
37        if max <= min {
38            return min;
39        }
40        let span = (u64::from(max) - u64::from(min)) + 1;
41        min + ((u64::from(self.next_u32()) % span) as u32)
42    }
43
44    /// Tirada de dado de `sides` caras, resultado en `[1, sides]`.
45    pub fn roll_die(&mut self, sides: u32) -> u32 {
46        self.range(1, sides.max(1))
47    }
48
49    /// Baraja un slice in-place con Fisher-Yates determinista.
50    pub fn shuffle<T>(&mut self, items: &mut [T]) {
51        for i in (1..items.len()).rev() {
52            let j = self.range(0, i as u32) as usize;
53            items.swap(i, j);
54        }
55    }
56}
