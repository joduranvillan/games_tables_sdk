33fn ensure_resources(world: &mut BevyWorld) {
34    if !world.contains_resource::<SpeedLimitResource>() {
35        world.insert_resource(SpeedLimitResource::default());
36    }
37    if !world.contains_resource::<RightOfWayResource>() {
38        world.insert_resource(RightOfWayResource::default());
39    }
40}

42fn validate_and_play_distance(
43    player: PlayerId,
44    value: u32,
45    world: &mut BevyWorld,
46) -> Result<Vec<Box<dyn Event>>, EngineError> {
47    ensure_resources(world);
48
49    let players = world.resource::<TurnResource>().0.players().to_vec();
50    let has_right_of_way = world
51        .resource::<RightOfWayResource>()
52        .active
53        .contains(&player);
54    let speed_limited = world
55        .resource::<SpeedLimitResource>()
56        .limited
57        .contains(&player);
58
59    let mut deck = world.resource_mut::<DeckResource>();
60    let is_speed_limited =
61        speed_limited || deck.hazard.get(&player).copied().flatten() == Some(Hazard::SpeedLimit);
62
63    if is_speed_limited && !has_right_of_way && value > 50 {
64        return Err(invalid(
65            "límite de velocidad: solo podés jugar cartas de 25 o 50 km",
66        ));
67    }
68
69    if let Some(h) = deck.hazard.get(&player).copied().flatten() {
70        if h != Hazard::SpeedLimit && (!has_right_of_way || h == Hazard::OutOfGas) {
71            return Err(invalid("estás detenido, jugá un remedio antes de avanzar"));
72        }
73    }
74
75    if value == 200 {
76        let count = deck.cards_200_played.get(&player).copied().unwrap_or(0);
77        if count >= 2 {
78            return Err(invalid("ya jugaste el máximo de 2 cartas de 200 km"));
79        }
80    }
81
82    let current = *deck.distance.get(&player).unwrap_or(&0);
83    if current + value > deck.target_distance {
84        return Err(invalid("esa jugada te pasaría de la meta"));
85    }
86
87    let new_distance = current + value;
88    deck.distance.insert(player, new_distance);
89
90    if value == 200 {
91        *deck.cards_200_played.entry(player).or_default() += 1;
92    }
93
94    let mut events: Vec<Box<dyn Event>> = vec![Box::new(DistancePlayedEvent {
95        player,
96        value,
97        new_distance,
98    })];
99
100    if value == 200 {
101        let all_played_2 = players
102            .iter()
103            .all(|p| deck.cards_200_played.get(p).copied().unwrap_or(0) >= 2);
104        if all_played_2 && deck.target_distance < 1000 {
105            deck.target_distance = 1000;
106            events.push(Box::new(TripExtendedEvent { new_target: 1000 }));
107        }
108    }
109
110    if new_distance == deck.target_distance {
111        events.push(Box::new(GameOverEvent { winner: player }));
112    }
113
114    Ok(events)
115}

117fn validate_and_play_hazard(
118    player: PlayerId,
119    target: PlayerId,
120    hazard: Hazard,
121    world: &mut BevyWorld,
122) -> Result<Vec<Box<dyn Event>>, EngineError> {
123    ensure_resources(world);
124
125    let target_row = world
126        .resource::<RightOfWayResource>()
127        .active
128        .contains(&target);
129
130    // Compatibilidad: RightOfWayResource bloquea todos los hazards excepto OutOfGas
131    if target_row && hazard != Hazard::OutOfGas {
132        return Err(invalid(
133            "el objetivo tiene derecho de vía activo y es inmune a este peligro",
134        ));
135    }
136
137    // Inmunidad específica por cartas de seguridad jugadas
138    let has_specific_safety = {
139        let deck = world.resource::<DeckResource>();
140        let safeties = deck.safeties_played.get(&target);
141        match hazard {
142            Hazard::SpeedLimit => safeties
143                .map(|s| s.contains(&Safety::RightOfWay))
144                .unwrap_or(false),
145            Hazard::OutOfGas => safeties
146                .map(|s| s.contains(&Safety::ExtraTank))
147                .unwrap_or(false),
148            Hazard::FlatTire => safeties
149                .map(|s| s.contains(&Safety::PunctureProof))
150                .unwrap_or(false),
151            Hazard::Accident => safeties
152                .map(|s| s.contains(&Safety::DrivingAce))
153                .unwrap_or(false),
154        }
155    };
156
157    if has_specific_safety {
158        return Err(invalid(
159            "el objetivo tiene una carta de seguridad que lo protege de este peligro",
160        ));
161    }
162
163    if hazard == Hazard::SpeedLimit {
164        let limited = world
165            .resource::<SpeedLimitResource>()
166            .limited
167            .contains(&target);
168        if limited {
169            return Err(invalid("el rival ya tiene un límite de velocidad activo"));
170        }
171        world
172            .resource_mut::<SpeedLimitResource>()
173            .limited
174            .insert(target);
175        let mut deck = world.resource_mut::<DeckResource>();
176        deck.hazard.insert(target, Some(Hazard::SpeedLimit));
177    } else {
178        let mut deck = world.resource_mut::<DeckResource>();
179        if deck.hazard.get(&target).copied().flatten().is_some() {
180            return Err(invalid("el rival ya está detenido por otro peligro"));
181        }
182        deck.hazard.insert(target, Some(hazard));
183    }
184
185    Ok(vec![Box::new(HazardPlayedEvent {
186        player,
187        target,
188        hazard,
189    })])
190}

192fn validate_and_play_remedy(
193    player: PlayerId,
194    hazard: Hazard,
195    world: &mut BevyWorld,
196) -> Result<Vec<Box<dyn Event>>, EngineError> {
197    ensure_resources(world);
198
199    if hazard == Hazard::SpeedLimit {
200        let limited = world
201            .resource::<SpeedLimitResource>()
202            .limited
203            .contains(&player);
204        let in_hazard = {
205            let deck = world.resource::<DeckResource>();
206            deck.hazard.get(&player).copied().flatten() == Some(Hazard::SpeedLimit)
207        };
208        if !limited && !in_hazard {
209            return Err(invalid("no tenés límite de velocidad que curar"));
210        }
211        world
212            .resource_mut::<SpeedLimitResource>()
213            .limited
214            .remove(&player);
215        if in_hazard {
216            let mut deck = world.resource_mut::<DeckResource>();
217            deck.hazard.insert(player, None);
218        }
219    } else {
220        let mut deck = world.resource_mut::<DeckResource>();
221        let current_hazard = deck.hazard.get(&player).copied().flatten();
222        if current_hazard != Some(hazard) {
223            return Err(invalid("ese remedio no cura tu situación actual"));
224        }
225        deck.hazard.insert(player, None);
226    }
227
228    Ok(vec![Box::new(RemedyPlayedEvent { player, hazard })])
229}

231fn validate_and_play_safety(
232    player: PlayerId,
233    safety: Safety,
234    world: &mut BevyWorld,
235) -> Result<Vec<Box<dyn Event>>, EngineError> {
236    ensure_resources(world);
237
238    match safety {
239        Safety::RightOfWay => {
240            world
241                .resource_mut::<RightOfWayResource>()
242                .active
243                .insert(player);
244            world
245                .resource_mut::<SpeedLimitResource>()
246                .limited
247                .remove(&player);
248
249            let mut deck = world.resource_mut::<DeckResource>();
250            deck.safeties_played
251                .entry(player)
252                .or_default()
253                .insert(Safety::RightOfWay);
254            if deck.hazard.get(&player).copied().flatten() != Some(Hazard::OutOfGas) {
255                deck.hazard.insert(player, None);
256            }
257        }
258        Safety::ExtraTank => {
259            let mut deck = world.resource_mut::<DeckResource>();
260            deck.safeties_played
261                .entry(player)
262                .or_default()
263                .insert(Safety::ExtraTank);
264            if deck.hazard.get(&player).copied().flatten() == Some(Hazard::OutOfGas) {
265                deck.hazard.insert(player, None);
266            }
267        }
268        Safety::PunctureProof => {
269            let mut deck = world.resource_mut::<DeckResource>();
270            deck.safeties_played
271                .entry(player)
272                .or_default()
273                .insert(Safety::PunctureProof);
274            if deck.hazard.get(&player).copied().flatten() == Some(Hazard::FlatTire) {
275                deck.hazard.insert(player, None);
276            }
277        }
278        Safety::DrivingAce => {
279            let mut deck = world.resource_mut::<DeckResource>();
280            deck.safeties_played
281                .entry(player)
282                .or_default()
283                .insert(Safety::DrivingAce);
284            if deck.hazard.get(&player).copied().flatten() == Some(Hazard::Accident) {
285                deck.hazard.insert(player, None);
286            }
287        }
288    }
289
290    Ok(vec![Box::new(SafetyPlayedEvent { player, safety })])
291}

293pub fn build_mille_bornes_rules() -> RuleEngine {
294    let mut rules = RuleEngine::new();
295
296    rules.register::<SetupDeckCommand, _>(|_cmd, world| {
297        ensure_resources(world);
298
299        let players: Vec<PlayerId> = world.resource::<TurnResource>().0.players().to_vec();
300
301        let mut deck = build_deck();
302        {
303            let mut rng = world.resource_mut::<RandomResource>();
304            rng.0.shuffle(&mut deck);
305        }
306
307        let mut deck_res = world.resource_mut::<DeckResource>();
308        deck_res.discard_pile.clear();
309        for player in &players {
310            let mut hand = Vec::new();
311            for _ in 0..MAX_HAND_SIZE {
312                if let Some(card) = deck.pop() {
313                    hand.push(card);
314                }
315            }
316            deck_res.hands.insert(*player, hand);
317            deck_res.distance.insert(*player, 0);
318            deck_res.hazard.insert(*player, None);
319            deck_res.cards_200_played.insert(*player, 0);
320        }
321        deck_res.draw_pile = deck;
322        deck_res.target_distance = TARGET_DISTANCE;
323
324        Ok(vec![])
325    });
326
327    rules.register::<DrawCardCommand, _>(|cmd, world| {
328        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
329            return Err(invalid("no es tu turno"));
330        }
331
332        let mut deck = world.resource_mut::<DeckResource>();
333        let hand_len = deck.hands.get(&cmd.player).map(Vec::len).unwrap_or(0);
334        if hand_len >= MAX_HAND_SIZE {
335            return Err(invalid("tu mano ya está llena, jugá una carta primero"));
336        }
337
338        let needs_reshuffle = deck.draw_pile.is_empty() && !deck.discard_pile.is_empty();
339        let discarded_for_shuffle = if needs_reshuffle {
340            std::mem::take(&mut deck.discard_pile)
341        } else {
342            Vec::new()
343        };
344        let _ = deck; // release mutable borrow before accessing RandomResource
345
346        let shuffled = if needs_reshuffle {
347            let mut d = discarded_for_shuffle;
348            world.resource_mut::<RandomResource>().0.shuffle(&mut d);
349            d
350        } else {
351            Vec::new()
352        };
353
354        let mut deck = world.resource_mut::<DeckResource>();
355        if needs_reshuffle {
356            deck.draw_pile = shuffled;
357        }
358
359        let card = deck
360            .draw_pile
361            .pop()
362            .ok_or_else(|| invalid("no quedan cartas en el mazo"))?;
363        deck.hands.entry(cmd.player).or_default().push(card);
364
365        Ok(vec![Box::new(CardDrawnEvent {
366            player: cmd.player,
367            card,
368        }) as Box<dyn Event>])
369    });
370
371    rules.register::<PlayCardCommand, _>(|cmd, world| {
372        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
373            return Err(invalid("no es tu turno"));
374        }
375
376        let opponent = world.resource::<TurnResource>().0.peek_next();
377        let card = {
378            let deck = world.resource::<DeckResource>();
379            let hand = deck
380                .hands
381                .get(&cmd.player)
382                .ok_or_else(|| invalid("no tenés mano en esta partida"))?;
383            if cmd.card_index >= hand.len() {
384                return Err(invalid("ese índice de carta no existe en tu mano"));
385            }
386            hand[cmd.card_index]
387        };
388
389        let is_safety = matches!(card, Card::Safety(_) | Card::RightOfWay);
390        let events = match card {
391            Card::Distance(value) => validate_and_play_distance(cmd.player, value, world)?,
392            Card::Hazard(hazard) => validate_and_play_hazard(cmd.player, opponent, hazard, world)?,
393            Card::Remedy(hazard) => validate_and_play_remedy(cmd.player, hazard, world)?,
394            Card::Safety(safety) => validate_and_play_safety(cmd.player, safety, world)?,
395            Card::RightOfWay => validate_and_play_safety(cmd.player, Safety::RightOfWay, world)?,
396        };
397
398        world
399            .resource_mut::<DeckResource>()
400            .hands
401            .get_mut(&cmd.player)
402            .unwrap()
403            .remove(cmd.card_index);
404
405        // Jugar una carta de seguridad otorga turno extra (no avanza el turno)
406        if !is_safety {
407            world.resource_mut::<TurnResource>().0.advance();
408        }
409
410        Ok(events)
411    });
412
413    rules.register::<PlayDistanceCommand, _>(|cmd, world| {
414        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
415            return Err(invalid("no es tu turno"));
416        }
417        let events = validate_and_play_distance(cmd.player, cmd.distance, world)?;
418        world.resource_mut::<TurnResource>().0.advance();
419        Ok(events)
420    });
421
422    rules.register::<PlayHazardCommand, _>(|cmd, world| {
423        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
424            return Err(invalid("no es tu turno"));
425        }
426        let events = validate_and_play_hazard(cmd.player, cmd.target, cmd.hazard, world)?;
427        world.resource_mut::<TurnResource>().0.advance();
428        Ok(events)
429    });
430
431    rules.register::<PlaySafetyCommand, _>(|cmd, world| {
432        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
433            return Err(invalid("no es tu turno"));
434        }
435        let events = validate_and_play_safety(cmd.player, cmd.safety, world)?;
436        let mut deck = world.resource_mut::<DeckResource>();
437        if let Some(hand) = deck.hands.get_mut(&cmd.player) {
438            if let Some(pos) = hand.iter().position(|&c| match c {
439                Card::Safety(s) if s == cmd.safety => true,
440                Card::RightOfWay if cmd.safety == Safety::RightOfWay => true,
441                _ => false,
442            }) {
443                hand.remove(pos);
444            }
445        }
446        // Turno extra por jugar carta de seguridad
447        Ok(events)
448    });
449
450    rules.register::<DiscardCardCommand, _>(|cmd, world| {
451        if !world.resource::<TurnResource>().0.is_current(cmd.player) {
452            return Err(invalid("no es tu turno"));
453        }
454
455        let mut deck = world.resource_mut::<DeckResource>();
456        let hand = deck
457            .hands
458            .get_mut(&cmd.player)
459            .ok_or_else(|| invalid("no tenés mano en esta partida"))?;
460        if cmd.card_index >= hand.len() {
461            return Err(invalid("ese índice de carta no existe en tu mano"));
462        }
463
464        let card = hand.remove(cmd.card_index);
465        deck.discard_pile.push(card);
466
467        let _ = deck;
468        world.resource_mut::<TurnResource>().0.advance();
469
470        Ok(vec![Box::new(CardDiscardedEvent {
471            player: cmd.player,
472            card,
473        }) as Box<dyn Event>])
474    });
475
476    rules.register::<CoupFourreCommand, _>(|cmd, world| {
477        ensure_resources(world);
478
479        let opponent = world
480            .resource::<TurnResource>()
481            .0
482            .players()
483            .iter()
484            .copied()
485            .find(|&p| p != cmd.player)
486            .ok_or_else(|| invalid("no hay rival"))?;
487
488        let mut deck = world.resource_mut::<DeckResource>();
489
490        let hazard = deck
491            .hazard
492            .get(&cmd.player)
493            .copied()
494            .flatten()
495            .ok_or_else(|| invalid("no tenés ningún peligro para responder con coup fourré"))?;
496
497        let matching_safety = match hazard {
498            Hazard::SpeedLimit => Safety::RightOfWay,
499            Hazard::OutOfGas => Safety::ExtraTank,
500            Hazard::FlatTire => Safety::PunctureProof,
501            Hazard::Accident => Safety::DrivingAce,
502        };
503
504        let hand = deck
505            .hands
506            .get_mut(&cmd.player)
507            .ok_or_else(|| invalid("no tenés mano en esta partida"))?;
508
509        let safety_pos = hand.iter().position(|&c| match c {
510            Card::Safety(s) if s == matching_safety => true,
511            Card::RightOfWay if matching_safety == Safety::RightOfWay => true,
512            _ => false,
513        });
514
515        let mut events: Vec<Box<dyn Event>> = Vec::new();
516
517        if let Some(pos) = safety_pos {
518            let card = hand.remove(pos);
519            let s = match card {
520                Card::Safety(sf) => sf,
521                _ => Safety::RightOfWay,
522            };
523            deck.safeties_played
524                .entry(cmd.player)
525                .or_default()
526                .insert(s);
527            deck.hazard.insert(cmd.player, None);
528            deck.hazard.insert(opponent, Some(hazard));
529
530            // Bono de +100 km por Coup Fourré
531            let current_dist = *deck.distance.get(&cmd.player).unwrap_or(&0);
532            let new_dist = (current_dist + 100).min(deck.target_distance);
533            deck.distance.insert(cmd.player, new_dist);
534
535            events.push(Box::new(SafetyPlayedEvent {
536                player: cmd.player,
537                safety: s,
538            }));
539            events.push(Box::new(DistancePlayedEvent {
540                player: cmd.player,
541                value: 100,
542                new_distance: new_dist,
543            }));
544            events.push(Box::new(HazardPlayedEvent {
545                player: cmd.player,
546                target: opponent,
547                hazard,
548            }));
549
550            if new_dist == deck.target_distance {
551                events.push(Box::new(GameOverEvent { winner: cmd.player }));
552            }
553
554            let _ = deck;
555            if s == Safety::RightOfWay {
556                world
557                    .resource_mut::<RightOfWayResource>()
558                    .active
559                    .insert(cmd.player);
560            }
561            if hazard == Hazard::SpeedLimit {
562                let mut speed_res = world.resource_mut::<SpeedLimitResource>();
563                speed_res.limited.remove(&cmd.player);
564                speed_res.limited.insert(opponent);
565            }
566        } else {
567            let remedy_pos = hand
568                .iter()
569                .position(|&c| c == Card::Remedy(hazard))
570                .ok_or_else(|| {
571                    invalid(
572                        "no tenés la carta de seguridad o el remedio correspondiente en la mano",
573                    )
574                })?;
575
576            hand.remove(remedy_pos);
577            deck.hazard.insert(cmd.player, None);
578            deck.hazard.insert(opponent, Some(hazard));
579
580            events.push(Box::new(RemedyPlayedEvent {
581                player: cmd.player,
582                hazard,
583            }));
584            events.push(Box::new(HazardPlayedEvent {
585                player: cmd.player,
586                target: opponent,
587                hazard,
588            }));
589
590            let _ = deck;
591            if hazard == Hazard::SpeedLimit {
592                let mut speed_res = world.resource_mut::<SpeedLimitResource>();
593                speed_res.limited.remove(&cmd.player);
594                speed_res.limited.insert(opponent);
595            }
596        }
597
598        Ok(events)
599    });
600
601    rules
