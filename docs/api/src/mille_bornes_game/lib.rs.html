31/// Arma la Session y la deja lista para jugar: ya corrió el
32/// `SetupDeckCommand` (mazo barajado, manos iniciales repartidas).
33pub fn new_mille_bornes_session(
34    session_id: SessionId,
35    seed: u64,
36    player_a: PlayerId,
37    player_b: PlayerId,
38) -> Session {
39    let mut session = Session::new_with_metadata(
40        session_id,
41        "Mille Bornes",
42        seed,
43        vec![player_a, player_b],
44        build_mille_bornes_rules(),
45    );
46    session.world_mut().insert_resource(DeckResource::default());
47    session
48        .world_mut()
49        .insert_resource(SpeedLimitResource::default());
50    session
51        .world_mut()
52        .insert_resource(RightOfWayResource::default());
53
54    // El comando de setup no tiene issuer, así que cualquier "sender" es
55    // válido para la validación de identidad — usamos el primer jugador
56    // por convención, no importa cuál, ya que `issuer()` es `None` acá.
57    session
58        .process_command(player_a, Box::new(SetupDeckCommand))
59        .expect("el setup del mazo no debería poder fallar");
60
61    session
62}
