1//! Inteligencia Artificial heurística basada en reglas para Mille Bornes.
2//!
3//! Prioridad de jugada (`Medium` / `Hard` / `Expert`):
4//! 1. `Safety` (`Card::Safety` / `Card::RightOfWay`) si está en la mano.
5//! 2. `Remedy` (`Card::Remedy`) si el jugador IA está bloqueado por un peligro o límite de velocidad.
6//! 3. `Hazard` (`Card::Hazard`) al oponente si tiene buen avance y es legal atacarlo.
7//! 4. `Distance` (`Card::Distance`) máxima posible que sea legal jugar.
8//!
9//! En `AiDifficulty::Easy`, ignora las prioridades 1 y 3.
10

27/// Comando devuelto por la IA de Mille Bornes.
28#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
29pub enum MilleBornesCommand {
30    PlayCard(PlayCardCommand),
31    PlaySafety(PlaySafetyCommand),
32    PlayHazard(PlayHazardCommand),
33    PlayDistance(PlayDistanceCommand),
34    DrawCard(DrawCardCommand),
35    DiscardCard(DiscardCardCommand),
36}

38impl MilleBornesCommand {
39    /// Convierte el comando de IA en un `Box<dyn Command>` ejecutable por `Session`.
40    pub fn into_command(self) -> Box<dyn Command> {
41        match self {
42            Self::PlayCard(cmd) => Box::new(cmd),
43            Self::PlaySafety(cmd) => Box::new(cmd),
44            Self::PlayHazard(cmd) => Box::new(cmd),
45            Self::PlayDistance(cmd) => Box::new(cmd),
46            Self::DrawCard(cmd) => Box::new(cmd),
47            Self::DiscardCard(cmd) => Box::new(cmd),
48        }
49    }
50
51    /// Índice de carta en la mano si el comando es `PlayCard` o `DiscardCard`.
52    pub fn card_index(&self) -> Option<usize> {
53        match self {
54            Self::PlayCard(cmd) => Some(cmd.card_index),
55            Self::DiscardCard(cmd) => Some(cmd.card_index),
56            _ => None,
57        }
58    }
59}

61fn resolve_opponent(state: &MilleBornesState, ai: PlayerId) -> PlayerId {
62    if let Some(turn) = state.world().get_resource::<TurnResource>() {
63        if let Some(&other) = turn.0.players().iter().find(|&&p| p != ai) {
64            return other;
65        }
66    }
67    if let Some(deck) = state.world().get_resource::<DeckResource>() {
68        if let Some(&other) = deck.hands.keys().find(|&&p| p != ai) {
69            return other;
70        }
71    }
72    if ai.0 == 1 {
73        PlayerId(2)
74    } else {
75        PlayerId(1)
76    }
77}

79fn can_play_hazard_on(
80    deck: &DeckResource,
81    speed_limit: Option<&SpeedLimitResource>,
82    right_of_way: Option<&RightOfWayResource>,
83    opponent: PlayerId,
84    hazard: Hazard,
85) -> bool {
86    let opp_has_row = right_of_way
87        .map(|r| r.active.contains(&opponent))
88        .unwrap_or(false);
89    if opp_has_row && hazard != Hazard::OutOfGas {
90        return false;
91    }
92
93    if hazard == Hazard::SpeedLimit {
94        let already_limited = speed_limit
95            .map(|s| s.limited.contains(&opponent))
96            .unwrap_or(false)
97            || deck.hazard.get(&opponent).copied().flatten() == Some(Hazard::SpeedLimit);
98        !already_limited
99    } else {
100        deck.hazard.get(&opponent).copied().flatten().is_none()
101    }
102}

104fn can_play_distance_value(
105    deck: &DeckResource,
106    speed_limit: Option<&SpeedLimitResource>,
107    right_of_way: Option<&RightOfWayResource>,
108    ai: PlayerId,
109    value: u32,
110) -> bool {
111    let has_row = right_of_way.map(|r| r.active.contains(&ai)).unwrap_or(false);
112    let is_speed_limited = speed_limit.map(|s| s.limited.contains(&ai)).unwrap_or(false)
113        || deck.hazard.get(&ai).copied().flatten() == Some(Hazard::SpeedLimit);
114
115    if is_speed_limited && !has_row && value > 50 {
116        return false;
117    }
118
119    if let Some(h) = deck.hazard.get(&ai).copied().flatten() {
120        if h != Hazard::SpeedLimit && (!has_row || h == Hazard::OutOfGas) {
121            return false;
122        }
123    }
124
125    if value == 200 {
126        let count_200 = deck.cards_200_played.get(&ai).copied().unwrap_or(0);
127        if count_200 >= 2 {
128            return false;
129        }
130    }
131
132    let current = deck.distance.get(&ai).copied().unwrap_or(0);
133    current + value <= deck.target_distance
134}

136/// Selecciona la mejor jugada para `ai` en Mille Bornes según `difficulty`:
137/// 1) Safety si en mano (ignorada en `Easy`),
138/// 2) Remedy si bloqueado,
139/// 3) Hazard al oponente si tiene buen avance (ignorada en `Easy`),
140/// 4) Distance máxima posible.
141pub fn best_move(
142    state: &MilleBornesState,
143    ai: PlayerId,
144    difficulty: AiDifficulty,
145) -> Option<MilleBornesCommand> {
146    let deck = state.world().get_resource::<DeckResource>()?;
147    let speed_limit = state.world().get_resource::<SpeedLimitResource>();
148    let right_of_way = state.world().get_resource::<RightOfWayResource>();
149    let hand = deck.hands.get(&ai)?;
150
151    if hand.is_empty() {
152        if !deck.draw_pile.is_empty() || !deck.discard_pile.is_empty() {
153            return Some(MilleBornesCommand::DrawCard(DrawCardCommand { player: ai }));
154        }
155        return None;
156    }
157
158    let opponent = resolve_opponent(state, ai);
159
160    // 1) Safety si en mano (ignorada en Easy)
161    if difficulty != AiDifficulty::Easy {
162        for (idx, card) in hand.iter().enumerate() {
163            if matches!(card, Card::Safety(_) | Card::RightOfWay) {
164                return Some(MilleBornesCommand::PlayCard(PlayCardCommand {
165                    player: ai,
166                    card_index: idx,
167                }));
168            }
169        }
170    }
171
172    // 2) Remedy si bloqueado
173    let current_hazard = deck.hazard.get(&ai).copied().flatten();
174    let is_speed_limited = speed_limit.map(|s| s.limited.contains(&ai)).unwrap_or(false)
175        || current_hazard == Some(Hazard::SpeedLimit);
176
177    if let Some(h) = current_hazard {
178        if h != Hazard::SpeedLimit {
179            if let Some(idx) = hand.iter().position(|&c| c == Card::Remedy(h)) {
180                return Some(MilleBornesCommand::PlayCard(PlayCardCommand {
181                    player: ai,
182                    card_index: idx,
183                }));
184            }
185        }
186    }
187    if is_speed_limited {
188        if let Some(idx) = hand
189            .iter()
190            .position(|&c| c == Card::Remedy(Hazard::SpeedLimit))
191        {
192            return Some(MilleBornesCommand::PlayCard(PlayCardCommand {
193                player: ai,
194                card_index: idx,
195            }));
196        }
197    }
198
199    // 3) Hazard al oponente si tiene buen avance (ignorada en Easy)
200    if difficulty != AiDifficulty::Easy {
201        let opp_distance = deck.distance.get(&opponent).copied().unwrap_or(0);
202        let ai_distance = deck.distance.get(&ai).copied().unwrap_or(0);
203        let opp_has_good_progress = opp_distance >= 50 || opp_distance > ai_distance;
204
205        if opp_has_good_progress {
206            for (idx, card) in hand.iter().enumerate() {
207                if let Card::Hazard(hazard) = *card {
208                    if can_play_hazard_on(deck, speed_limit, right_of_way, opponent, hazard) {
209                        return Some(MilleBornesCommand::PlayCard(PlayCardCommand {
210                            player: ai,
211                            card_index: idx,
212                        }));
213                    }
214                }
215            }
216        }
217    }
218
219    // 4) Distance máxima posible
220    let mut best_distance_idx: Option<usize> = None;
221    let mut best_distance_val: u32 = 0;
222
223    for (idx, card) in hand.iter().enumerate() {
224        if let Card::Distance(val) = *card {
225            if can_play_distance_value(deck, speed_limit, right_of_way, ai, val)
226                && val > best_distance_val
227            {
228                best_distance_val = val;
229                best_distance_idx = Some(idx);
230            }
231        }
232    }
233
234    if let Some(idx) = best_distance_idx {
235        return Some(MilleBornesCommand::PlayCard(PlayCardCommand {
236            player: ai,
237            card_index: idx,
238        }));
239    }
240
241    // Si no puede jugar distancia y no está en Easy, intenta jugar cualquier Hazard válido aunque el rival tenga 0 km
242    if difficulty != AiDifficulty::Easy {
243        for (idx, card) in hand.iter().enumerate() {
244            if let Card::Hazard(hazard) = *card {
245                if can_play_hazard_on(deck, speed_limit, right_of_way, opponent, hazard) {
246                    return Some(MilleBornesCommand::PlayCard(PlayCardCommand {
247                        player: ai,
248                        card_index: idx,
249                    }));
250                }
251            }
252        }
253    }
254
255    // Fallback: robar si hay hueco en mano, o descartar una carta no jugable
256    if hand.len() < MAX_HAND_SIZE && (!deck.draw_pile.is_empty() || !deck.discard_pile.is_empty()) {
257        return Some(MilleBornesCommand::DrawCard(DrawCardCommand { player: ai }));
258    }
259
260    Some(MilleBornesCommand::DiscardCard(DiscardCardCommand {
261        player: ai,
262        card_index: 0,
263    }))
264}

266#[cfg(test)]
267mod tests {
268    use super::*;
269    use crate::card::Safety;
270    use crate::new_mille_bornes_session;
271
272    const P1: PlayerId = PlayerId(1);
273    const P2: PlayerId = PlayerId(2);
274
275    #[test]
276    fn mille_bornes_ai_plays_safety_first() {
277        let mut session = new_mille_bornes_session(SessionId(1), 42, P1, P2);
278
279        {
280            let mut deck = session.world_mut().resource_mut::<DeckResource>();
281            deck.hands.insert(
282                P1,
283                vec![
284                    Card::Distance(100),
285                    Card::Hazard(Hazard::FlatTire),
286                    Card::Safety(Safety::RightOfWay),
287                    Card::Distance(200),
288                ],
289            );
290            deck.distance.insert(P2, 150);
291        }
292
293        // En dificultad Hard/Medium, la prioridad 1 es jugar la carta Safety (índice 2)
294        let hard_cmd = best_move(&session, P1, AiDifficulty::Hard).unwrap();
295        assert_eq!(
296            hard_cmd,
297            MilleBornesCommand::PlayCard(PlayCardCommand {
298                player: P1,
299                card_index: 2,
300            })
301        );
302
303        // En dificultad Easy, ignora las prioridades 1 (Safety) y 3 (Hazard) y juega la mayor Distance (200 en índice 3)
304        let easy_cmd = best_move(&session, P1, AiDifficulty::Easy).unwrap();
305        assert_eq!(
306            easy_cmd,
307            MilleBornesCommand::PlayCard(PlayCardCommand {
308                player: P1,
309                card_index: 3,
310            })
311        );
312    }
