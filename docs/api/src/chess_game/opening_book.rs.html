4//! Libro de aperturas ECO compacto (A00-E99) indexado por hash Zobrist de 64 bits.
5//!
6//! En los niveles `AiDifficulty::Hard` y `AiDifficulty::Expert`, la IA consulta
7//! `OpeningBook::lookup(zobrist_hash)` para responder instantáneamente con aperturas
8//! teóricas ponderadas por frecuencia de uso.
9

19/// Capacidad máxima de posiciones en el libro compacto para mantener el binario < 100 KB.
20pub const MAX_BOOK_POSITIONS: usize = 5000;
21
22/// Bytes binarios compactos del libro ECO embebidos en tiempo de compilación.
23pub const COMPACT_BOOK_BYTES: &[u8] = include_bytes!("../data/openings_eco_compact.bin");
24
25/// Movimiento de libro ponderado por frecuencia de uso en partidas teóricas ECO.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub struct ChessMove {
28    /// Casilla de origen.
29    pub from: Position,
30    /// Casilla de destino.
31    pub to: Position,
32    /// Peso / frecuencia relativa en el repertorio ECO.
33    pub weight: u16,
34}
35
36impl ChessMove {
37    /// Convierte el movimiento de libro en un `MovePieceCommand` ejecutable para `player`.
38    pub fn to_command(self, player: PlayerId) -> MovePieceCommand {
39        MovePieceCommand {
40            player,
41            from: self.from,
42            to: self.to,
43            promote_to: None,
44        }
45    }
46}
47
48/// Calcula el hash Zobrist determinista de 64 bits para una posición de ajedrez.
49pub fn compute_zobrist_hash(board: &BoardResource, sides: &Sides, turn: PlayerId) -> u64 {
50    fn splitmix64(mut x: u64) -> u64 {
51        x = x.wrapping_add(0x9e3779b97f4a7c15);
52        x = (x ^ (x >> 30)).wrapping_mul(0xbf58476d1ce4e5b9);
53        x = (x ^ (x >> 27)).wrapping_mul(0x94d049bb133111eb);
54        x ^ (x >> 31)
55    }
56
57    let mut hash = 0u64;
58    for (&pos, piece) in &board.pieces {
59        let color_idx: u64 = if piece.owner == sides.white { 0 } else { 1 };
60        let kind_idx: u64 = match piece.kind {
61            PieceKind::Pawn => 1,
62            PieceKind::Knight => 2,
63            PieceKind::Bishop => 3,
64            PieceKind::Rook => 4,
65            PieceKind::Queen => 5,
66            PieceKind::King => 6,
67        };
68        let square_idx = u64::from(pos.row) * 8 + u64::from(pos.col);
69        let key_seed = (square_idx << 8) | (color_idx << 4) | kind_idx;
70        hash ^= splitmix64(key_seed);
71    }
72
73    if turn == sides.black {
74        hash ^= splitmix64(0xFFFF_0001);
75    }
76    hash
77}

79/// Libro de aperturas ECO (A00-E99) indexado por hash Zobrist.
80pub struct OpeningBook {
81    entries: HashMap<u64, Vec<ChessMove>>,
82}
83
84static GLOBAL_BOOK: OnceLock<OpeningBook> = OnceLock::new();
85
86impl OpeningBook {
87    /// Devuelve la instancia global del libro de aperturas cargada en memoria.
88    pub fn global() -> &'static Self {
89        GLOBAL_BOOK.get_or_init(Self::load_default)
90    }
91
92    /// Busca el mejor movimiento teórico (mayor peso) para `zobrist_hash`.
93    pub fn lookup(zobrist_hash: u64) -> Option<ChessMove> {
94        let list = Self::global().entries.get(&zobrist_hash)?;
95        list.iter().copied().max_by_key(|m| m.weight)
96    }
97
98    /// Consulta el libro sólo si `difficulty.use_opening_book()` es `true`; de lo contrario retorna `None`.
99    pub fn lookup_with_difficulty(
100        zobrist_hash: u64,
101        difficulty: AiDifficulty,
102    ) -> Option<ChessMove> {
103        if !difficulty.use_opening_book() {
104            return None;
105        }
106        Self::lookup(zobrist_hash)
107    }
108
109    /// Consulta el libro para la sesión actual si `difficulty.use_opening_book()` está activo.
110    pub fn lookup_session(
111        session: &Session,
112        player: PlayerId,
113        difficulty: AiDifficulty,
114    ) -> Option<MovePieceCommand> {
115        if !difficulty.use_opening_book() {
116            return None;
117        }
118        let board = session.world().get_resource::<BoardResource>()?;
119        let sides = session.world().get_resource::<Sides>()?;
120        let zh = compute_zobrist_hash(board, sides, player);
121        let book_mv = Self::lookup(zh)?;
122        Some(book_mv.to_command(player))
123    }
124
125    fn load_default() -> Self {
126        let mut entries: HashMap<u64, Vec<ChessMove>> = HashMap::new();
127
128        // 1. Decodificar entradas del archivo binario compacto (registros de 14 bytes)
129        for chunk in COMPACT_BOOK_BYTES.chunks_exact(14) {
130            if entries.len() >= MAX_BOOK_POSITIONS {
131                break;
132            }
133            let zh = u64::from_le_bytes([
134                chunk[0], chunk[1], chunk[2], chunk[3], chunk[4], chunk[5], chunk[6], chunk[7],
135            ]);
136            let from = Position::new(chunk[8], chunk[9]);
137            let to = Position::new(chunk[10], chunk[11]);
138            let weight = u16::from_le_bytes([chunk[12], chunk[13]]);
139            if from.row < 8 && from.col < 8 && to.row < 8 && to.col < 8 {
140                entries
141                    .entry(zh)
142                    .or_default()
143                    .push(ChessMove { from, to, weight });
144            }
145        }
146
147        // 2. Poblar líneas ECO principales (A00-E99) sobre el hash Zobrist canónico del motor
148        type EcoCoord = [u8; 2];
149        type EcoMovePair = (EcoCoord, EcoCoord);
150        type EcoLine<'a> = (&'a [EcoMovePair], u16);
151
152        let p1 = PlayerId(1);
153        let p2 = PlayerId(2);
154        let eco_lines: &[EcoLine<'_>] = &[
155            // C50 Giuoco Piano / Italiana: 1.e4 e5 2.Nf3 Nc6 3.Bc4 Bc5
156            (
157                &[
158                    ([1, 4], [3, 4]),
159                    ([6, 4], [4, 4]),
160                    ([0, 6], [2, 5]),
161                    ([7, 1], [5, 2]),
162                    ([0, 5], [3, 2]),
163                    ([7, 5], [4, 2]),
164                ],
165                100,
166            ),
167            // C60 Ruy López / Española: 1.e4 e5 2.Nf3 Nc6 3.Bb5 a6
168            (
169                &[
170                    ([1, 4], [3, 4]),
171                    ([6, 4], [4, 4]),
172                    ([0, 6], [2, 5]),
173                    ([7, 1], [5, 2]),
174                    ([0, 5], [4, 1]),
175                    ([6, 0], [5, 0]),
176                ],
177                95,
178            ),
179            // B20 Defensa Siciliana: 1.e4 c5 2.Nf3 d6 3.d4 cxd4
180            (
181                &[
182                    ([1, 4], [3, 4]),
183                    ([6, 2], [4, 2]),
184                    ([0, 6], [2, 5]),
185                    ([6, 3], [5, 3]),
186                    ([1, 3], [3, 3]),
187                    ([4, 2], [3, 3]),
188                ],
189                90,
190            ),
191            // D06 Gambito de Dama: 1.d4 d5 2.c4 e6 3.Nc3 Nf6
192            (
193                &[
194                    ([1, 3], [3, 3]),
195                    ([6, 3], [4, 3]),
196                    ([1, 2], [3, 2]),
197                    ([6, 4], [5, 4]),
198                    ([0, 1], [2, 2]),
199                    ([7, 6], [5, 5]),
200                ],
201                88,
202            ),
203            // E60 Defensa India de Rey: 1.d4 Nf6 2.c4 g6 3.Nc3 Bg7
204            (
205                &[
206                    ([1, 3], [3, 3]),
207                    ([7, 6], [5, 5]),
208                    ([1, 2], [3, 2]),
209                    ([6, 6], [5, 6]),
210                    ([0, 1], [2, 2]),
211                    ([7, 5], [6, 6]),
212                ],
213                85,
214            ),
215        ];
216
217        for &(moves, base_weight) in eco_lines {
218            let mut sim = crate::new_chess_session(SessionId(999), 1, p1, p2);
219            for (ply, &([fr, fc], [tr, tc])) in moves.iter().enumerate() {
220                if entries.len() >= MAX_BOOK_POSITIONS {
221                    break;
222                }
223                let turn = sim.current_player();
224                let zh = {
225                    let b = sim.world().get_resource::<BoardResource>().unwrap();
226                    let s = sim.world().get_resource::<Sides>().unwrap();
227                    compute_zobrist_hash(b, s, turn)
228                };
229                let candidate = ChessMove {
230                    from: Position::new(fr, fc),
231                    to: Position::new(tr, tc),
232                    weight: base_weight.saturating_sub(ply as u16),
233                };
234                let list = entries.entry(zh).or_default();
235                if !list
236                    .iter()
237                    .any(|m| m.from == candidate.from && m.to == candidate.to)
238                {
239                    list.push(candidate);
240                }
241                let _ = sim.process_command(turn, Box::new(candidate.to_command(turn)));
242            }
243        }
244
245        Self { entries }
246    }
247}

249#[cfg(test)]
250mod tests {
251    use super::*;
252    use crate::new_chess_session;
253
254    const P1: PlayerId = PlayerId(1);
255    const P2: PlayerId = PlayerId(2);
256
257    #[test]
258    fn test_opening_book_lookup_initial_position() {
259        let session = new_chess_session(SessionId(1), 1, P1, P2);
260        let board = session.world().get_resource::<BoardResource>().unwrap();
261        let sides = session.world().get_resource::<Sides>().unwrap();
262        let zh = compute_zobrist_hash(board, sides, P1);
263
264        let mv = OpeningBook::lookup(zh);
265        assert!(
266            mv.is_some(),
267            "La posición inicial debe estar cubierta en el OpeningBook"
268        );
269        let chosen = mv.unwrap();
270        // 1.e4 (row 1, col 4 -> row 3, col 4) tiene el mayor peso (100)
271        assert_eq!(chosen.from, Position::new(1, 4));
272        assert_eq!(chosen.to, Position::new(3, 4));
273    }
274
275    #[test]
276    fn test_opening_book_respects_difficulty_flag() {
277        let session = new_chess_session(SessionId(1), 1, P1, P2);
278        assert!(OpeningBook::lookup_session(&session, P1, AiDifficulty::Easy).is_none());
279        assert!(OpeningBook::lookup_session(&session, P1, AiDifficulty::Medium).is_none());
280        assert!(OpeningBook::lookup_session(&session, P1, AiDifficulty::Hard).is_some());
281        assert!(OpeningBook::lookup_session(&session, P1, AiDifficulty::Expert).is_some());
282    }
283
284    #[test]
285    fn test_opening_book_binary_size_under_100kb() {
286        assert!(
287            !COMPACT_BOOK_BYTES.is_empty(),
288            "El archivo binario de aperturas no debe estar vacío"
289        );
290        assert!(
291            COMPACT_BOOK_BYTES.len() < 100 * 1024,
292            "El binario compacto debe pesar menos de 100 KB"
293        );
294    }
