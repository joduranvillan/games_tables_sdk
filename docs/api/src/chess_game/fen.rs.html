11pub fn load_chess_position_from_fen(session: &mut Session, fen: &str) -> Result<(), EngineError> {
12    let (board, active_player) = parse_fen_with_player(fen)?;
13    session.world_mut().insert_resource(board);
14    if let Some(mut turn) = session
15        .world_mut()
16        .get_resource_mut::<open_table_engine::turn::TurnResource>()
17    {
18        turn.0.set_current_player(active_player);
19    }
20    Ok(())
21}
22
23pub fn from_fen(fen: &str) -> Result<BoardResource, EngineError> {
24    let (board, _) = parse_fen_with_player(fen)?;
25    Ok(board)
26}

28pub fn parse_fen_with_player(fen: &str) -> Result<(BoardResource, PlayerId), EngineError> {
29    let parts: Vec<&str> = fen.split_whitespace().collect();
30    if parts.is_empty() {
31        return Err(EngineError::InvalidCommand {
32            reason: "FEN vacío".into(),
33        });
34    }
35
36    let p1 = PlayerId(1);
37    let p2 = PlayerId(2);
38
39    let mut pieces = HashMap::new();
40    let ranks: Vec<&str> = parts[0].split('/').collect();
41    if ranks.len() != 8 {
42        return Err(EngineError::InvalidCommand {
43            reason: format!("FEN inválido: se esperaban 8 filas, hay {}", ranks.len()),
44        });
45    }
46
47    for (rank_idx, rank_str) in ranks.iter().enumerate() {
48        let row = (7 - rank_idx) as u8;
49        let mut col: u8 = 0;
50        for c in rank_str.chars() {
51            if let Some(digit) = c.to_digit(10) {
52                col += digit as u8;
53            } else {
54                if col >= 8 {
55                    return Err(EngineError::InvalidCommand {
56                        reason: "FEN inválido: columna fuera de límites".into(),
57                    });
58                }
59                let owner = if c.is_uppercase() { p1 } else { p2 };
60                let kind = match c.to_ascii_lowercase() {
61                    'p' => PieceKind::Pawn,
62                    'n' => PieceKind::Knight,
63                    'b' => PieceKind::Bishop,
64                    'r' => PieceKind::Rook,
65                    'q' => PieceKind::Queen,
66                    'k' => PieceKind::King,
67                    _ => {
68                        return Err(EngineError::InvalidCommand {
69                            reason: format!("Pieza FEN desconocida: {c}"),
70                        });
71                    }
72                };
73                pieces.insert(Position::new(row, col), Piece { owner, kind });
74                col += 1;
75            }
76        }
77    }
78
79    let active_player = if parts.len() > 1 && parts[1] == "b" {
80        p2
81    } else {
82        p1
83    };
84
85    let mut white_king_moved = true;
86    let mut white_rook_a_moved = true;
87    let mut white_rook_h_moved = true;
88    let mut black_king_moved = true;
89    let mut black_rook_a_moved = true;
90    let mut black_rook_h_moved = true;
91
92    if parts.len() > 2 {
93        let castling = parts[2];
94        if castling.contains('K') {
95            white_king_moved = false;
96            white_rook_h_moved = false;
97        }
98        if castling.contains('Q') {
99            white_king_moved = false;
100            white_rook_a_moved = false;
101        }
102        if castling.contains('k') {
103            black_king_moved = false;
104            black_rook_h_moved = false;
105        }
106        if castling.contains('q') {
107            black_king_moved = false;
108            black_rook_a_moved = false;
109        }
110    }
111
112    let mut en_passant_target = None;
113    if parts.len() > 3 && parts[3] != "-" {
114        let ep_str = parts[3];
115        if ep_str.len() == 2 {
116            let ep_bytes = ep_str.as_bytes();
117            let c = ep_bytes[0] as i32 - b'a' as i32;
118            let r = ep_bytes[1] as i32 - b'1' as i32;
119            if (0..8).contains(&c) && (0..8).contains(&r) {
120                en_passant_target = Some(Position::new(r as u8, c as u8));
121            }
122        }
123    }
124
125    let board = BoardResource {
126        pieces,
127        en_passant_target,
128        white_king_moved,
129        white_rook_a_moved,
130        white_rook_h_moved,
131        black_king_moved,
132        black_rook_a_moved,
133        black_rook_h_moved,
134    };
135
136    Ok((board, active_player))
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142
143    #[test]
144    fn test_cargar_posicion_desde_fen_inicializa_tablero_correcto() {
145        let initial_fen = "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1";
146        let (board_init, active_init) = parse_fen_with_player(initial_fen).unwrap();
147        assert_eq!(board_init.pieces.len(), 32);
148        assert_eq!(active_init, PlayerId(1));
149        assert_eq!(
150            board_init.get(Position::new(1, 4)),
151            Some(Piece {
152                owner: PlayerId(1),
153                kind: PieceKind::Pawn
154            })
155        );
156        assert_eq!(
157            board_init.get(Position::new(7, 4)),
158            Some(Piece {
159                owner: PlayerId(2),
160                kind: PieceKind::King
161            })
162        );
163
164        let middlegame_fen = "r1bqk2r/pp1pppbp/2n2np1/8/3NP3/2N1B3/PPP2PPP/R2QKB1R w KQkq - 3 7";
165        let (board_mid, active_mid) = parse_fen_with_player(middlegame_fen).unwrap();
166        assert_eq!(active_mid, PlayerId(1));
167        assert_eq!(
168            board_mid.get(Position::new(3, 3)),
169            Some(Piece {
170                owner: PlayerId(1),
171                kind: PieceKind::Knight
172            })
173        );
174        assert_eq!(
175            board_mid.get(Position::new(5, 2)),
176            Some(Piece {
177                owner: PlayerId(2),
178                kind: PieceKind::Knight
179            })
180        );
181        assert_eq!(
182            board_mid.get(Position::new(2, 4)),
183            Some(Piece {
184                owner: PlayerId(1),
185                kind: PieceKind::Bishop
186            })
187        );
188    }
