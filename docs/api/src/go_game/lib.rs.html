32pub fn new_go_session_with_size(
33    session_id: SessionId,
34    seed: u64,
35    size: usize,
36    black: PlayerId,
37    white: PlayerId,
38) -> Session {
39    let valid_size = if matches!(size, 9 | 13 | 19) { size } else { 9 };
40    let mut session =
41        Session::new_with_metadata(session_id, "Go", seed, vec![black, white], build_go_rules());
42    let board = BoardResource::new(valid_size, black, white).with_komi(13);
43    let mut history = BoardHistoryResource::new();
44    history.history.insert(board.compute_hash());
45    session.world_mut().insert_resource(board);
46    session.world_mut().insert_resource(history);
47    session
48        .world_mut()
49        .insert_resource(DeadMarkingResource::default());
50    session
51}

53pub fn new_go_session_with_komi(
54    session_id: SessionId,
55    seed: u64,
56    black: PlayerId,
57    white: PlayerId,
58    komi_half_points: u32,
59) -> Session {
60    let mut session =
61        Session::new_with_metadata(session_id, "Go", seed, vec![black, white], build_go_rules());
62    let board = BoardResource::new(9, black, white).with_komi(komi_half_points);
63    let mut history = BoardHistoryResource::new();
64    history.history.insert(board.compute_hash());
65    session.world_mut().insert_resource(board);
66    session.world_mut().insert_resource(history);
67    session
68        .world_mut()
69        .insert_resource(DeadMarkingResource::default());
70    session
