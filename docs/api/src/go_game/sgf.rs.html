9fn sgf_coord(x: usize, y: usize) -> String {
10    let col = (b'a' + (x.min(25) as u8)) as char;
11    let row = (b'a' + (y.min(25) as u8)) as char;
12    format!("{col}{row}")
13}
14
15impl BoardResource {
16    /// Exporta el estado actual del tablero de Go en formato estándar SGF (Smart Game Format FF\[4\]).
17    pub fn export_sgf(&self) -> String {
18        let mut sgf = format!(
19            "(;GM[1]FF[4]CA[UTF-8]AP[OpenTable]SZ[{}]KM[6.5]PB[Black]PW[White]",
20            self.size
21        );
22
23        for y in 0..self.size {
24            for x in 0..self.size {
25                if let Some(stone) = self.get_stone(x, y) {
26                    let tag = match stone {
27                        Stone::Black => "B",
28                        Stone::White => "W",
29                    };
30                    let coord = sgf_coord(x, y);
31                    sgf.push_str(&format!(";{tag}[{coord}]"));
32                }
33            }
34        }
35
36        sgf.push(')');
37        sgf
38    }
39}

41/// Rasgo de extensión para exportar una partida de Go a formato SGF desde `Session`.
42pub trait GoSgfExport {
43    fn export_sgf(&self) -> String;
44}
45
46impl GoSgfExport for Session {
47    fn export_sgf(&self) -> String {
48        export_session_sgf(self)
49    }
50}
51
52pub fn export_session_sgf(session: &Session) -> String {
53    let (size, black_player) = session
54        .world()
55        .get_resource::<BoardResource>()
56        .map(|b| (b.size, b.black_player))
57        .unwrap_or((9, PlayerId(1)));
58
59    let mut sgf = format!("(;GM[1]FF[4]CA[UTF-8]AP[OpenTable]SZ[{size}]KM[6.5]PB[Black]PW[White]");
60
61    let mut nodes_written = 0usize;
62
63    for (_tick, _elapsed, cmd) in session.command_log() {
64        let any_cmd = cmd.as_any();
65        if let Some(place) = any_cmd.downcast_ref::<PlaceStoneCommand>() {
66            let color = if place.player == black_player {
67                "B"
68            } else {
69                "W"
70            };
71            let coord = sgf_coord(place.x, place.y);
72            sgf.push_str(&format!(";{color}[{coord}]"));
73            nodes_written += 1;
74        } else if let Some(pass) = any_cmd.downcast_ref::<PassCommand>() {
75            let color = if pass.player == black_player {
76                "B"
77            } else {
78                "W"
79            };
80            sgf.push_str(&format!(";{color}[]"));
81            nodes_written += 1;
82        } else if let Some(handicap) = any_cmd.downcast_ref::<PlaceHandicapCommand>() {
83            if !handicap.positions.is_empty() {
84                sgf.push_str(&format!("HA[{}]", handicap.positions.len()));
85                sgf.push_str("AB");
86                for &(hx, hy) in &handicap.positions {
87                    let coord = sgf_coord(hx as usize, hy as usize);
88                    sgf.push_str(&format!("[{coord}]"));
89                }
90                nodes_written += 1;
91            }
92        }
93    }
94
95    if nodes_written == 0 {
96        if let Some(board) = session.world().get_resource::<BoardResource>() {
97            return board.export_sgf();
98        }
99    }
100
101    sgf.push(')');
102    sgf
103}
104
105pub fn export_to_sgf(session: &Session) -> String {
106    export_session_sgf(session)
107}

109#[cfg(test)]
110mod tests {
111    use super::*;
112    use crate::{new_go_session, PassCommand, PlaceStoneCommand};
113    use open_table_core::ids::{PlayerId, SessionId};
114
115    #[test]
116    fn test_export_sgf_incluye_metadatos_tamano_y_secuencia_de_piedras() {
117        let black = PlayerId(1);
118        let white = PlayerId(2);
119        let mut session = new_go_session(SessionId(1), 42, black, white);
120
121        // Negro juega en (4, 4) -> 'ee'
122        session
123            .process_command(
124                black,
125                Box::new(PlaceStoneCommand {
126                    player: black,
127                    x: 4,
128                    y: 4,
129                }),
130            )
131            .unwrap();
132
133        // Blanco juega en (2, 2) -> 'cc'
134        session
135            .process_command(
136                white,
137                Box::new(PlaceStoneCommand {
138                    player: white,
139                    x: 2,
140                    y: 2,
141                }),
142            )
143            .unwrap();
144
145        // Negro pasa -> ';B[]'
146        session
147            .process_command(black, Box::new(PassCommand { player: black }))
148            .unwrap();
149
150        let sgf = session.export_sgf();
151        assert!(sgf.starts_with("(;GM[1]FF[4]"));
152        assert!(sgf.contains("SZ[9]"));
153        assert!(sgf.contains(";B[ee];W[cc];B[]"));
154        assert!(sgf.ends_with(')'));
155    }
156
157    #[test]
158    fn test_board_resource_export_sgf_directo() {
159        let mut board = BoardResource::new(9, PlayerId(1), PlayerId(2));
160        board.set_stone(0, 0, Some(Stone::Black));
161        board.set_stone(8, 8, Some(Stone::White));
162
163        let sgf = board.export_sgf();
164        assert!(sgf.contains("SZ[9]"));
165        assert!(sgf.contains(";B[aa]"));
166        assert!(sgf.contains(";W[ii]"));
167    }
