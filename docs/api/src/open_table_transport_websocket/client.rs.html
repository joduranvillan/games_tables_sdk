27/// Intervalo de heartbeat ping/pong (15 segundos) para mantener vivas las conexiones
28/// detrás de proxies reversos y routers NAT.
29pub const HEARTBEAT_INTERVAL: Duration = Duration::from_secs(15);

31/// Número máximo de intentos de reconexión automática antes de declarar desconexión definitiva.
32pub const MAX_RECONNECT_ATTEMPTS: u32 = 5;
33
34/// Secuencia de backoff exponencial por intento en milisegundos (`500ms`, `1s`, `2s`, `4s`, hasta 5 intentos).
35pub const RECONNECT_BACKOFF_MS: [u64; 5] = [500, 1000, 2000, 4000, 4000];
36
37/// Devuelve la duración de espera de backoff para el intento `attempt` (`1..=5`).
38pub const fn reconnect_backoff_duration(attempt: u32) -> Duration {
39    let idx = if attempt == 0 {
40        0
41    } else if (attempt as usize) <= RECONNECT_BACKOFF_MS.len() {
42        (attempt as usize) - 1
43    } else {
44        RECONNECT_BACKOFF_MS.len() - 1
45    };
46    Duration::from_millis(RECONNECT_BACKOFF_MS[idx])
47}

49/// Estado de reconexión automática del cliente WebSocket Relay.
50#[derive(Debug, Clone, Copy, PartialEq, Eq)]
51pub enum ReconnectState {
52    /// Conexión establecida normalmente.
53    Connected,
54    /// Reconectando automáticamente con backoff exponencial (`attempt` de `1` a `max_attempts`).
55    Reconnecting { attempt: u32, max_attempts: u32 },
56    /// Desconectado definitivamente (sin sala activa o tras agotar los `max_attempts` intentos).
57    Disconnected,
58}

60/// Política configurable de reconexión con backoff exponencial para [`WsRelayClient`].
61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62pub struct ReconnectPolicy {
63    pub max_attempts: u32,
64    pub backoff_ms: [u64; 5],
65}

67impl Default for ReconnectPolicy {
68    fn default() -> Self {
69        Self {
70            max_attempts: MAX_RECONNECT_ATTEMPTS,
71            backoff_ms: RECONNECT_BACKOFF_MS,
72        }
73    }
74}

76impl ReconnectPolicy {
77    /// Duración de espera antes de ejecutar el intento `attempt` (`1..=max_attempts`).
78    pub fn delay_for_attempt(&self, attempt: u32) -> Duration {
79        if self.backoff_ms.is_empty() {
80            return Duration::from_millis(500);
81        }
82        let idx = attempt
83            .saturating_sub(1)
84            .min((self.backoff_ms.len() - 1) as u32) as usize;
85        Duration::from_millis(self.backoff_ms[idx])
86    }
87}

101    pub fn connect_as_new_player(
102        addr: impl ToSocketAddrs,
103        player_name: &str,
104    ) -> io::Result<(Self, PlayerId, String)> {
105        let sock_addr = addr
106            .to_socket_addrs()?
107            .next()
108            .ok_or_else(|| io::Error::new(ErrorKind::InvalidInput, "Dirección inválida"))?;
109
110        let url = format!("ws://{sock_addr}");
111        let (tx_to_ws, mut rx_from_client) = mpsc::unbounded_channel::<Vec<u8>>();
112        let (tx_handshake, mut rx_handshake) =
113            mpsc::unbounded_channel::<io::Result<(PlayerId, String)>>();
114
115        let incoming: IncomingQueue = Arc::new(Mutex::new(VecDeque::new()));
116        let incoming_clone = Arc::clone(&incoming);
117        let player_name = player_name.to_string();
118
119        let handle = thread::spawn(move || {
120            let rt = tokio::runtime::Builder::new_current_thread()
121                .enable_all()
122                .build()
123                .expect("Failed to build Tokio runtime");
124
125            rt.block_on(async move {
126                let ws_stream = match connect_async(&url).await {
127                    Ok((ws, _)) => ws,
128                    Err(e) => {
129                        let _ =
130                            tx_handshake.send(Err(io::Error::new(ErrorKind::ConnectionRefused, e)));
131                        return;
132                    }
133                };
134
135                let (mut ws_sender, mut ws_receiver) = ws_stream.split();
136
137                // Send Handshake
138                let req = HandshakeRequest::Join {
139                    player_name,
140                    protocol_version: PROTOCOL_VERSION.to_string(),
141                    route: None,
142                };
143                let req_bytes = serde_json::to_vec(&req).unwrap();
144                if ws_sender.send(wrap_binary(&req_bytes)).await.is_err() {
145                    let _ = tx_handshake.send(Err(io::Error::new(
146                        ErrorKind::ConnectionReset,
147                        "Send error",
148                    )));
149                    return;
150                }
151
152                // Wait for response
153                let resp_msg = match ws_receiver.next().await {
154                    Some(Ok(msg)) => extract_bytes(&msg),
155                    _ => None,
156                };
157
158                let (player_id, token) = match resp_msg {
159                    Some(bytes) => match serde_json::from_slice::<HandshakeResponse>(&bytes) {
160                        Ok(HandshakeResponse::Welcome {
161                            player_id,
162                            reconnect_token,
163                            ..
164                        }) => (player_id, reconnect_token),
165                        Ok(HandshakeResponse::Rejected { reason }) => {
166                            let _ = tx_handshake
167                                .send(Err(io::Error::new(ErrorKind::PermissionDenied, reason)));
168                            return;
169                        }
170                        _ => {
171                            let _ = tx_handshake.send(Err(io::Error::new(
172                                ErrorKind::InvalidData,
173                                "Respuesta inválida",
174                            )));
175                            return;
176                        }
177                    },
178                    None => {
179                        let _ = tx_handshake.send(Err(io::Error::new(
180                            ErrorKind::UnexpectedEof,
181                            "EOF handshake",
182                        )));
183                        return;
184                    }
185                };
186
187                let _ = tx_handshake.send(Ok((player_id, token)));
188
189                let send_task = tokio::spawn(async move {
190                    let mut heartbeat = tokio::time::interval(HEARTBEAT_INTERVAL);
191                    // El primer tick ocurre inmediatamente; lo consumimos para esperar 15s reales.
192                    heartbeat.tick().await;
193
194                    loop {
195                        tokio::select! {
196                            maybe_bytes = rx_from_client.recv() => {
197                                match maybe_bytes {
198                                    Some(bytes) => {
199                                        if ws_sender.send(wrap_binary(&bytes)).await.is_err() {
200                                            break;
201                                        }
202                                    }
203                                    None => break,
204                                }
205                            }
206                            _ = heartbeat.tick() => {
207                                if ws_sender.send(Message::Ping(Vec::new().into())).await.is_err() {
208                                    break;
209                                }
210                            }
211                        }
212                    }
213                });
214
215                while let Some(msg) = ws_receiver.next().await {
216                    match msg {
217                        Ok(m) => {
218                            if let Some(bytes) = extract_bytes(&m) {
219                                incoming_clone
220                                    .lock()
221                                    .unwrap()
222                                    .push_back((HOST_PLAYER_ID, bytes));
223                            }
224                        }
225                        Err(_) => break,
226                    }
227                }
228
229                send_task.abort();
230            });
231        });
232
233        match rx_handshake.blocking_recv() {
234            Some(Ok((player_id, token))) => Ok((
235                Self {
236                    tx: tx_to_ws,
237                    incoming,
238                    _runtime_thread: Some(handle),
239                },
240                player_id,
241                token,
242            )),
243            Some(Err(err)) => Err(err),
244            None => Err(io::Error::new(
245                ErrorKind::ConnectionAborted,
246                "Canal cerrado",
247            )),
248        }
249    }

319#[derive(Debug, Serialize, Deserialize)]
320struct EncodedRelayPayload {
321    #[serde(rename = "__ot_b64_hex")]
322    hex: String,
323}
324
325fn encode_bytes_to_value(bytes: &[u8]) -> Value {
326    if let Ok(s) = std::str::from_utf8(bytes) {
327        if let Ok(json_val) = serde_json::from_str::<Value>(s) {
328            return json_val;
329        }
330        return Value::String(s.to_string());
331    }
332    let mut hex = String::with_capacity(bytes.len() * 2);
333    for &b in bytes {
334        use std::fmt::Write;
335        let _ = write!(&mut hex, "{b:02x}");
336    }
337    serde_json::to_value(EncodedRelayPayload { hex }).unwrap_or(Value::Null)
338}
339
340fn decode_value_to_bytes(val: &Value) -> Vec<u8> {
341    if let Ok(encoded) = serde_json::from_value::<EncodedRelayPayload>(val.clone()) {
342        let mut out = Vec::with_capacity(encoded.hex.len() / 2);
343        let bytes = encoded.hex.as_bytes();
344        let mut i = 0;
345        while i + 1 < bytes.len() {
346            if let Ok(s) = std::str::from_utf8(&bytes[i..i + 2]) {
347                if let Ok(b) = u8::from_str_radix(s, 16) {
348                    out.push(b);
349                }
350            }
351            i += 2;
352        }
353        return out;
354    }
355    match val {
356        Value::String(s) => s.as_bytes().to_vec(),
357        other => serde_json::to_vec(other).unwrap_or_default(),
358    }
359}

361fn generate_fallback_room_code() -> String {
362    let mut buf = [0u8; 4];
363    if getrandom::fill(&mut buf).is_err() {
364        let ts = std::time::SystemTime::now()
365            .duration_since(std::time::UNIX_EPOCH)
366            .map(|d| d.as_nanos() as u64)
367            .unwrap_or(0x5A5A_A5A5);
368        for (i, b) in buf.iter_mut().enumerate() {
369            *b = ((ts >> (i * 8)) & 0xFF) as u8;
370        }
371    }
372    buf.iter()
373        .map(|&b| RELAY_CODE_ALPHABET[(b as usize) % RELAY_CODE_ALPHABET.len()] as char)
374        .collect()
375}
376
377fn is_valid_relay_code(code: &str) -> bool {
378    code.len() == 4
379        && code
380            .bytes()
381            .all(|b| RELAY_CODE_ALPHABET.contains(&b.to_ascii_uppercase()))
382}

468    /// Conecta al servidor Relay en `relay_url` con token opcional y una [`ReconnectPolicy`] específica.
469    pub fn connect_with_policy(
470        relay_url: &str,
471        auth_token: Option<&str>,
472        policy: ReconnectPolicy,
473    ) -> Self {
474        let client_id = LOCAL_HUB_CLIENT_ID.fetch_add(1, Ordering::Relaxed);
475        let url = relay_url.trim().to_string();
476        let initial_token = auth_token
477            .map(str::trim)
478            .filter(|t| !t.is_empty())
479            .map(ToOwned::to_owned);
480        let events: Arc<Mutex<VecDeque<RelayClientEvent>>> = Arc::new(Mutex::new(VecDeque::new()));
481        let incoming_payloads: IncomingQueue = Arc::new(Mutex::new(VecDeque::new()));
482        let connected_remote = Arc::new(AtomicBool::new(false));
483        let reconnect_state = Arc::new(Mutex::new(ReconnectState::Connected));
484        let shutdown_flag = Arc::new(AtomicBool::new(false));
485
486        let (cmd_tx, mut cmd_rx) = mpsc::unbounded_channel::<OutboundRelayCommand>();
487        let (conn_init_tx, conn_init_rx) = oneshot::channel::<bool>();
488
489        let events_clone = Arc::clone(&events);
490        let incoming_clone = Arc::clone(&incoming_payloads);
491        let connected_flag = Arc::clone(&connected_remote);
492        let reconnect_state_clone = Arc::clone(&reconnect_state);
493        let shutdown_flag_clone = Arc::clone(&shutdown_flag);
494        let url_for_thread = url.clone();
495
496        let worker = thread::spawn(move || {
497            let rt = match tokio::runtime::Builder::new_current_thread()
498                .enable_all()
499                .build()
500            {
501                Ok(rt) => rt,
502                Err(_) => {
503                    let _ = conn_init_tx.send(false);
504                    return;
505                }
506            };
507
508            rt.block_on(async move {
509                let connect_fut = connect_async(&url_for_thread);
510                let ws_stream =
511                    match tokio::time::timeout(Duration::from_millis(400), connect_fut).await {
512                        Ok(Ok((ws, _))) => {
513                            connected_flag.store(true, Ordering::SeqCst);
514                            *reconnect_state_clone.lock().unwrap() = ReconnectState::Connected;
515                            let _ = conn_init_tx.send(true);
516                            ws
517                        }
518                        _ => {
519                            connected_flag.store(false, Ordering::SeqCst);
520                            let _ = conn_init_tx.send(false);
521                            return;
522                        }
523                    };
524
525                let (mut ws_sink, mut ws_source) = ws_stream.split();
526                let mut current_auth_token = initial_token;
527                if let Some(ref tok) = current_auth_token {
528                    let auth_msg = serde_json::json!({
529                        "op": "auth",
530                        "token": tok,
531                    })
532                    .to_string();
533                    if ws_sink.send(Message::Text(auth_msg.into())).await.is_err() {
534                        connected_flag.store(false, Ordering::SeqCst);
535                        return;
536                    }
537                }
538                let mut heartbeat = tokio::time::interval(HEARTBEAT_INTERVAL);
539                heartbeat.tick().await;
540
541                let mut pending_create: Option<CreateReplySender> = None;
542                let mut pending_join: Option<(String, JoinReplySender)> = None;
543                let mut pending_rejoin: Option<RejoinReplySender> = None;
544
545                let mut active_room_code: Option<String> = None;
546                let mut active_player_id: u32 = 0;
547                let mut intentional_leave = false;
548
549                loop {
550                    let connection_dropped = tokio::select! {
551                        maybe_cmd = cmd_rx.recv() => {
552                            match maybe_cmd {
553                                Some(OutboundRelayCommand::Auth { token }) => {
554                                    current_auth_token = Some(token.clone());
555                                    let msg = serde_json::json!({
556                                        "op": "auth",
557                                        "token": token,
558                                    }).to_string();
559                                    ws_sink.send(Message::Text(msg.into())).await.is_err()
560                                }
561                                Some(OutboundRelayCommand::Create { game_id, reply }) => {
562                                    intentional_leave = false;
563                                    pending_create = Some(reply);
564                                    let msg = serde_json::json!({
565                                        "op": "create",
566                                        "game_id": game_id,
567                                        "is_public": true,
568                                    }).to_string();
569                                    if ws_sink.send(Message::Text(msg.into())).await.is_err() {
570                                        if let Some(r) = pending_create.take() {
571                                            let _ = r.send(Err("websocket send failed".to_string()));
572                                        }
573                                        true
574                                    } else {
575                                        false
576                                    }
577                                }
578                                Some(OutboundRelayCommand::Join { room_code, reply }) => {
579                                    intentional_leave = false;
580                                    pending_join = Some((room_code.clone(), reply));
581                                    let msg = serde_json::json!({
582                                        "op": "join",
583                                        "room_code": room_code,
584                                    }).to_string();
585                                    if ws_sink.send(Message::Text(msg.into())).await.is_err() {
586                                        if let Some((_, r)) = pending_join.take() {
587                                            let _ = r.send(Err("websocket send failed".to_string()));
588                                        }
589                                        true
590                                    } else {
591                                        false
592                                    }
593                                }
594                                Some(OutboundRelayCommand::Rejoin { room_code, player_id, reply }) => {
595                                    intentional_leave = false;
596                                    pending_rejoin = Some(reply);
597                                    let msg = match player_id {
598                                        Some(pid) if !pid.trim().is_empty() => serde_json::json!({
599                                            "op": "rejoin_room",
600                                            "room_code": room_code,
601                                            "player_id": pid,
602                                        }),
603                                        _ => serde_json::json!({
604                                            "op": "rejoin_room",
605                                            "room_code": room_code,
606                                        }),
607                                    }.to_string();
608                                    if ws_sink.send(Message::Text(msg.into())).await.is_err() {
609                                        if let Some(r) = pending_rejoin.take() {
610                                            let _ = r.send(Err("websocket send failed".to_string()));
611                                        }
612                                        true
613                                    } else {
614                                        false
615                                    }
616                                }
617                                Some(OutboundRelayCommand::Relay { data }) => {
618                                    let msg = serde_json::json!({
619                                        "op": "relay",
620                                        "data": data,
621                                    }).to_string();
622                                    ws_sink.send(Message::Text(msg.into())).await.is_err()
623                                }
624                                Some(OutboundRelayCommand::Leave) => {
625                                    intentional_leave = true;
626                                    active_room_code = None;
627                                    active_player_id = 0;
628                                    let msg = r#"{"op":"leave"}"#.to_string();
629                                    let _ = ws_sink.send(Message::Text(msg.into())).await;
630                                    false
631                                }
632                                None => {
633                                    break;
634                                }
635                            }
636                        }
637                        _ = heartbeat.tick() => {
638                            let ping_json = r#"{"op":"ping"}"#.to_string();
639                            if ws_sink.send(Message::Text(ping_json.into())).await.is_err() {
640                                true
641                            } else {
642                                ws_sink.send(Message::Ping(Vec::new().into())).await.is_err()
643                            }
644                        }
645                        maybe_ws = ws_source.next() => {
646                            match maybe_ws {
647                                Some(Ok(Message::Text(text))) => {
648                                    if let Ok(val) = serde_json::from_str::<Value>(&text) {
649                                        let op = val.get("op").and_then(|v| v.as_str()).unwrap_or("");
650                                        match op {
651                                            "created" => {
652                                                let code = val.get("room_code")
653                                                    .and_then(|v| v.as_str())
654                                                    .unwrap_or("")
655                                                    .to_string();
656                                                active_room_code = Some(code.clone());
657                                                active_player_id = 1;
658                                                if let Some(reply) = pending_create.take() {
659                                                    let _ = reply.send(Ok(code.clone()));
660                                                }
661                                                events_clone.lock().unwrap().push_back(
662                                                    RelayClientEvent::RoomCreated { room_code: code }
663                                                );
664                                            }
665                                            "joined" => {
666                                                let pid = val.get("player_id")
667                                                    .and_then(|v| v.as_u64())
668                                                    .unwrap_or(2) as u32;
669                                                let code = pending_join
670                                                    .as_ref()
671                                                    .map(|(c, _)| c.clone())
672                                                    .or_else(|| active_room_code.clone())
673                                                    .unwrap_or_default();
674                                                if !code.is_empty() {
675                                                    active_room_code = Some(code.clone());
676                                                }
677                                                active_player_id = pid;
678                                                if let Some((_, reply)) = pending_join.take() {
679                                                    let _ = reply.send(Ok(pid));
680                                                }
681                                                events_clone.lock().unwrap().push_back(
682                                                    RelayClientEvent::RoomJoined {
683                                                        success: true,
684                                                        player_id: PlayerId(pid),
685                                                        room_code: code,
686                                                    }
687                                                );
688                                            }
689                                            "rejoined" => {
690                                                let pid = val.get("player_id")
691                                                    .and_then(|v| v.as_u64())
692                                                    .map(|p| p as u32)
693                                                    .unwrap_or_else(|| active_player_id.max(1));
694                                                let code = val.get("room_code")
695                                                    .and_then(|v| v.as_str())
696                                                    .map(ToOwned::to_owned)
697                                                    .or_else(|| active_room_code.clone())
698                                                    .unwrap_or_default();
699                                                let seed = val.get("seed")
700                                                    .and_then(|v| v.as_u64())
701                                                    .unwrap_or(1);
702                                                let commands = val.get("commands")
703                                                    .and_then(|v| v.as_array())
704                                                    .cloned()
705                                                    .unwrap_or_default();
706                                                if !code.is_empty() {
707                                                    active_room_code = Some(code.clone());
708                                                }
709                                                active_player_id = pid;
710                                                if let Some(reply) = pending_rejoin.take() {
711                                                    let _ = reply.send(Ok((pid, seed, commands.clone())));
712                                                }
713                                                events_clone.lock().unwrap().push_back(
714                                                    RelayClientEvent::ConnectionRestored {
715                                                        room_code: code,
716                                                        player_id: PlayerId(pid),
717                                                        seed,
718                                                        commands,
719                                                    }
720                                                );
721                                            }
722                                            "peer_joined" | "opponent_reconnected" => {
723                                                let pid = val.get("player_id")
724                                                    .and_then(|v| v.as_u64())
725                                                    .unwrap_or(2) as u32;
726                                                events_clone.lock().unwrap().push_back(
727                                                    RelayClientEvent::PeerJoined {
728                                                        player_id: PlayerId(pid),
729                                                    }
730                                                );
731                                            }
732                                            "peer_left" => {
733                                                let pid = val.get("player_id")
734                                                    .and_then(|v| v.as_u64())
735                                                    .unwrap_or(0) as u32;
736                                                events_clone.lock().unwrap().push_back(
737                                                    RelayClientEvent::PeerLeft {
738                                                        player_id: PlayerId(pid),
739                                                    }
740                                                );
741                                            }
742                                            "relay" => {
743                                                let from = val.get("from")
744                                                    .and_then(|v| v.as_u64())
745                                                    .unwrap_or(0) as u32;
746                                                let bytes = val.get("data")
747                                                    .map(decode_value_to_bytes)
748                                                    .unwrap_or_default();
749                                                incoming_clone.lock().unwrap().push_back((PlayerId(from), bytes.clone()));
750                                                events_clone.lock().unwrap().push_back(
751                                                    RelayClientEvent::PayloadReceived {
752                                                        from: PlayerId(from),
753                                                        data: bytes,
754                                                    }
755                                                );
756                                            }
757                                            "pong" => {
758                                                events_clone.lock().unwrap().push_back(RelayClientEvent::PongReceived);
759                                            }
760                                            "error" => {
761                                                let msg = val.get("message")
762                                                    .or_else(|| val.get("reason"))
763                                                    .and_then(|v| v.as_str())
764                                                    .unwrap_or("relay error")
765                                                    .to_string();
766                                                if let Some(reply) = pending_create.take() {
767                                                    let _ = reply.send(Err(msg.clone()));
768                                                }
769                                                if let Some((code, reply)) = pending_join.take() {
770                                                    let _ = reply.send(Err(msg.clone()));
771                                                    events_clone.lock().unwrap().push_back(
772                                                        RelayClientEvent::RoomJoined {
773                                                            success: false,
774                                                            player_id: PlayerId(0),
775                                                            room_code: code,
776                                                        }
777                                                    );
778                                                }
779                                                if let Some(reply) = pending_rejoin.take() {
780                                                    let _ = reply.send(Err(msg.clone()));
781                                                }
782                                                events_clone.lock().unwrap().push_back(
783                                                    RelayClientEvent::Error { message: msg }
784                                                );
785                                            }
786                                            _ => {}
787                                        }
788                                    }
789                                    false
790                                }
791                                Some(Ok(Message::Pong(_))) => {
792                                    events_clone.lock().unwrap().push_back(RelayClientEvent::PongReceived);
793                                    false
794                                }
795                                Some(Ok(Message::Close(_))) | None | Some(Err(_)) => true,
796                                _ => false,
797                            }
798                        }
799                    };
800
801                    if !connection_dropped {
802                        continue;
803                    }
804
805                    connected_flag.store(false, Ordering::SeqCst);
806
807                    // Si no hay partida activa o el cierre fue voluntario, desconectar inmediatamente.
808                    let Some(target_room) = active_room_code.clone() else {
809                        *reconnect_state_clone.lock().unwrap() = ReconnectState::Disconnected;
810                        events_clone
811                            .lock()
812                            .unwrap()
813                            .push_back(RelayClientEvent::Disconnected);
814                        break;
815                    };
816                    if intentional_leave || shutdown_flag_clone.load(Ordering::SeqCst) {
817                        *reconnect_state_clone.lock().unwrap() = ReconnectState::Disconnected;
818                        events_clone
819                            .lock()
820                            .unwrap()
821                            .push_back(RelayClientEvent::Disconnected);
822                        break;
823                    }
824
825                    // Intentar reconexión automática con backoff exponencial (hasta max_attempts = 5)
826                    let mut restored = false;
827                    let max_attempts = policy.max_attempts.max(1);
828                    for attempt in 1..=max_attempts {
829                        if shutdown_flag_clone.load(Ordering::SeqCst) {
830                            break;
831                        }
832                        *reconnect_state_clone.lock().unwrap() = ReconnectState::Reconnecting {
833                            attempt,
834                            max_attempts,
835                        };
836                        events_clone
837                            .lock()
838                            .unwrap()
839                            .push_back(RelayClientEvent::Reconnecting {
840                                attempt,
841                                max_attempts,
842                            });
843
844                        let delay = policy.delay_for_attempt(attempt);
845                        tokio::time::sleep(delay).await;
846                        if shutdown_flag_clone.load(Ordering::SeqCst) {
847                            break;
848                        }
849
850                        let Ok(Ok((new_ws, _))) = tokio::time::timeout(
851                            Duration::from_millis(600),
852                            connect_async(&url_for_thread),
853                        )
854                        .await
855                        else {
856                            continue;
857                        };
858
859                        let (mut new_sink, mut new_source) = new_ws.split();
860
861                        if let Some(ref tok) = current_auth_token {
862                            let auth_msg = serde_json::json!({
863                                "op": "auth",
864                                "token": tok,
865                            })
866                            .to_string();
867                            if new_sink.send(Message::Text(auth_msg.into())).await.is_err() {
868                                continue;
869                            }
870                        }
871
872                        let rejoin_msg = match (&current_auth_token, active_player_id) {
873                            (None, pid) if pid > 0 => serde_json::json!({
874                                "op": "rejoin_room",
875                                "room_code": target_room,
876                                "player_id": format!("player{pid}"),
877                            }),
878                            _ => serde_json::json!({
879                                "op": "rejoin_room",
880                                "room_code": target_room,
881                            }),
882                        }
883                        .to_string();
884
885                        if new_sink
886                            .send(Message::Text(rejoin_msg.into()))
887                            .await
888                            .is_err()
889                        {
890                            continue;
891                        }
892
893                        // Esperar respuesta `rejoined` del servidor Relay
894                        let wait_deadline =
895                            tokio::time::Instant::now() + Duration::from_millis(800);
896                        let mut attempt_rejoined = None;
897                        while let Ok(Some(Ok(msg))) =
898                            tokio::time::timeout_at(wait_deadline, new_source.next()).await
899                        {
900                            if let Message::Text(text) = msg {
901                                if let Ok(val) = serde_json::from_str::<Value>(&text) {
902                                    let op = val.get("op").and_then(|v| v.as_str()).unwrap_or("");
903                                    if op == "rejoined" {
904                                        let pid = val
905                                            .get("player_id")
906                                            .and_then(|v| v.as_u64())
907                                            .map(|p| p as u32)
908                                            .unwrap_or_else(|| active_player_id.max(1));
909                                        let r_code = val
910                                            .get("room_code")
911                                            .and_then(|v| v.as_str())
912                                            .unwrap_or(&target_room)
913                                            .to_string();
914                                        let seed =
915                                            val.get("seed").and_then(|v| v.as_u64()).unwrap_or(1);
916                                        let commands = val
917                                            .get("commands")
918                                            .and_then(|v| v.as_array())
919                                            .cloned()
920                                            .unwrap_or_default();
921                                        attempt_rejoined = Some((r_code, pid, seed, commands));
922                                        break;
923                                    } else if op == "error" {
924                                        break;
925                                    }
926                                }
927                            } else if matches!(msg, Message::Close(_)) {
928                                break;
929                            }
930                        }
931
932                        if let Some((r_code, pid, seed, commands)) = attempt_rejoined {
933                            ws_sink = new_sink;
934                            ws_source = new_source;
935                            active_room_code = Some(r_code.clone());
936                            active_player_id = pid;
937                            connected_flag.store(true, Ordering::SeqCst);
938                            *reconnect_state_clone.lock().unwrap() = ReconnectState::Connected;
939                            events_clone.lock().unwrap().push_back(
940                                RelayClientEvent::ConnectionRestored {
941                                    room_code: r_code,
942                                    player_id: PlayerId(pid),
943                                    seed,
944                                    commands,
945                                },
946                            );
947                            heartbeat.reset();
948                            restored = true;
949                            break;
950                        }
951                    }
952
953                    if !restored {
954                        connected_flag.store(false, Ordering::SeqCst);
955                        *reconnect_state_clone.lock().unwrap() = ReconnectState::Disconnected;
956                        events_clone
957                            .lock()
958                            .unwrap()
959                            .push_back(RelayClientEvent::Disconnected);
960                        break;
961                    }
962                }
963            });
964        });
965
966        let _ = conn_init_rx.blocking_recv();
967
968        Self {
969            client_id,
970            relay_url: url,
971            cmd_tx: Some(cmd_tx),
972            events,
973            incoming_payloads,
974            connected_remote,
975            reconnect_state,
976            shutdown_flag,
977            current_room: None,
978            player_id: PlayerId(0),
979            _worker: Some(worker),
980        }
981    }

1021    /// Solicita reconectar explícitamente al asiento previo en la sala `code` enviando `{"op": "rejoin_room"}`.
1022    pub fn rejoin_room(&mut self, code: &str) -> EngineResult<PlayerId> {
1023        let normalized = code.trim().to_ascii_uppercase();
1024        if !is_valid_relay_code(&normalized) {
1025            return Err(EngineError::Network("Código de sala inválido".to_string()));
1026        }
1027        if self.is_remote_connected() {
1028            if let Some(tx) = &self.cmd_tx {
1029                let (reply_tx, reply_rx) = oneshot::channel();
1030                let pid_hint = if self.player_id.0 > 0 {
1031                    Some(format!("player{}", self.player_id.0))
1032                } else {
1033                    None
1034                };
1035                if tx
1036                    .send(OutboundRelayCommand::Rejoin {
1037                        room_code: normalized.clone(),
1038                        player_id: pid_hint,
1039                        reply: reply_tx,
1040                    })
1041                    .is_ok()
1042                {
1043                    match reply_rx.blocking_recv() {
1044                        Ok(Ok((pid, _seed, _commands))) => {
1045                            let player = PlayerId(pid);
1046                            self.current_room = Some(normalized);
1047                            self.player_id = player;
1048                            return Ok(player);
1049                        }
1050                        Ok(Err(err)) => {
1051                            if self.current_room.as_deref() == Some(normalized.as_str())
1052                                && self.player_id.0 > 0
1053                            {
1054                                return Ok(self.player_id);
1055                            }
1056                            return Err(EngineError::Network(err));
1057                        }
1058                        Err(e) => return Err(EngineError::Network(e.to_string())),
1059                    }
1060                }
1061            }
1062        }
1063
1064        // Fallback en hub local en memoria
1065        let mut hub = local_relay_hub().lock().unwrap();
1066        if let Some(room) = hub.get_mut(&normalized) {
1067            let assigned_pid = if self.player_id.0 > 0 {
1068                self.player_id
1069            } else {
1070                let pid = PlayerId(room.next_player_id);
1071                room.next_player_id += 1;
1072                pid
1073            };
1074            if !room
1075                .members
1076                .iter()
1077                .any(|(cid, _, _)| *cid == self.client_id)
1078            {
1079                room.members
1080                    .push((self.client_id, assigned_pid, Arc::clone(&self.events)));
1081            }
1082            drop(hub);
1083
1084            self.current_room = Some(normalized.clone());
1085            self.player_id = assigned_pid;
1086            self.events
1087                .lock()
1088                .unwrap()
1089                .push_back(RelayClientEvent::RoomJoined {
1090                    success: true,
1091                    player_id: assigned_pid,
1092                    room_code: normalized,
1093                });
1094            return Ok(assigned_pid);
1095        }
1096        drop(hub);
1097
1098        Err(EngineError::Network(
1099            "No conectado a un servidor Relay remoto".to_string(),
1100        ))
1101    }

1171    /// Solicita unirse a la sala con código `code` y devuelve el `PlayerId` asignado.
1172    pub fn join_room(&mut self, code: &str) -> EngineResult<PlayerId> {
1173        let normalized = code.trim().to_ascii_uppercase();
1174        if !is_valid_relay_code(&normalized) {
1175            self.events
1176                .lock()
1177                .unwrap()
1178                .push_back(RelayClientEvent::RoomJoined {
1179                    success: false,
1180                    player_id: PlayerId(0),
1181                    room_code: normalized,
1182                });
1183            return Err(EngineError::Network("Código de sala inválido".to_string()));
1184        }
1185
1186        if self.is_remote_connected() {
1187            if let Some(tx) = &self.cmd_tx {
1188                let (reply_tx, reply_rx) = oneshot::channel();
1189                if tx
1190                    .send(OutboundRelayCommand::Join {
1191                        room_code: normalized.clone(),
1192                        reply: reply_tx,
1193                    })
1194                    .is_ok()
1195                {
1196                    match reply_rx.blocking_recv() {
1197                        Ok(Ok(pid)) => {
1198                            let player = PlayerId(pid);
1199                            self.current_room = Some(normalized);
1200                            self.player_id = player;
1201                            return Ok(player);
1202                        }
1203                        Ok(Err(err)) => {
1204                            return Err(EngineError::Network(err));
1205                        }
1206                        Err(_) => {}
1207                    }
1208                }
1209            }
1210        }
1211
1212        // Fallback en hub local en memoria
1213        self.leave_local_hub();
1214        let mut hub = local_relay_hub().lock().unwrap();
1215        let Some(room) = hub.get_mut(&normalized) else {
1216            drop(hub);
1217            self.events
1218                .lock()
1219                .unwrap()
1220                .push_back(RelayClientEvent::RoomJoined {
1221                    success: false,
1222                    player_id: PlayerId(0),
1223                    room_code: normalized,
1224                });
1225            return Err(EngineError::Network("Sala no encontrada".to_string()));
1226        };
1227
1228        let assigned_pid = PlayerId(room.next_player_id);
1229        room.next_player_id += 1;
1230        for (_, _, member_events) in &room.members {
1231            member_events
1232                .lock()
1233                .unwrap()
1234                .push_back(RelayClientEvent::PeerJoined {
1235                    player_id: assigned_pid,
1236                });
1237        }
1238        room.members
1239            .push((self.client_id, assigned_pid, Arc::clone(&self.events)));
1240        drop(hub);
1241
1242        self.current_room = Some(normalized.clone());
1243        self.player_id = assigned_pid;
1244        self.events

1248                success: true,
1249                player_id: assigned_pid,

1255    /// Envía datos a los demás participantes de la sala actual a través del Relay.
1256    pub fn send_relay_data(&mut self, bytes: &[u8]) -> EngineResult<()> {
1257        if self.is_remote_connected() {
1258            if let Some(tx) = &self.cmd_tx {
1259                let data_val = encode_bytes_to_value(bytes);
1260                return tx
1261                    .send(OutboundRelayCommand::Relay { data: data_val })
1262                    .map_err(|e| EngineError::Network(e.to_string()));
1263            }
1264        }
1265
1266        let Some(code) = &self.current_room else {
1267            return Err(EngineError::Network(
1268                "El cliente no está en una sala".into(),
1269            ));
1270        };
1271        let hub = local_relay_hub().lock().unwrap();
1272        if let Some(room) = hub.get(code) {
1273            for (cid, _, member_events) in &room.members {
1274                if *cid != self.client_id {
1275                    member_events
1276                        .lock()
1277                        .unwrap()
1278                        .push_back(RelayClientEvent::PayloadReceived {
1279                            from: self.player_id,
1280                            data: bytes.to_vec(),
1281                        });
1282                }
1283            }
1284            Ok(())
1285        } else {
1286            Err(EngineError::Network("Sala no encontrada".into()))
1287        }
1288    }

1290    /// Abandona la sala actual y notifica a los participantes restantes.
1291    pub fn leave_room(&mut self) -> EngineResult<()> {
1292        if self.is_remote_connected() {
1293            if let Some(tx) = &self.cmd_tx {
1294                let _ = tx.send(OutboundRelayCommand::Leave);
1295            }
1296        }
1297        self.leave_local_hub();
1298        self.current_room = None;
1299        self.player_id = PlayerId(0);
1300        Ok(())
1301    }

1303    fn leave_local_hub(&mut self) {
1304        let Some(code) = self.current_room.take() else {
1305            return;
1306        };
1307        let mut hub = local_relay_hub().lock().unwrap();
1308        let mut remove_room = false;
1309        if let Some(room) = hub.get_mut(&code) {
1310            room.members.retain(|(cid, _, _)| *cid != self.client_id);
1311            for (_, _, member_events) in &room.members {
1312                member_events
1313                    .lock()
1314                    .unwrap()
1315                    .push_back(RelayClientEvent::PeerLeft {
1316                        player_id: self.player_id,
1317                    });
1318            }
1319            remove_room = room.members.is_empty();
1320        }
1321        if remove_room {
1322            hub.remove(&code);
1323        }
1324    }

1327    pub fn poll_relay_events(&mut self) -> Vec<RelayClientEvent> {
1328        let mut q = self.events.lock().unwrap();
1329        q.drain(..).collect()
1330    }

1349    fn poll_received(&mut self) -> Vec<(PlayerId, Vec<u8>)> {
1350        let mut out: Vec<(PlayerId, Vec<u8>)> =
1351            self.incoming_payloads.lock().unwrap().drain(..).collect();
1352        let mut ev_queue = self.events.lock().unwrap();
1353        let mut remaining = VecDeque::new();
1354        while let Some(ev) = ev_queue.pop_front() {
1355            if let RelayClientEvent::PayloadReceived { from, data } = ev {
1356                out.push((from, data));
1357            } else {
1358                remaining.push_back(ev);
1359            }
1360        }
1361        *ev_queue = remaining;
1362        out
1363    }
