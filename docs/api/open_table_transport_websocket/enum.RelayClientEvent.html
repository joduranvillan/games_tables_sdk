pub enum RelayClientEvent {
    RoomCreated {
        room_code: String,
    },
    RoomJoined {
        success: bool,
        player_id: PlayerId,
        room_code: String,
    },
    PeerJoined {
        player_id: PlayerId,
    },
    PeerLeft {
        player_id: PlayerId,
    },
    PayloadReceived {
        from: PlayerId,
        data: Vec<u8>,
    },
    PongReceived,
    Reconnecting {
        attempt: u32,
        max_attempts: u32,
    },
    ConnectionRestored {
        room_code: String,
        player_id: PlayerId,
        seed: u64,
        commands: Vec<Value>,
    },
    Disconnected,
    Error {
        message: String,
    },
}
